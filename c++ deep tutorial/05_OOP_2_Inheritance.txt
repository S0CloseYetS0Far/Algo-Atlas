I. Basic Inheritance Syntax: Starting with Shape Classes
------------------------------------

Inheritance is one of the three core mechanisms of object-oriented programming. Through inheritance, a
derived class can reuse the base class's member variables and member functions, and add new functionality on top of them.

[Example 1-1] Base class Shape and derived class Circle:

```cpp
#include <iostream>
#include <string>
#include <cmath>

// Base class
class Shape {
protected:               // protected members: accessible to derived classes, not to outside code
    double x, y;         // coordinates
    std::string color;

public:
    Shape(double xVal, double yVal, const std::string& c)
        : x(xVal), y(yVal), color(c) {
        std::cout << "[Shape] constructed at (" << x << ", " << y << ") color: " << color << std::endl;
    }

    ~Shape() {
        std::cout << "[Shape] destroyed" << std::endl;
    }

    void move(double dx, double dy) {
        x += dx;
        y += dy;
        std::cout << "[Shape] moved to (" << x << ", " << y << ")" << std::endl;
    }

    // const member functions
    double getX() const { return x; }
    double getY() const { return y; }
    const std::string& getColor() const { return color; }

    void setColor(const std::string& c) { color = c; }
};

// Derived class: public inheritance
class Circle : public Shape {
private:
    double radius;

public:
    Circle(double xVal, double yVal, const std::string& c, double r)
        : Shape(xVal, yVal, c),   // must call the base class constructor
          radius(r) {
        std::cout << "[Circle] constructed radius: " << radius << std::endl;
    }

    ~Circle() {
        std::cout << "[Circle] destroyed radius: " << radius << std::endl;
    }

    double area() const {
        return M_PI * radius * radius;
    }

    double perimeter() const {
        return 2 * M_PI * radius;
    }

    double getRadius() const { return radius; }

    // Redefines the base class's move function (not an override, since there is no virtual)
    void move(double dx, double dy) {
        // Can call the base class's move
        Shape::move(dx, dy);  // call the base class version
        std::cout << "[Circle] also moved radius: " << radius << std::endl;
    }

    void print() const {
        std::cout << "Circle{" << color << "} center(" << x << "," << y
                  << ") radius " << radius
                  << " area " << area() << std::endl;
    }
};

// Another derived class
class Rectangle : public Shape {
private:
    double width, height;

public:
    Rectangle(double xVal, double yVal, const std::string& c,
              double w, double h)
        : Shape(xVal, yVal, c), width(w), height(h) {
        std::cout << "[Rectangle] constructed " << width << "x" << height << std::endl;
    }

    ~Rectangle() {
        std::cout << "[Rectangle] destroyed" << std::endl;
    }

    double area() const { return width * height; }
    double getWidth() const { return width; }
    double getHeight() const { return height; }
};

int main() {
    std::cout << "===== Construction phase =====" << std::endl;
    Circle c(100, 200, "red", 50);
    std::cout << std::endl;

    Rectangle r(0, 0, "blue", 100, 60);
    std::cout << std::endl;

    // Access base class members
    c.print();
    std::cout << "Circle x coordinate: " << c.getX() << std::endl;

    // Call the base class's move
    c.move(10, 20);

    // Call derived class methods
    std::cout << "Circle area: " << c.area() << std::endl;
    std::cout << "Rectangle area: " << r.area() << std::endl;

    std::cout << "\n===== Destruction phase =====" << std::endl;
    return 0;
    // Destruction order: Rectangle -> Shape, then Circle -> Shape
}
/* Output order (shows the construction chain and destruction chain):
===== Construction phase =====
[Shape] constructed at (100, 200) color: red
[Circle] constructed radius: 50

[Shape] constructed at (0, 0) color: blue
[Rectangle] constructed 100x60

===== Destruction phase =====
[Rectangle] destroyed         <- derived class destroyed first
[Shape] destroyed             <- base class destroyed afterwards
[Circle] destroyed            <- derived class destroyed first
[Shape] destroyed             <- base class destroyed afterwards
*/
```

II. Low-Level Computer Principles: Memory Layout of Inheritance
------------------------------------

[Principle 2-1] Memory layout of a derived-class object

A derived-class object contains a complete base class subobject in memory,
laid out contiguously with the members added by the derived class.

Take the following two classes as an example:

```cpp
class Base {
public:
    int baseVal;        // 4 bytes
    char baseChar;      // 1 byte
    // [padding: 3 bytes]
    // Base class subobject size: 8 bytes
};

class Derived : public Base {
public:
    int derivedVal;     // 4 bytes
    double derivedDbl;  // 8 bytes
    // [padding: 4 bytes] final alignment to an 8-byte boundary
};
```

Memory layout of a Derived object (64-bit system):

  Offset    Content                  Belongs to
  ─────────────────────────────────────────
  0x00    baseVal (4 bytes)       <- base class subobject starts
  0x04    baseChar (1 byte)
  0x05    [padding: 3 bytes]
  0x08    derivedVal (4 bytes)    <- derived class members start
  0x0C    [padding: 4 bytes]
  0x10    derivedDbl (8 bytes)
  ─────────────────────────────────────────
  Total size: 24 bytes = sizeof(Base)=8 + padding=4 + derivedVal=4 + padding=4 +
                     derivedDbl=8
  The base class part comes first, the derived class part comes after!

Verify with code:

```cpp
#include <iostream>

class Base {
public:
    int baseVal;
    char baseChar;
};

class Derived : public Base {
public:
    int derivedVal;
    double derivedDbl;
};

int main() {
    std::cout << "sizeof(Base): " << sizeof(Base) << std::endl;       // 8
    std::cout << "sizeof(Derived): " << sizeof(Derived) << std::endl; // 24

    Derived d;
    // The address of object d and the address of its base class subobject are the same
    std::cout << "&d: " << &d << std::endl;
    std::cout << "&d.baseVal: " << &d.baseVal << std::endl;
    // They are equal! This proves the base class subobject sits at the start of the derived object

    return 0;
}
```

[Principle 2-2] Upcasting -- implicit conversion of pointers/references

Because the base class subobject sits at the start of the derived object, converting a derived-class pointer
to a base-class pointer costs nothing at the address level:

```cpp
class Animal {
public:
    int age;  // 0x00
};

class Dog : public Animal {
public:
    int breedID;   // 0x04
    double weight; // 0x08
};

// Upcasting:
Dog dog;
dog.age = 5;
dog.breedID = 10;

Animal* animalPtr = &dog;  // implicit upcast
// animalPtr and &dog point to the same memory address (same address value)
// but the static type of animalPtr is Animal*, so it can only access Animal's members

// Under the hood:
// dog object memory layout: [age=4B][padding=4B][breedID=4B][weight=8B]
// animalPtr points to the start address, which is the same as pointing to the base class subobject
// The compiler restricts access to the first 8 bytes (the Animal range)

// Safety rules:
// Upcasting is always safe -- every Dog is an Animal
// Downcasting requires dynamic_cast (Chapter 6) -- an Animal is not necessarily a Dog
```

[Principle 2-3] The construction chain and the destruction chain

```cpp
#include <iostream>

class GrandParent {
public:
    GrandParent()  { std::cout << "GrandParent constructed" << std::endl; }
    ~GrandParent() { std::cout << "GrandParent destroyed" << std::endl; }
};

class Parent : public GrandParent {
public:
    Parent()  { std::cout << "  Parent constructed" << std::endl; }
    ~Parent() { std::cout << "  Parent destroyed" << std::endl; }
};

class Child : public Parent {
public:
    Child()  { std::cout << "    Child constructed" << std::endl; }
    ~Child() { std::cout << "    Child destroyed" << std::endl; }
};

int main() {
    Child c;
    return 0;
}
/* Output:
GrandParent constructed   <- the most distant ancestor is constructed first
  Parent constructed      <- then the direct base class
    Child constructed     <- the derived class is constructed last
    Child destroyed       <- the derived class is destroyed first
  Parent destroyed        <- then the direct base class
GrandParent destroyed     <- the most distant ancestor last
*/

// Rules:
// Construction order: base -> derived (from ancestor to descendant)
// Destruction order: derived -> base (from descendant to ancestor, exactly reversed)
// It's like building a house: lay the foundation first (base class), then build the upper floors (derived class)
// Demolishing the house: tear down the upper floors first, then the foundation
```

The three phases of the construction chain (when each class's constructor runs):
  1. The base class constructor runs (initializing the base class subobject)
  2. Member variables are initialized (initializer list executed in declaration order)
  3. The constructor body runs

[Principle 2-4] How different inheritance modes affect accessibility

C++ provides three inheritance modes: public, protected, and private. They affect the access level of
base class members inside the derived class.

  Inheritance │ Base public members in derived │ Base protected members in derived │ Base private members in derived
  ─────────┼───────────────────────┼───────────────────────────┼─────────────────────────
  public    │ stay public           │ stay protected              │ inaccessible
  protected │ become protected      │ stay protected              │ inaccessible
  private   │ become private        │ become private              │ inaccessible

```cpp
class Base {
public:
    int pub;
protected:
    int prot;
private:
    int priv;  // can never be accessed directly by a derived class
};

class PublicDerived : public Base {
    // pub  is still public
    // prot is still protected
    // priv is inaccessible
public:
    void test() {
        pub = 10;   // OK, still public
        prot = 20;  // OK, still protected
        // priv = 30;  // Error! private members are inaccessible
    }
};

class ProtectedDerived : protected Base {
    // pub  becomes protected
    // prot becomes protected
    // priv is inaccessible
public:
    void test() {
        pub = 10;   // OK, now protected
        prot = 20;  // OK, still protected
    }
};

class PrivateDerived : private Base {
    // pub  becomes private
    // prot becomes private
    // priv is inaccessible
public:
    void test() {
        pub = 10;   // OK, now private
        prot = 20;  // OK, now private
    }
};
```

Guide to choosing an inheritance mode:
  public inheritance    -- an "is-a" relationship, the most common (Dog is an Animal)
  private inheritance   -- "implemented-in-terms-of" (implementing the derived class using the base class; an alternative to composition)
  protected inheritance -- rarely used, unless you really need to control access for descendants

[Principle 2-5] Redefining vs Overriding

These are two concepts that are easily confused:

```cpp
class Base {
public:
    void show() { std::cout << "Base::show()" << std::endl; }
    virtual void display() { std::cout << "Base::display()" << std::endl; }
};

class Derived : public Base {
public:
    // Redefining -- non-virtual function
    void show() { std::cout << "Derived::show()" << std::endl; }
    // Overriding -- virtual function, covered in detail in Chapter 6
    void display() override { std::cout << "Derived::display()" << std::endl; }
};

int main() {
    Derived d;
    d.show();           // calls Derived::show() (static binding)
    d.Base::show();     // explicitly calls Base::show()

    Base* ptr = &d;
    ptr->show();        // calls Base::show()! because show is not virtual
    ptr->display();     // calls Derived::display()! because display is virtual (dynamic binding)
    return 0;
}
```

The problem with redefinition:
  Defining a non-virtual function in the derived class with the same name as one in the base class hides ALL base class functions with that name (including overloads)
  -> this is called "Name Hiding"

2.1 Inheritance Syntax and the Three Inheritance Modes
------------------------------------

```cpp
// Base class
class Vehicle {
public:
    void start() { std::cout << "Vehicle started" << std::endl; }
protected:
    int speed;
private:
    std::string vin;  // vehicle identification number
};

// Public inheritance (most common)
class Car : public Vehicle {
    // start()  -> public
    // speed    -> protected
    // vin      -> inaccessible
};

// Protected inheritance
class Boat : protected Vehicle {
    // start()  -> protected
    // speed    -> protected
    // vin      -> inaccessible
};

// Private inheritance
class Bicycle : private Vehicle {
    // start()  -> private
    // speed    -> private
    // vin      -> inaccessible
};
```

2.2 Derived Class Constructors -- Calling the Base Class Constructor
------------------------------------

```cpp
#include <iostream>
#include <string>

class Person {
protected:
    std::string name;
    int age;
public:
    Person(const std::string& n, int a) : name(n), age(a) {
        std::cout << "Person(" << name << ", " << age << ") constructed\n";
    }
};

class Teacher : public Person {
private:
    std::string subject;
    int experience;
public:
    // The derived class constructor must call the base class constructor
    Teacher(const std::string& n, int a,
            const std::string& sub, int exp)
        : Person(n, a),      // call the base class constructor
          subject(sub),
          experience(exp) {
        std::cout << "Teacher(" << name << ", "
                  << subject << ") constructed\n";
    }

    // You can explicitly call multiple base class constructors (with multiple inheritance)
    // You can also delegate to another constructor of the same class
    Teacher(const std::string& n, int a, const std::string& sub)
        : Teacher(n, a, sub, 0) { }  // delegating constructor (C++11)
};

class Professor : public Teacher {
private:
    bool tenured;
public:
    Professor(const std::string& n, int a, const std::string& sub,
              int exp, bool t)
        : Teacher(n, a, sub, exp), tenured(t) {
        std::cout << "Professor constructed\n";
    }
};
```

Initialization order (cannot be changed):
  1. Virtual base classes (if any; covered in detail in Section 3) -- depth-first, left to right
  2. Direct non-virtual base classes -- in the order they are declared in the inheritance list
  3. This class's member variables -- in declaration order
  4. The constructor body

2.3 Destructors in Inheritance
------------------------------------

```cpp
#include <iostream>

class Base {
public:
    Base()  { std::cout << "Base::Base()\n"; }
    ~Base() { std::cout << "Base::~Base()\n"; }
};

class Derived : public Base {
private:
    int* dynamicData;
public:
    Derived() : dynamicData(new int[100]) {
        std::cout << "Derived::Derived()\n";
    }

    // The destructor automatically runs the derived destructor body first, then calls the base destructor
    ~Derived() {
        delete[] dynamicData;
        std::cout << "Derived::~Derived()\n";
    }
};

// Important: if a base class pointer points to a derived object and you delete through the base pointer:
// Base* ptr = new Derived();
// delete ptr;  // if ~Base() is not a virtual destructor, only ~Base() is called
//              // causing dynamicData to leak!
// Solution -> Chapter 6: virtual destructors
```

Destruction order (guaranteed):
  1. This class's destructor body runs
  2. This class's member variables are destroyed in reverse declaration order
  3. Direct non-virtual base classes are destroyed in reverse declaration order
  4. Virtual base classes are destroyed

2.4 Redefinition and Name Hiding
------------------------------------

```cpp
class Base {
public:
    void func(int x)       { std::cout << "Base::func(int)\n"; }
    void func(double x)    { std::cout << "Base::func(double)\n"; }
    void func(int x, int y){ std::cout << "Base::func(int,int)\n"; }
};

class Derived : public Base {
public:
    // Redefining just one version hides ALL of the base class's func overloads!
    void func(int x) { std::cout << "Derived::func(int)\n"; }
};

int main() {
    Derived d;
    d.func(10);           // OK: calls Derived::func(int)
    // d.func(3.14);      // Error! Base::func(double) is hidden!
    // d.func(10, 20);    // Error! Base::func(int,int) is hidden!

    d.Base::func(3.14);   // the base class version can be called explicitly
    d.Base::func(10, 20); // the base class version can be called explicitly
    return 0;
}
```

2.5 Changing Access Levels in Inheritance with using Declarations
------------------------------------

```cpp
class Base {
protected:
    void helperFunc() { std::cout << "Protected helper\n"; }
    int value;
public:
    void publicFunc() { std::cout << "Public function\n"; }
    void overloaded(int x) { std::cout << "overloaded(int)\n"; }
    void overloaded(double x) { std::cout << "overloaded(double)\n"; }
};

class Derived : private Base {  // private inheritance: all base members become private
public:
    // using declarations raise the access level
    using Base::publicFunc;     // publicFunc becomes public in Derived
    using Base::value;          // value becomes public in Derived

    // A using declaration solves name hiding -- it imports all overloads from the base class
    using Base::overloaded;

    void overloaded(int x) {   // redefine one version
        std::cout << "Derived::overloaded(int)\n";
    }
    // Now both Derived's version and the base class's double version can be used
};

int main() {
    Derived d;
    d.publicFunc();      // OK (the using declaration raised the access level)
    d.value = 42;        // OK (the using declaration raised the access level)
    d.overloaded(10);    // Derived::overloaded(int)
    d.overloaded(3.14);  // Base::overloaded(double) -- made visible by the using declaration!
    return 0;
}
```

2.6 Multiple Inheritance and the Diamond Inheritance Problem
------------------------------------

[Multiple Inheritance]

```cpp
#include <iostream>

class Flyable {
public:
    void fly() { std::cout << "Flying" << std::endl; }
};

class Swimmable {
public:
    void swim() { std::cout << "Swimming" << std::endl; }
};

// Multiple inheritance
class Duck : public Flyable, public Swimmable {
public:
    void quack() { std::cout << "Quack!" << std::endl; }
};

int main() {
    Duck d;
    d.fly();    // inherited from Flyable
    d.swim();   // inherited from Swimmable
    d.quack();  // Duck's own method
    return 0;
}
```

[The Diamond Inheritance Problem]

```cpp
class Animal {
public:
    int age;
    Animal() : age(0) { std::cout << "Animal()\n"; }
};

class Mammal : public Animal {
    // contains one Animal subobject
};

class Bird : public Animal {
    // contains another Animal subobject!
};

// Diamond inheritance: Bat inherits Animal through both Mammal and Bird
//          Animal
//         /      \
//     Mammal    Bird
//         \      /
//           Bat
class Bat : public Mammal, public Bird {
public:
    // Bat has two age members! Mammal::age and Bird::age
    void printAge() {
        // std::cout << age;     // Error! Ambiguous -- which age?
        std::cout << Mammal::age << std::endl;  // the path must be specified explicitly
    }
};
```

[Virtual Inheritance -- Solving Diamond Inheritance]

```cpp
class Animal {
public:
    int age;
    Animal() : age(0) { std::cout << "Animal()\n"; }
};

// Virtual inheritance: guarantees only one Animal subobject in the most derived class
class Mammal : virtual public Animal {
    // ...
};

class Bird : virtual public Animal {
    // ...
};

// Now Bat has only one Animal subobject!
class Bat : public Mammal, public Bird {
public:
    void printAge() {
        std::cout << age << std::endl;  // OK! There is only one age
    }
};

// How virtual inheritance changes the memory layout:
// Normal inheritance: the base class subobject is embedded in the derived class
// Virtual inheritance: the derived class stores a pointer (vbptr) to the shared base class subobject,
//         and the base class subobject is placed at the end of the object
// This adds the overhead of indirect access, so use virtual inheritance only when necessary
```

Things to note about virtual inheritance:
  1. The virtual base class's constructor is called by the most derived class
  2. Virtual base classes are initialized before any non-virtual base classes
  3. Virtual inheritance increases object size (an extra vbptr pointer)

2.7 The final Keyword Prevents Inheritance (C++11)
------------------------------------

```cpp
// A final class -- cannot be inherited from
class Singleton final {
public:
    static Singleton& getInstance() {
        static Singleton instance;
        return instance;
    }
private:
    Singleton() = default;
};

// class DerivedSingleton : public Singleton {};  // Error! Cannot inherit from a final class

// final can also be used on virtual functions (see Chapter 6)
class Base {
public:
    virtual void doSomething() = 0;  // pure virtual function
    virtual void cannotOverride() final { }  // a virtual function that cannot be overridden
};

class Derived : public Base {
public:
    void doSomething() override { }  // OK
    // void cannotOverride() override { }  // Error! A final function cannot be overridden
};
```

Case 1: Graphics Drawing System Basics (Shape/Circle/Rectangle)
------------------------------------

[Real-life scenario] The shape-management layer of drawing software (such as Windows Paint or Photoshop).
The base class defines common attributes (position, color), and derived classes define concrete shapes.

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <memory>
#include <cmath>
#include <iomanip>

// ====== Debug macro (Chapter 2 knowledge) ======
#ifdef DRAW_DEBUG
    #define DRAW_LOG(msg) std::cout << "[DRAW] " << msg << std::endl
#else
    #define DRAW_LOG(msg)
#endif

namespace graphics {

// ====== Base class: Shape ======
class Shape {
protected:
    double x, y;           // shape coordinates (top-left corner or center)
    std::string color;     // color
    int lineWidth;         // line width
    static int shapeCount; // total number of shapes

public:
    Shape(double xVal, double yVal, const std::string& c, int lw = 1)
        : x(xVal), y(yVal), color(c), lineWidth(lw) {
        shapeCount++;
        DRAW_LOG("Shape created: (" + std::to_string(x) + "," +
                 std::to_string(y) + ") " + c);
    }

    Shape(const Shape& other)
        : x(other.x), y(other.y), color(other.color),
          lineWidth(other.lineWidth) {
        shapeCount++;
        DRAW_LOG("Shape copied");
    }

    virtual ~Shape() {
        shapeCount--;
        DRAW_LOG("Shape destroyed");
    }

    // ====== Basic operations ======
    void move(double dx, double dy) {
        x += dx;
        y += dy;
    }

    void setColor(const std::string& c) { color = c; }
    void setLineWidth(int lw) { lineWidth = lw; }

    // const member functions
    double getX() const { return x; }
    double getY() const { return y; }
    const std::string& getColor() const { return color; }
    int getLineWidth() const { return lineWidth; }

    virtual double area() const = 0;      // pure virtual function -- covered in detail in Chapter 6
    virtual double perimeter() const = 0; // pure virtual function
    virtual void draw() const = 0;        // pure virtual function

    virtual void printInfo() const {
        std::cout << "Shape{" << color << "} at (" << x << "," << y
                  << ") lineWidth=" << lineWidth;
    }

    static int getShapeCount() { return shapeCount; }
};

int Shape::shapeCount = 0;

// ====== Derived class: Circle ======
class Circle : public Shape {
private:
    double radius;

public:
    Circle(double xVal, double yVal, const std::string& c, double r, int lw = 1)
        : Shape(xVal, yVal, c, lw), radius(r) {
        DRAW_LOG("Circle: radius=" + std::to_string(radius));
    }

    double area() const override {
        return M_PI * radius * radius;
    }

    double perimeter() const override {
        return 2 * M_PI * radius;
    }

    void draw() const override {
        std::cout << "Drawing circle: center(" << x << "," << y
                  << ") radius " << radius << " color " << color
                  << " area " << std::fixed << std::setprecision(2) << area()
                  << std::endl;
    }

    void printInfo() const override {
        Shape::printInfo();
        std::cout << " type=Circle radius=" << radius
                  << " area=" << area() << std::endl;
    }

    double getRadius() const { return radius; }
    void setRadius(double r) { radius = r; }
};

// ====== Derived class: Rectangle ======
class Rectangle : public Shape {
private:
    double width, height;

public:
    Rectangle(double xVal, double yVal, const std::string& c,
              double w, double h, int lw = 1)
        : Shape(xVal, yVal, c, lw), width(w), height(h) {
        DRAW_LOG("Rectangle: " + std::to_string(width) + "x" +
                 std::to_string(height));
    }

    double area() const override {
        return width * height;
    }

    double perimeter() const override {
        return 2 * (width + height);
    }

    void draw() const override {
        std::cout << "Drawing rectangle: top-left(" << x << "," << y
                  << ") width " << width << " height " << height
                  << " color " << color << " area " << area() << std::endl;
    }

    void printInfo() const override {
        Shape::printInfo();
        std::cout << " type=Rectangle " << width << "x" << height
                  << " area=" << area() << std::endl;
    }

    double getWidth() const { return width; }
    double getHeight() const { return height; }
};

// ====== Derived class: Triangle ======
class Triangle : public Shape {
private:
    double base, triHeight;  // base length and height

public:
    Triangle(double xVal, double yVal, const std::string& c,
             double b, double h, int lw = 1)
        : Shape(xVal, yVal, c, lw), base(b), triHeight(h) {
        DRAW_LOG("Triangle: base=" + std::to_string(base) +
                 " height=" + std::to_string(triHeight));
    }

    double area() const override {
        return 0.5 * base * triHeight;
    }

    double perimeter() const override {
        // Simplified: isosceles triangle approximation
        double halfBase = base / 2.0;
        double side = std::sqrt(halfBase * halfBase +
                                triHeight * triHeight);
        return base + 2 * side;
    }

    void draw() const override {
        std::cout << "Drawing triangle: apex(" << x << "," << y
                  << ") base " << base << " height " << triHeight
                  << " color " << color << " area " << area() << std::endl;
    }

    void printInfo() const override {
        Shape::printInfo();
        std::cout << " type=Triangle base=" << base
                  << " height=" << triHeight << " area=" << area()
                  << std::endl;
    }
};

}  // namespace graphics
```

Case 3: Company Payroll Management System (Employee Inheritance Hierarchy)
------------------------------------

[Real-life scenario] A company's HR department needs to manage salary calculations for all kinds of employees.
Different positions (manager, salesperson, intern) use different salary formulas.
Inheritance is used for code reuse, laying the groundwork for polymorphism.

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <memory>
#include <iomanip>

// ====== Debug macro ======
#define HR_DEBUG 1
#if HR_DEBUG
    #define HR_LOG(msg) std::cout << "[HR] " << msg << std::endl
#else
    #define HR_LOG(msg)
#endif

namespace company {

// ====== Base class: Employee ======
class Employee {
protected:
    std::string name;
    std::string employeeID;
    double baseSalary;          // base salary
    int yearsOfService;         // years of service

    static int totalEmployees;

public:
    Employee(const std::string& n, const std::string& id,
             double base, int years)
        : name(n), employeeID(id), baseSalary(base),
          yearsOfService(years) {
        totalEmployees++;
        HR_LOG("Employee hired: " + name + " (" + id + ")");
    }

    virtual ~Employee() {
        totalEmployees--;
        HR_LOG("Employee left: " + name);
    }

    // ====== Accessors ======
    const std::string& getName()       const { return name; }
    const std::string& getID()         const { return employeeID; }
    double getBaseSalary()             const { return baseSalary; }
    int getYearsOfService()            const { return yearsOfService; }

    // Calculate salary -- each derived class has its own implementation
    virtual double calculateSalary() const {
        // Base salary + seniority allowance (2% increase per year)
        return baseSalary * (1.0 + yearsOfService * 0.02);
    }

    virtual std::string getRole() const = 0;  // pure virtual function

    virtual void display() const {
        std::cout << std::left
                  << std::setw(15) << name
                  << std::setw(10) << employeeID
                  << std::setw(12) << getRole()
                  << "$" << std::fixed << std::setprecision(2)
                  << std::setw(10) << calculateSalary();
    }

    static int getTotalEmployees() { return totalEmployees; }
};

int Employee::totalEmployees = 0;

// ====== Derived class: Manager ======
class Manager : public Employee {
private:
    double managementAllowance;   // management allowance
    int teamSize;                 // number of people managed

public:
    Manager(const std::string& n, const std::string& id,
            double base, int years, double allowance, int team)
        : Employee(n, id, base, years),
          managementAllowance(allowance), teamSize(team) {
        HR_LOG("Manager created: " + n);
    }

    // Manager salary = base salary + seniority allowance + management allowance + team bonus
    double calculateSalary() const override {
        double base = Employee::calculateSalary();
        // Management allowance + team bonus (500 per team member)
        return base + managementAllowance + teamSize * 500.0;
    }

    std::string getRole() const override {
        return "Manager";
    }

    void display() const override {
        Employee::display();
        std::cout << " (manages " << teamSize << " people)";
        std::cout << std::endl;
    }
};

// ====== Derived class: SalesPerson ======
class SalesPerson : public Employee {
private:
    double commissionRate;      // commission rate (e.g. 0.05 = 5%)
    double monthlySales;        // monthly sales

public:
    SalesPerson(const std::string& n, const std::string& id,
                double base, int years, double rate, double sales)
        : Employee(n, id, base, years),
          commissionRate(rate), monthlySales(sales) {
        HR_LOG("SalesPerson created: " + n);
    }

    // Salesperson salary = base salary + seniority allowance + sales commission
    double calculateSalary() const override {
        double base = Employee::calculateSalary();
        return base + monthlySales * commissionRate;
    }

    std::string getRole() const override {
        return "Sales";
    }

    void display() const override {
        Employee::display();
        std::cout << " (sales $" << monthlySales << ")";
        std::cout << std::endl;
    }
};

// ====== Derived class: Intern ======
class Intern : public Employee {
private:
    double stipend;   // internship stipend (fixed)
    std::string school;

public:
    Intern(const std::string& n, const std::string& id,
           double base, int years, double stip, const std::string& sch)
        : Employee(n, id, base, years), stipend(stip), school(sch) {
        HR_LOG("Intern created: " + n);
    }

    // Interns get no seniority allowance, only a fixed stipend
    double calculateSalary() const override {
        return baseSalary + stipend;  // use base salary + stipend directly
    }

    std::string getRole() const override {
        return "Intern";
    }

    void display() const override {
        Employee::display();
        std::cout << " (" << school << ")";
        std::cout << std::endl;
    }
};

}  // namespace company

// ========== Demo code ==========
int main() {
    using namespace company;

    std::cout << "===== Company Payroll Management System =====\n" << std::endl;

    // Manage all employees through base class pointers
    std::vector<std::unique_ptr<Employee>> payroll;

    payroll.push_back(std::make_unique<Manager>(
        "Alice Wang", "E001", 15000.0, 5, 5000.0, 8));
    payroll.push_back(std::make_unique<Manager>(
        "Bob Li",     "E002", 12000.0, 3, 3000.0, 4));
    payroll.push_back(std::make_unique<SalesPerson>(
        "Charlie Zhang", "E003", 8000.0, 2, 0.05, 50000.0));
    payroll.push_back(std::make_unique<SalesPerson>(
        "Diana Chen", "E004", 8000.0, 1, 0.05, 80000.0));
    payroll.push_back(std::make_unique<Intern>(
        "Edward Liu", "E005", 3000.0, 0, 2000.0, "Tsinghua University"));
    payroll.push_back(std::make_unique<Intern>(
        "Fiona Wu",   "E006", 3000.0, 0, 2000.0, "Peking University"));

    // Print the payroll table
    std::cout << std::left
              << std::setw(15) << "Name"
              << std::setw(10) << "Emp ID"
              << std::setw(12) << "Role"
              << std::setw(12) << "Salary"
              << "Notes" << std::endl;
    std::cout << std::string(70, '-') << std::endl;

    double totalPayroll = 0.0;
    for (const auto& emp : payroll) {
        emp->display();
        totalPayroll += emp->calculateSalary();
    }

    std::cout << std::string(70, '-') << std::endl;
    std::cout << "Total employees: " << Employee::getTotalEmployees() << std::endl;
    std::cout << "Total monthly payroll: $" << std::fixed << std::setprecision(2)
              << totalPayroll << std::endl;

    // Salary statistics by department
    std::cout << "\n===== Salary Statistics by Department =====" << std::endl;
    double managerTotal = 0, salesTotal = 0, internTotal = 0;
    for (const auto& emp : payroll) {
        if (emp->getRole() == "Manager")
            managerTotal += emp->calculateSalary();
        else if (emp->getRole() == "Sales")
            salesTotal += emp->calculateSalary();
        else if (emp->getRole() == "Intern")
            internTotal += emp->calculateSalary();
    }
    std::cout << "Management: $" << managerTotal << std::endl;
    std::cout << "Sales: $" << salesTotal << std::endl;
    std::cout << "Interns: $" << internTotal << std::endl;

    return 0;
}
```

Concepts applied:
  - Inheritance hierarchy (Employee -> Manager/SalesPerson/Intern)
  - Construction chain
  - Using virtual functions (calculateSalary, getRole)
  - Using using declarations to change access levels (Employee base class functions)
  - Namespace company
  - Managing dynamic memory with unique_ptr
  - Debug macro
  - Upcasting

------------------------------------

[Real-life scenario] A university needs to manage students, teachers, and teaching assistants (who are both students and teachers).

```cpp

// ====== TeachingAssistant: multiply inherits Student and Teacher ======

    taCharlie->study();     // Charlie studying as a student
    taCharlie->teach();     // Charlie teaching as a teacher
    taCharlie->assistCourse();  // Charlie tutoring as a TA

    // Show the benefit of virtual inheritance: no ambiguous-access problems
    std::cout << "\n===== Virtual Inheritance Check =====" << std::endl;
    std::cout << "TA Charlie's name: " << taCharlie->getName() << std::endl;
    std::cout << "TA Charlie's age: " << taCharlie->getAge() << std::endl;
    // Without virtual inheritance, the two lines above would produce an "ambiguous" error!
    // Because there would be two Person subobjects (one from Student, one from Teacher)
    // Virtual inheritance guarantees there is only one Person subobject

    return 0;
}
```

  - Solving diamond inheritance with virtual inheritance
  - The most derived class must explicitly construct the virtual base class
  - Managing objects with shared_ptr (Chapter 4)
  - static member counters

I. An Alternative to Inheritance: Composition vs Inheritance
------------------------------------

Composition is usually preferable to inheritance. In C++, when the relationship between two classes is "has-a",
you should use composition rather than inheritance.

```cpp
// ❌ Bad use of inheritance -- a "has-a" relationship should not use inheritance
class Engine { /*...*/ };
class Car : public Engine {  // Car IS-A Engine? That makes no sense!
    // semantically wrong
};

// ✅ Correct use of composition -- a "has-a" relationship
class Engine { /*...*/ };
class Car {
    Engine engine;  // Car HAS-A Engine -- semantically correct
    // ...
};

// Deciding by use case:
// Inheritance (IS-A):  Dog IS-A Animal,  Circle IS-A Shape
// Composition (HAS-A): Car HAS-A Engine,  Person HAS-A Address
// When in doubt, prefer composition -- lower coupling, more flexibility
```

[Why Composition Is Preferable to Inheritance]
1. Composition has lower coupling -- changing the internals of a composed object does not affect the outer class
2. Composition is more flexible -- the composed object can be swapped at runtime
3. Inheritance adds class-hierarchy complexity and reduces maintainability
4. Inheritance exposes the base class's implementation details (especially protected members)
5. Composition supports the "Strategy pattern" -- change behavior by swapping the composed object

```cpp
// Strategy pattern example: replacing inheritance with composition
#include <iostream>
#include <memory>

// Strategy interface (polymorphic behavior via composition)
class FlyBehavior {
public:
    virtual void fly() = 0;
    virtual ~FlyBehavior() = default;
};

class FlyWithWings : public FlyBehavior {
public:
    void fly() override { std::cout << "Flying with wings!" << std::endl; }
};

class NoFly : public FlyBehavior {
public:
    void fly() override { std::cout << "Cannot fly." << std::endl; }
};

class Duck {
private:
    std::unique_ptr<FlyBehavior> flyBehavior;  // composition
public:
    Duck(std::unique_ptr<FlyBehavior> fb)
        : flyBehavior(std::move(fb)) { }

    void performFly() {
        flyBehavior->fly();  // delegate to the composed object
    }

    void setFlyBehavior(std::unique_ptr<FlyBehavior> fb) {
        flyBehavior = std::move(fb);  // swap behavior at runtime!
    }
};

// This way Duck doesn't need a complex inheritance hierarchy; composition provides flexible behavior changes
```

II. Practical Uses of protected and private Inheritance
------------------------------------

```cpp
// private inheritance: "implemented-in-terms-of"
// A class wants to reuse the base class's implementation without exposing the base class's interface

class Timer {
public:
    void start() { /* ... */ }
    void stop()  { /* ... */ }
    int elapsed() const { return 0; }
};

// ❌ Using public inheritance -- exposes all of Timer's interface
class Stopwatch_public : public Timer {
    // Users can call start()/stop()/elapsed() through Stopwatch
    // but these names may not have the right meaning for a stopwatch
};

// ✅ Using private inheritance -- hides Timer's interface
class Stopwatch : private Timer {
public:
    void begin() { start(); }         // expose suitable interface names
    void end()   { stop(); }
    int  result() const { return elapsed(); }

    // Selectively expose some base class interfaces
    using Timer::elapsed;  // raise elapsed to public
};

// Equivalent composition implementation (usually preferred):
class Stopwatch_composition {
    Timer timer;
public:
    void begin() { timer.start(); }
    void end()   { timer.stop(); }
    int  result() const { return timer.elapsed(); }
};
```

III. Name Lookup Rules in Inheritance
------------------------------------

C++ name lookup in an inheritance hierarchy follows these rules:

```cpp
class Base {
public:
    void func(int)  { std::cout << "Base::func(int)\n"; }
    void func(double){ std::cout << "Base::func(double)\n"; }
};

class Intermediate : public Base {
public:
    // If any function named func is defined in Intermediate,
    // all functions named func in Base are hidden!
    void func(char) { std::cout << "Intermediate::func(char)\n"; }
};

class Derived : public Intermediate {
public:
    void func(long) { std::cout << "Derived::func(long)\n"; }
};

int main() {
    Derived d;
    d.func(10);         // Derived::func(long) -- int can convert to long
    d.func('a');        // Derived::func(long) -- char can convert to long
                        // Intermediate::func(char) is hidden!
    // d.func(3.14);    // Error! All of Base's func are hidden
    d.Base::func(3.14); // the Base version must be called explicitly

    // Use using to solve name hiding
    // Add to Derived: using Intermediate::func;
    // Add to Intermediate: using Base::func;
    return 0;
}

// Name lookup rule: once the compiler finds the first matching name, it stops searching upward
// Lookup order: derived class -> direct base class -> indirect base class -> ...
// As soon as a function with the same name (regardless of parameters) is found at some level, the upward search stops
// This is "Name Hiding" (also called "Function Hiding")
```

IV. Construction and Destruction in Inheritance -- Common Pitfalls
------------------------------------

```cpp
#include <iostream>
#include <string>

class Base {
protected:
    std::string resource;
public:
    // Note: the constructor body may throw an exception!
    Base(const std::string& name) : resource(name) {
        std::cout << "Base::Base() for " << resource << std::endl;
    }

    // Virtual destructor -- if this class is used as a base class
    virtual ~Base() {
        std::cout << "Base::~Base() for " << resource << std::endl;
    }
};

class Derived : public Base {
    int* dynamicArray;
    size_t size;
public:
    Derived(const std::string& name, size_t n)
        : Base(name),    // if this throws, Derived will not be constructed
          size(n),
          dynamicArray(new int[n]) {  // if this throws, Base is destroyed automatically
        std::cout << "Derived::Derived() allocating " << n << " ints"
                  << std::endl;
        // Risk: if the code below throws, you must make sure resources are cleaned up
    }

    ~Derived() override {
        delete[] dynamicArray;
        std::cout << "Derived::~Derived() freed " << size << " ints"
                  << std::endl;
    }

    // Rule: manage resources with RAII; avoid handling complex logic manually in constructors/destructors
};

// ⚠️ The pitfall of calling virtual functions in constructors/destructors:
class Base2 {
public:
    Base2() {
        std::cout << "Base2::Base2()" << std::endl;
        process();  // calls Base2::process(), not Derived2::process()!
    }
    virtual void process() { std::cout << "Base2::process()" << std::endl; }
    virtual ~Base2() {
        std::cout << "Base2::~Base2()" << std::endl;
        process();  // also calls Base2::process()
    }
};

class Derived2 : public Base2 {
public:
    Derived2() { std::cout << "Derived2::Derived2()" << std::endl; }
    void process() override {
        std::cout << "Derived2::process()" << std::endl;
    }
};

// Reason: during construction/destruction the vptr points to the current class's vtable,
// not the most derived class's vtable. This behavior is mandated by the C++ standard
// to prevent access to uninitialized derived-class members.
```

V. Construction Order in Multiple Inheritance Explained
------------------------------------

```cpp
#include <iostream>

class A {
public:
    A() { std::cout << "A()" << std::endl; }
};

class B {
public:
    B() { std::cout << "B()" << std::endl; }
};

class C : virtual public A {  // C virtually inherits A
public:
    C() { std::cout << "C()" << std::endl; }
};

class D : virtual public A {  // D virtually inherits A
public:
    D() { std::cout << "D()" << std::endl; }
};

class E : public B {          // E non-virtually inherits B
public:
    E() { std::cout << "E()" << std::endl; }
};

// A complex multiple inheritance hierarchy:
//      A (virtual)
//     / \
//    C   D    B
//    |   |   /
//    |   |  /
//     \  | /
//       F
// (C and D virtually inherit A, E normally inherits B)
class F : public C, public D, public E {
public:
    F() { std::cout << "F()" << std::endl; }
};

// Construction order:
// 1. Virtual base class A (because C and D virtually inherit A, A is constructed by the most derived class F)
// 2. Non-virtual direct base classes, in declaration order: C, D, E
//    - C (A already constructed) -> only C's own part is constructed
//    - D (A already constructed) -> only D's own part is constructed
//    - E -> first constructs B (non-virtual base class), then E itself
// 3. F's own members
// 4. F's constructor body
//
// Output:
// A()
// C()
// D()
// B()
// E()
// F()

// Destruction order: strictly the reverse of construction order
// F() -> E() -> B() -> D() -> C() -> A()
```

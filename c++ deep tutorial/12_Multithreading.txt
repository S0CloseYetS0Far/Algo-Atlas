[12.1.1 Basic Example: Creating and Waiting for Threads]

C++11 introduced the std::thread class, making multithreaded programming simple and
cross-platform. Here is the most basic example of creating and waiting for threads:

// ============================================================
// Example 12-1: The most basic thread creation
// ============================================================
#include <iostream>
#include <thread>
#include <string>

void print_message(const std::string& msg) {
    std::cout << "[Thread " << std::this_thread::get_id()
              << "] " << msg << std::endl;
}

int main() {
    std::cout << "[Main] Starting. Main thread ID: "
              << std::this_thread::get_id() << std::endl;

    // Create threads: pass a function and its arguments
    std::thread t1(print_message, "Hello from thread 1");
    std::thread t2(print_message, "Hello from thread 2");

    // join: the main thread waits for the child threads to finish
    t1.join();
    t2.join();

    std::cout << "[Main] All threads finished." << std::endl;
    return 0;
}
// Compile: g++ -std=c++17 -pthread 12-1.cpp -o 12-1

[12.1.2 The Difference Between join and detach]

// ============================================================
// Example 12-2: join vs detach
// ============================================================
#include <iostream>
#include <thread>
#include <chrono>
#include <atomic>

std::atomic<bool> g_running{true};

void worker_thread(int id) {
    while (g_running.load(std::memory_order_relaxed)) {
        std::cout << "Worker " << id << " is running..." << std::endl;
        std::this_thread::sleep_for(std::chrono::milliseconds(500));
    }
    std::cout << "Worker " << id << " stopped." << std::endl;
}

void independent_task() {
    std::this_thread::sleep_for(std::chrono::seconds(2));
    std::cout << "[Detached] Independent task done!" << std::endl;
}

int main() {
    // === join example ===
    std::thread joined_thread(worker_thread, 1);
    std::this_thread::sleep_for(std::chrono::seconds(2));
    g_running.store(false);
    joined_thread.join();  // The main thread blocks here until worker_thread finishes
    // After join, the thread state can be checked safely
    std::cout << "joined_thread is joinable? "
              << joined_thread.joinable() << std::endl;  // 0 (false)

    // === detach example ===
    // detach separates the thread from the std::thread object; the thread runs in the background
    // After detaching it cannot be joined; resources are reclaimed automatically when it ends
    std::thread detached_thread(independent_task);
    detached_thread.detach();
    // detached_thread.join();  // Error! A detached thread cannot be joined
    std::cout << "[Main] detached_thread is joinable? "
              << detached_thread.joinable() << std::endl;  // 0 (false)
    std::cout << "[Main] Main thread continues without waiting..."
              << std::endl;

    // Must give the detached thread enough time to finish
    std::this_thread::sleep_for(std::chrono::seconds(3));
    std::cout << "[Main] Exiting." << std::endl;
    return 0;
}

[12.1.3 Creating Threads with Lambdas]

// ============================================================
// Example 12-3: Creating threads with lambdas and capture lists
// ============================================================
#include <iostream>
#include <thread>
#include <vector>

int main() {
    int shared_counter = 0;
    std::vector<std::thread> threads;

    // Capture by reference: multiple threads operate on the same variable (demo only; needs synchronization in practice)
    for (int i = 0; i < 5; ++i) {
        threads.emplace_back([&shared_counter, i]() {
            // Each thread performs some operation on shared_counter
            int local = shared_counter;
            std::this_thread::sleep_for(std::chrono::milliseconds(10));
            shared_counter = local + i + 1;
            std::cout << "Thread " << i
                      << " set counter to " << shared_counter << std::endl;
        });
    }

    // Capture by value: each thread operates independently
    std::thread independent([i = 42]() {
        std::cout << "Independent thread with captured value: "
                  << i << std::endl;
    });

    for (auto& t : threads) t.join();
    independent.join();

    std::cout << "Final counter: " << shared_counter << std::endl;
    return 0;
}

[12.1.4 Low-Level Principles: The Relationship Between CPU Cores and Threads]
================================================================================

I. Physical Core
    An independent compute unit that physically exists on the CPU chip. Each core has its own:
   - Arithmetic Logic Unit (ALU)
   - Floating-Point Unit (FPU)
   - L1/L2 caches
   - Instruction decoder
   An 8-core CPU can truly execute 8 instructions at the same time (true parallelism).

II. Hyper-Threading / SMT
    Intel's Hyper-Threading technology (AMD calls it SMT, Simultaneous Multi-Threading)
    makes one physical core appear as two logical cores. How it works:
   - A physical core contains two sets of registers and program counters (PC)
   - But they share a single set of execution units such as the ALU/FPU
   - When thread A is waiting on memory due to a cache miss, the CPU can switch to running thread B
   - The performance gain is usually 15%-30%, not a true doubling
   - Because the execution units are shared, the two hyper-threads compete for resources

    So a "8-core 16-thread" CPU really has only 8 physical cores. For compute-bound tasks,
    more threads than physical cores gives no performance gain (it actually hurts due to
    context switching). For I/O-bound tasks (e.g. waiting on network or disk),
    hyper-threading can effectively fill the waiting time.

III. How std::thread relates to the operating system
    On Linux, std::thread calls pthread_create under the hood to create a POSIX thread.
    On Windows, it calls CreateThread to create a Win32 thread.
    On macOS, it is based on Mach threads and POSIX threads.

    On every platform, threads created by std::thread are managed by the OS kernel; they
    are true Kernel-Level Threads (KLT) and share the scheduler's run queues with processes.

    By contrast, the "coroutines" or "green threads" implemented by some languages are
    scheduled in user space without going through the OS. Although C++20 introduced
    coroutines, std::thread is always a kernel thread.


================================================================================
[12.1.5 Low-Level Principles: OS Thread Scheduling]
================================================================================

I. Round-Robin
    The most basic scheduling algorithm: each ready thread is given a fixed time slice
    (usually 10ms-100ms; Linux CFS uses dynamic time slices). When the slice is used up,
    the OS forcibly switches threads. This Preemptive Scheduling guarantees that every
    thread gets CPU time.

    The art of choosing the time slice:
   - Time slice too short → context switch overhead dominates, throughput drops
   - Time slice too long → interactive response slows down, users feel "lag"
   - Linux CFS uses a dynamically computed time slice: target_latency / nr_running

II. CFS (Completely Fair Scheduler)
    Since 2.6.23, Linux has used CFS instead of the O(1) scheduler. Core ideas:
   - Each thread maintains a virtual runtime (vruntime)
   - vruntime = actual runtime × (nice_0_weight / current thread weight)
   - The scheduler always picks the thread with the smallest vruntime to run
   - Threads are managed in a red-black tree (the data structure underlying std::map),
     so picking the next thread is O(log n)
   - The vruntime of high-priority threads grows slowly, so they get more CPU time

    CFS fairness means: within one scheduling period, all runnable threads get CPU shares
    proportional to their weights. For example: thread A with nice=-5 (high priority,
    higher weight), thread B with nice=0 (normal), and thread C with nice=+5 (low priority)
    get CPU shares in a ratio of roughly 3:2:1.

III. Scheduling policy types
    Linux supports several scheduling policies:
   - SCHED_OTHER (CFS): the default time-sharing policy, used for normal processes
   - SCHED_FIFO: real-time first-in-first-out; higher priority preempts lower, used for
     latency-sensitive tasks
   - SCHED_RR: real-time round-robin; like FIFO but with a time slice limit
   - SCHED_DEADLINE: deadline scheduling; guarantees tasks complete before their deadline

    These can be set with pthread_attr_setschedpolicy, but real-time scheduling is
    generally not recommended for ordinary programs.


================================================================================
[12.1.6 Low-Level Principles: The Memory Sharing Model of Processes and Threads]
================================================================================

I. Process memory layout (review of Chapter 3)
    ┌─────────────┐ High address
    │  Stack      │ ← Function call frames, local variables, return addresses
    ├─────────────┤
    │     ↓↓↓     │
    │   (gap)     │
    │     ↑↑↑     │
    ├─────────────┤
    │  Heap       │ ← Dynamically allocated memory (new/malloc)
    ├─────────────┤
    │  BSS        │ ← Uninitialized global/static variables
    ├─────────────┤
    │  Data       │ ← Initialized global/static variables
    ├─────────────┤
    │  Text       │ ← The program's machine instructions (read-only)
    └─────────────┘ Low address

II. Resources shared between threads
   All threads in the same process share:
   - Heap: new/malloc from all threads allocate from the same heap and can access each other's data
   - Global/static variables: the .data and .bss segments are visible to all threads
   - Text segment: threads of the same process execute the same program code
   - File descriptors: a file opened by thread A can also be read/written by thread B
     (the file descriptor table is per-process)
   - Signal handlers: a process-level setting (but a signal is usually delivered to one
     particular thread of the process)
   - Process attributes such as the process ID, working directory, and resource limits

III. Resources private to each thread
   Each thread has its own:
   - Stack: each thread has its own call stack, usually allocated in the high-address
     region of the process address space (Linux default stack size is 8MB, adjustable
     with pthread_attr_setstacksize)
   - Register set: including general-purpose registers, program counter (PC),
     stack pointer (SP), flags register, etc.
   - Thread ID (tid): the OS-level thread identifier
   - errno: the C standard library errno is thread-local (implemented via TLS)
   - Signal Mask: each thread can block different signals

IV. Problems caused by sharing memory
   Sharing is convenient, but it is also the root of Data Races.
   When thread A and thread B access the same memory location at the same time:
   - At least one of them is a write
   - There is no synchronization (locks, atomic operations, etc.)
   then the program's behavior is Undefined Behavior.

   For example, two threads both execute counter++:
   - Thread A reads counter=0
   - Thread B reads counter=0  (both read 0)
   - Thread A computes counter+1=1
   - Thread B computes counter+1=1
   - Thread A writes counter=1
   - Thread B writes counter=1  (overwriting A's write!)
   In the end counter=1 instead of the expected 2. This is a classic Lost Update.


================================================================================
[12.1.7 Low-Level Principles: The Cost of Thread Context Switches]
================================================================================

When the OS decides to switch from one thread to another, it performs these steps:

I. Steps of a context switch
   1. Save the current thread's state
      - Save all register values to the current thread's kernel stack
      - Save the program counter (address of the next instruction)
      - Save the stack pointer
      - Save other CPU status registers (such as the flags register)

   2. Update scheduling information
      - Move the current thread to the appropriate queue (ready queue or wait queue)
      - Update the thread's vruntime (CFS) or other scheduling statistics

   3. Pick the next thread
      - CFS picks the thread with the smallest vruntime from the red-black tree
      - Or wakes up a thread from a wait queue whose condition is satisfied

   4. Restore the next thread's state
      - Restore register values from the new thread's kernel stack
      - Restore the program counter
      - Restore the stack pointer, switching to the new thread's stack
      - Update the MMU (Memory Management Unit) page table pointer (if switching
        processes rather than threads)

II. Analyzing the cost of a context switch
   - Direct cost: the instructions to save/restore registers (typically ~100-200 CPU cycles)
   - Indirect cost -- cache invalidation:
     * Data in the L1/L2/L3 caches belongs to the previous thread; the new thread gets
       cache misses when accessing its own data
     * The TLB (Translation Lookaside Buffer, the page table cache) must also be reloaded
     * The branch predictor's history becomes useless and must be retrained
     * The indirect cost can be 10-100x larger than the direct cost

   - Real numbers (rough estimates):
     * Thread switch (same process): about 1000-5000 CPU cycles
     * Process switch (different processes): about 5000-50000 CPU cycles (TLB and page
       tables must be flushed)

III. Strategies to reduce context switches
   1. Number of threads ≈ number of logical CPU cores (for compute-bound tasks)
   2. Use a thread pool to reuse threads and avoid frequent creation/destruction
   3. Avoid heavy lock contention to reduce threads blocking and waiting
   4. Use lock-free data structures to reduce blocking
   5. Set CPU affinity (pthread_setaffinity_np) to reduce thread migration between cores
      // CPU affinity example
      #include <pthread.h>
      cpu_set_t cpuset;
      CPU_ZERO(&cpuset);
      CPU_SET(0, &cpuset);  // Bind to core 0
      pthread_setaffinity_np(pthread_self(), sizeof(cpu_set_t), &cpuset);


================================================================================
[12.1.8 Low-Level Principles: Memory Barriers and the Cache Coherence Protocol (MESI)]
================================================================================

I. The cache coherence problem
   In modern CPUs, every core has its own L1/L2 cache. When core A modifies the value of variable x:
   - The value of x in A's L1 cache is updated
   - But the value of x in core B's L1 cache is still the old one
   - This makes different cores see inconsistent values of x

   CPU designers solve this with cache coherence protocols. The most common is MESI.

II. The MESI protocol: four cache line states
   Each Cache Line (usually 64 bytes) is in one of four states:

   M (Modified):
     The data in this cache line has been modified by this core and differs from the
     copies in other cores. This core has exclusive ownership. Other cores cannot read
     it until the line is written back to main memory.

   E (Exclusive):
     The data in this cache line matches main memory, and only this core has a copy.
     This core can modify it directly (moving to M) without notifying other cores.

   S (Shared):
     The data in this cache line matches main memory, and multiple cores may have copies.
     To modify it, the core must first send a message invalidating the other copies.

   I (Invalid):
     This cache line is invalid and cannot be used. Reading the data requires fetching
     it from another core or from main memory.

   Transition rules:
   - A core reads data: if other cores have a copy → S state, otherwise → E state
   - A core writes data: other cores' copies must first be invalidated → E or M state
   - Another core writes: this core's copy → I state

III. Message passing in the MESI protocol
   CPUs send messages over the bus (or the QPI/Infinity Fabric interconnect in modern CPUs):
   - BusRdX: "I want to read and modify; please invalidate your copies"
   - BusUpgr: "I want to modify; please invalidate your copies"
   - BusWB: "I am writing back to main memory"
   - Flush: "I am answering your read request"

IV. Memory Barrier / Fence
   Even with the MESI protocol, the CPU and compiler may still reorder instructions
   (for performance). A memory barrier is an instruction that prevents reordering:

   1. Compiler barrier: asm volatile("" ::: "memory") prevents the compiler from reordering instructions
   2. CPU barrier: std::atomic_thread_fence() or inline assembly mfence/lfence/sfence

   Types of memory barriers:
   - Acquire Barrier: operations after the barrier are not reordered before it
     // Usually used after a read, to guarantee the value read is up to date
   - Release Barrier: operations before the barrier are not reordered after it
     // Usually used before a write, to guarantee the written value is visible to other cores
   - Full Barrier: prevents reordering in both directions

   In C++, this is controlled through std::atomic's memory_order:
   - memory_order_relaxed: guarantees atomicity only, no ordering
   - memory_order_acquire: acquire semantics
   - memory_order_release: release semantics
   - memory_order_acq_rel: acquire + release
   - memory_order_seq_cst: sequential consistency (default, strongest guarantee, but most expensive)


================================================================================
[12.1.9 Low-Level Principles: The Nature of Data Races]
================================================================================

The C++ standard's definition of a data race ([intro.races]):
Two expression evaluations, in the same thread or different threads, where one modifies
a memory location and the other reads or also modifies the same memory location, and
these evaluations are not ordered by a happens-before relationship, constitute a data
race. A data race results in undefined behavior.

I. The three conditions of a data race (all are required)
   1. Multiple threads access the same memory location
   2. At least one thread performs a write
   3. There is no synchronization between the accesses (no lock / no atomic operation /
      no proper happens-before relationship)

II. Common data race scenarios
   1. Incrementing a simple shared variable
      // Bad example -- data race!
      int counter = 0;
      void increment() {
          counter++;  // read-modify-write, not atomic
      }

   2. Check-Then-Act
      // Bad example
      if (list.size() > 0) {
          auto item = list.back();  // Another thread may already have popped it
          list.pop_back();
      }

   3. Temporarily broken invariants
      // Bad example
      struct Node { Node* prev; Node* next; };
      // While modifying a doubly linked list, the intermediate state may have inconsistent .prev and .next
      // Another thread may happen to observe this inconsistent state

   4. Partial writes of non-atomic types
      // Bad example
      struct LargeData { int64_t a; int64_t b; int64_t c; };
      // On a 32-bit platform, a 64-bit operation may be split into two 32-bit operations
      // Another thread may read a "half-written" value

III. How to detect data races
   1. Thread Sanitizer (tsan): add -fsanitize=thread at compile time
   $ g++ -std=c++17 -fsanitize=thread -g prog.cpp -o prog -pthread
   $ ./prog
   // TSan reports every data race it detects

   2. Helgrind (a Valgrind tool):
   $ valgrind --tool=helgrind ./prog

   3. Code review: check that every access path to shared data has proper synchronization

IV. Ways to prevent data races
   1. Mutexes (std::mutex): protect critical sections
   2. Atomic operations (std::atomic): suitable for simple read-modify-write operations
   3. Immutable data: data that is never modified after creation is inherently thread-safe
   4. Thread-local storage (thread_local): each thread has its own copy
   5. Message passing: threads share no data and communicate only through queues

[12.2.1 Complete Usage of the <thread> Header]

I. The std::thread class

// ============================================================
// Example 12-4: Various ways to construct a thread
// ============================================================
#include <iostream>
#include <thread>
#include <functional>
#include <chrono>

// 1. Plain function
void func(int a, double b) {
    std::cout << "func: " << a << ", " << b << std::endl;
}

// 2. Function object (Functor)
struct Task {
    int id;
    Task(int i) : id(i) {}
    void operator()() const {
        std::cout << "Task " << id << " executing." << std::endl;
    }
};

// 3. Class member functions
class Worker {
public:
    void process(int value) {
        std::cout << "Worker::process(" << value << ")" << std::endl;
    }
    void process_const(int value) const {
        std::cout << "Worker::process_const(" << value << ")" << std::endl;
    }
};

int main() {
    // Way 1: plain function + arguments
    std::thread t1(func, 42, 3.14);

    // Way 2: function object
    Task task(100);
    std::thread t2(task);
    // Note: std::thread t2(Task(100)); may be parsed as a function declaration!
    // Use extra parentheses: std::thread t2((Task(100)));
    // Or use braces: std::thread t2{Task(100)};

    // Way 3: lambda expression
    std::thread t3([](int x, int y) {
        std::cout << "Lambda: " << x + y << std::endl;
    }, 10, 20);

    // Way 4: member function + object pointer/reference + arguments
    Worker w;
    std::thread t4(&Worker::process, &w, 99);

    // Way 5: const member function
    std::thread t5(&Worker::process_const, &w, 88);

    // Way 6: std::ref wraps a reference argument (thread passes by value by default)
    int value = 5;
    std::thread t6([](int& v) {
        v = 100;  // Modify the referenced value
    }, std::ref(value));

    // Way 7: std::function
    std::function<void()> f = []() {
        std::cout << "std::function thread" << std::endl;
    };
    std::thread t7(f);

    // Way 8: move construction
    std::thread t8([](){ std::cout << "T8" << std::endl; });
    std::thread t9(std::move(t8));  // t8 no longer owns a thread
    // t8.join(); // Error! t8 is empty

    t1.join(); t2.join(); t3.join();
    t4.join(); t5.join(); t6.join();
    t7.join(); t9.join();

    std::cout << "After t6, value = " << value << std::endl;  // 100
    return 0;
}


II. std::thread member functions

// ============================================================
// Example 12-5: join, detach, joinable
// ============================================================
#include <iostream>
#include <thread>
#include <chrono>

void work() {
    std::this_thread::sleep_for(std::chrono::seconds(1));
}

int main() {
    std::thread t;

    // joinable() - checks whether the thread can be joined (i.e. is associated with an active thread)
    std::cout << "Default constructed, joinable: "
              << t.joinable() << std::endl;  // false

    t = std::thread(work);
    std::cout << "After assignment, joinable: "
              << t.joinable() << std::endl;  // true

    t.join();
    std::cout << "After join, joinable: "
              << t.joinable() << std::endl;  // false

    // If the thread is still joinable when destroyed, std::terminate is called!
    // std::thread dt(work);
    // return 0; // dt destroyed → terminate!

    // Safe pattern: make sure to join or detach before destruction
    std::thread safe_t(work);
    if (safe_t.joinable()) {
        safe_t.join();  // or safe_t.detach();
    }
    return 0;
}


III. Functions in the std::this_thread namespace

// ============================================================
// Example 12-6: this_thread functions
// ============================================================
#include <iostream>
#include <thread>
#include <chrono>
#include <sstream>

std::string current_thread_id_str() {
    std::ostringstream oss;
    oss << std::this_thread::get_id();
    return oss.str();
}

void sleepy_worker() {
    std::cout << "Worker starting on thread: "
              << current_thread_id_str() << std::endl;

    // sleep_for: block the current thread for the given duration
    std::cout << "  Sleeping for 500ms..." << std::endl;
    std::this_thread::sleep_for(std::chrono::milliseconds(500));

    // sleep_until: block until the given time point
    auto wake_time = std::chrono::steady_clock::now()
                     + std::chrono::milliseconds(500);
    std::cout << "  Sleeping until..." << std::endl;
    std::this_thread::sleep_until(wake_time);

    std::cout << "Worker finished on thread: "
              << current_thread_id_str() << std::endl;
}

void yield_example() {
    // yield: voluntarily give up the CPU time slice so other threads can run
    // Usually used to optimize spin locks or busy waiting
    for (int i = 0; i < 100; ++i) {
        // ... do some lightweight work
        if (i % 10 == 0) {
            std::this_thread::yield();  // Give other threads a chance
            std::cout << "Yield at iteration " << i << std::endl;
        }
        std::this_thread::sleep_for(std::chrono::milliseconds(1));
    }
}

int main() {
    std::cout << "Main thread ID: " << current_thread_id_str() << std::endl;

    // get_id: returns a std::thread::id
    std::thread t1(sleepy_worker);
    std::cout << "t1 thread ID: " << t1.get_id() << std::endl;

    // hardware_concurrency: returns the number of concurrent threads supported by the hardware
    std::cout << "Hardware concurrency (logical cores): "
              << std::thread::hardware_concurrency() << std::endl;

    // yield example
    std::thread t2(yield_example);

    t1.join();
    t2.join();

    // native_handle: get the underlying OS thread handle (for system calls)
    // pthread_t handle = t1.native_handle();  // Linux
    // HANDLE handle = t1.native_handle();     // Windows

    return 0;
}

I. std::mutex, the basic mutex

// ============================================================
// Example 12-7: Basic mutex usage
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>

std::mutex mtx;
int shared_counter = 0;

void safe_increment(int times) {
    for (int i = 0; i < times; ++i) {
        mtx.lock();           // Acquire the lock
        ++shared_counter;     // Critical Section
        mtx.unlock();         // Release the lock
    }
}

void unsafe_increment(int times) {
    for (int i = 0; i < times; ++i) {
        ++shared_counter;     // Data race!
    }
}

int main() {
    int iterations = 100000;

    // Unsafe version: result is nondeterministic
    shared_counter = 0;
    std::thread u1(unsafe_increment, iterations);
    std::thread u2(unsafe_increment, iterations);
    u1.join(); u2.join();
    std::cout << "Unsafe result:  " << shared_counter
              << " (expected " << (2 * iterations) << ")" << std::endl;

    // Safe version: result is deterministic
    shared_counter = 0;
    std::thread s1(safe_increment, iterations);
    std::thread s2(safe_increment, iterations);
    s1.join(); s2.join();
    std::cout << "Safe result:    " << shared_counter
              << " (expected " << (2 * iterations) << ")" << std::endl;

    return 0;
}


II. std::lock_guard: RAII-style lock management

// ============================================================
// Example 12-8: lock_guard usage
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>
#include <vector>

class ThreadSafeCounter {
private:
    mutable std::mutex mtx_;
    int value_ = 0;

public:
    void increment() {
        std::lock_guard<std::mutex> lock(mtx_);
        // lock_guard calls mtx_.lock() on construction
        ++value_;
        // lock_guard calls mtx_.unlock() on destruction -- automatic release, exception-safe!
    }

    void decrement() {
        std::lock_guard<std::mutex> lock(mtx_);
        --value_;
    }

    int get_value() const {
        std::lock_guard<std::mutex> lock(mtx_);
        return value_;
    }
};

int main() {
    ThreadSafeCounter counter;
    std::vector<std::thread> threads;

    for (int i = 0; i < 10; ++i) {
        threads.emplace_back([&counter]() {
            for (int j = 0; j < 1000; ++j) {
                counter.increment();
            }
        });
    }

    for (auto& t : threads) t.join();
    std::cout << "Final count: " << counter.get_value() << std::endl;
    return 0;
}


III. std::unique_lock: advanced lock management

// ============================================================
// Example 12-9: unique_lock usage
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>
#include <chrono>

std::mutex resource_mtx;
int resource = 0;

void flexible_locker(int id) {
    // unique_lock provides flexible lock management:
    // 1. Deferred locking (defer_lock)
    // 2. Try locking (try_to_lock)
    // 3. Adopt an already-locked mutex (adopt_lock)

    // Way 1: normal construction, locks immediately
    {
        std::unique_lock<std::mutex> lock(resource_mtx);
        resource += 10;
        std::cout << "Thread " << id
                  << " (normal): resource = " << resource << std::endl;
    } // lock destroyed, unlocks automatically

    // Way 2: deferred locking, lock manually later
    {
        std::unique_lock<std::mutex> lock(resource_mtx, std::defer_lock);
        // Do some preparation that doesn't need the lock...
        std::cout << "Thread " << id << " preparing..." << std::endl;
        lock.lock();  // Only now acquire the lock
        resource += 20;
        std::cout << "Thread " << id
                  << " (deferred): resource = " << resource << std::endl;
        // lock.unlock();  // Can unlock manually early
    }

    // Way 3: try locking (try_to_lock)
    {
        std::unique_lock<std::mutex> lock(resource_mtx, std::try_to_lock);
        if (lock.owns_lock()) {
            std::cout << "Thread " << id << " got the lock!" << std::endl;
            resource += 30;
        } else {
            std::cout << "Thread " << id
                      << " could not get lock, doing other work..."
                      << std::endl;
        }
    }

    // Way 4: adopt an already-locked mutex
    {
        resource_mtx.lock();
        // ... do something while holding the lock
        std::unique_lock<std::mutex> lock(resource_mtx, std::adopt_lock);
        // Now lock manages this mutex and will unlock it on destruction
        resource += 40;
    }

    // unique_lock also supports move semantics
    std::unique_lock<std::mutex> lock1(resource_mtx);
    // std::unique_lock<std::mutex> lock2 = lock1;  // Error! Not copyable
    std::unique_lock<std::mutex> lock3(std::move(lock1));  // OK, movable
}

int main() {
    std::thread t1(flexible_locker, 1);
    std::thread t2(flexible_locker, 2);
    t1.join(); t2.join();
    std::cout << "Final resource: " << resource << std::endl;
    return 0;
}


IV. std::scoped_lock (C++17): lock management with deadlock avoidance

// ============================================================
// Example 12-10: scoped_lock usage
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>

std::mutex mtx_a, mtx_b;

void deadlock_prone(int id) {
    // Dangerous: if thread 1 locks mtx_a then mtx_b,
    // and thread 2 locks mtx_b then mtx_a, they may deadlock!
    // Use std::lock or std::scoped_lock to avoid this
    {
        std::scoped_lock lock(mtx_a, mtx_b);
        // scoped_lock is equivalent to std::lock(mtx_a, mtx_b) and then adopting both locks
        // std::lock uses a deadlock-avoidance algorithm (try_lock + back-off)
        std::cout << "Thread " << id << " locked both mutexes safely."
                  << std::endl;
    }
}

void manual_deadlock_avoidance(int id) {
    // Manually avoiding deadlock in C++11/14
    std::lock(mtx_a, mtx_b);  // Lock multiple mutexes atomically
    std::lock_guard<std::mutex> lock_a(mtx_a, std::adopt_lock);
    std::lock_guard<std::mutex> lock_b(mtx_b, std::adopt_lock);
    // or use unique_lock
    std::cout << "Thread " << id << " (manual) locked both." << std::endl;
}

int main() {
    std::thread t1(deadlock_prone, 1);
    std::thread t2(deadlock_prone, 2);
    std::thread t3(manual_deadlock_avoidance, 3);
    t1.join(); t2.join(); t3.join();
    return 0;
}


V. std::recursive_mutex: the recursive mutex

// ============================================================
// Example 12-11: recursive_mutex usage
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>

class RecursiveCounter {
private:
    std::recursive_mutex mtx_;
    int value_ = 0;

public:
    void increment() {
        std::lock_guard<std::recursive_mutex> lock(mtx_);
        ++value_;
    }

    void add_multiple(int count) {
        std::lock_guard<std::recursive_mutex> lock(mtx_);
        for (int i = 0; i < count; ++i) {
            increment();  // Calls increment, which acquires the same mutex again
            // With a plain mutex, this would deadlock!
            // recursive_mutex lets the same thread acquire it multiple times
        }
    }

    int get_value() const {
        std::lock_guard<std::recursive_mutex> lock(mtx_);
        return value_;
    }
};

int main() {
    RecursiveCounter rc;
    rc.add_multiple(5);  // Calls increment 5 times
    std::cout << "Value: " << rc.get_value() << std::endl;  // 5
    return 0;
}


VI. std::timed_mutex: a mutex with timeouts

// ============================================================
// Example 12-12: timed_mutex usage
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>
#include <chrono>

std::timed_mutex tmtx;

void try_lock_for_example(int id) {
    // try_lock_for: try to acquire the lock within the given duration
    auto timeout = std::chrono::milliseconds(100);
    if (tmtx.try_lock_for(timeout)) {
        std::cout << "Thread " << id
                  << " got lock within 100ms." << std::endl;
        std::this_thread::sleep_for(std::chrono::milliseconds(200));
        tmtx.unlock();
    } else {
        std::cout << "Thread " << id
                  << " timed out after 100ms." << std::endl;
    }
}

void try_lock_until_example(int id) {
    // try_lock_until: try to acquire the lock before the given time point
    auto deadline = std::chrono::steady_clock::now()
                    + std::chrono::milliseconds(150);
    if (tmtx.try_lock_until(deadline)) {
        std::cout << "Thread " << id
                  << " got lock before deadline." << std::endl;
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        tmtx.unlock();
    } else {
        std::cout << "Thread " << id
                  << " missed the deadline." << std::endl;
    }
}

int main() {
    std::thread t1(try_lock_for_example, 1);
    std::thread t2(try_lock_until_example, 2);
    t1.join(); t2.join();
    return 0;
}

// ============================================================

// ============================================================
#include <iostream>
#include <thread>
#include <mutex>
#include <vector>

class Singleton {
private:
    static std::unique_ptr<Singleton> instance_;
    static std::once_flag init_flag_;

    ~Singleton() { std::cout << "Singleton destroyed." << std::endl; }

    Singleton& operator=(const Singleton&) = delete;

        std::call_once(init_flag_, []() {

        });

    }

}

// ============================================================
// Example 12-14: Basic condition_variable usage
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <queue>
#include <chrono>

std::mutex mtx;
std::condition_variable cv;
std::queue<int> data_queue;
bool done = false;

void producer() {
    for (int i = 1; i <= 10; ++i) {
        {
            std::lock_guard<std::mutex> lock(mtx);
            data_queue.push(i);
            std::cout << "[Producer] Pushed: " << i << std::endl;
        }
        // Notify one waiting consumer (notify_one)
        // Notifying after or before unlocking both work, but notifying after unlocking may slightly reduce context switches
        cv.notify_one();
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
    }

    // Notify all consumers that the producer is done
    {
        std::lock_guard<std::mutex> lock(mtx);
        done = true;
    }
    cv.notify_all();
}

void consumer(int id) {
    while (true) {
        std::unique_lock<std::mutex> lock(mtx);

        // How wait behaves:
        // 1. Check the condition (lambda returns false), release the mutex, the thread waits
        // 2. On notification, reacquire the mutex and check the condition
        // 3. When the condition is true, wait returns (with the mutex still locked)
        cv.wait(lock, []{ return !data_queue.empty() || done; });

        if (!data_queue.empty()) {
            int value = data_queue.front();
            data_queue.pop();
            std::cout << "[Consumer " << id << "] Popped: "
                      << value << std::endl;
        }

        if (done && data_queue.empty()) {
            break;
        }
    }
}

int main() {
    std::thread p(producer);
    std::thread c1(consumer, 1);
    std::thread c2(consumer, 2);

    p.join(); c1.join(); c2.join();
    std::cout << "All done." << std::endl;
    return 0;
}


[About Spurious Wakeups]

A spurious wakeup means a thread waiting on a condition variable may be woken up even
though no notify was sent. This is due to implementation details at the OS and hardware
level. Therefore, a condition variable wait must re-check the condition using a loop or
a wait with a predicate:

// Wrong -- can be fooled by a spurious wakeup
while (true) {
    std::unique_lock<std::mutex> lock(mtx);
    // cv.wait(lock); // No predicate!
    // process(data); // Assumes there is data once woken -- wrong!
}

// Correct approach 1 -- use wait with a predicate
cv.wait(lock, []{ return !data_queue.empty(); });

// Correct approach 2 -- a manual while loop
while (data_queue.empty()) {
    cv.wait(lock);
}

[notify_one vs notify_all]
- notify_one: wakes up one of the threads waiting on the condition variable (which one is unspecified)
  Use case: the producer produces one task at a time, and one consumer can handle it
- notify_all: wakes up all waiting threads
  Use case: a state change affects all waiting threads (e.g. reset, shutdown)


[Condition Variable Waits with Timeouts]

// ============================================================
// Example 12-15: wait_for / wait_until
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <chrono>

std::mutex mtx2;
std::condition_variable cv2;
bool ready = false;

void waiter() {
    std::unique_lock<std::mutex> lock(mtx2);

    // wait_for: wait for the given duration
    std::cout << "Waiting for 500ms..." << std::endl;
    if (cv2.wait_for(lock, std::chrono::milliseconds(500))) {
        std::cout << "Condition satisfied!" << std::endl;
    } else {
        std::cout << "Timeout! ready is still false." << std::endl;
        std::cout << "ready = " << ready << std::endl;
    }

    // wait_until: wait until the given time point
    auto deadline = std::chrono::steady_clock::now()
                    + std::chrono::milliseconds(300);
    if (cv2.wait_until(lock, deadline, []{ return ready; })) {
        std::cout << "Condition met before deadline." << std::endl;
    } else {
        std::cout << "Deadline passed." << std::endl;
    }
}

int main() {
    std::thread w(waiter);
    std::this_thread::sleep_for(std::chrono::milliseconds(200));
    {
        std::lock_guard<std::mutex> lock(mtx2);
        ready = true;
    }
    cv2.notify_one();
    w.join();
    return 0;
}

// ============================================================
// Example 12-16: atomic<T> basic types
// ============================================================
#include <iostream>
#include <thread>
#include <atomic>
#include <vector>
#include <cassert>

std::atomic<int> atomic_counter{0};
std::atomic<bool> atomic_flag_ready{false};
std::atomic<long long> atomic_large{0};

// Main atomic operations
void atomic_basics() {
    // store: atomic write
    atomic_counter.store(42, std::memory_order_relaxed);

    // load: atomic read
    int val = atomic_counter.load(std::memory_order_relaxed);
    std::cout << "Stored value: " << val << std::endl;

    // fetch_add: atomic addition, returns the old value
    int old = atomic_counter.fetch_add(10);
    std::cout << "After fetch_add(10): old=" << old
              << ", new=" << atomic_counter.load() << std::endl;

    // fetch_sub: atomic subtraction
    atomic_counter.fetch_sub(5);
    std::cout << "After fetch_sub(5): " << atomic_counter.load() << std::endl;

    // fetch_and, fetch_or, fetch_xor: atomic bitwise operations
    int bit_flags = 0b1010;
    std::atomic<int> atomic_flags{bit_flags};
    atomic_flags.fetch_or(0b0101);
    std::cout << "After fetch_or: " << atomic_flags.load()
              << " (binary: 1111)" << std::endl;

    // Compare-And-Swap (CAS)
    int expected = 47;
    // If atomic_counter's current value equals expected, set it to 53 and return true
    // Otherwise expected is updated to the current value and false is returned
    bool success = atomic_counter.compare_exchange_strong(expected, 53);
    std::cout << "CAS success: " << success
              << ", expected now: " << expected << std::endl;

    // compare_exchange_weak: may fail spuriously, but performs better
    // Usually used in a loop
    int expected_weak = atomic_counter.load();
    while (!atomic_counter.compare_exchange_weak(expected_weak, 100)) {
        // expected_weak is updated to the current value
        // Retry in the loop
    }
    std::cout << "After CAS loop: " << atomic_counter.load() << std::endl;

    // exchange: atomically swap the value, returns the old value
    old = atomic_counter.exchange(999);
    std::cout << "exchange: old=" << old
              << ", new=" << atomic_counter.load() << std::endl;

    // is_lock_free: checks whether this atomic type uses hardware atomic instructions (lock-free)
    std::cout << "atomic<int> is lock-free: "
              << atomic_counter.is_lock_free() << std::endl;
    // Basic types such as int/pointer/bool are usually lock-free
    // Large structs may not be

    // Special operations for the bool type
    atomic_flag_ready.store(false);
    bool old_bool = atomic_flag_ready.exchange(true);
    std::cout << "atomic<bool> exchange: old=" << old_bool << std::endl;
}


// ============================================================
// Example 12-17: atomic_flag, the atomic flag (lock-free boolean)
// ============================================================
#include <atomic>
#include <thread>

class SpinLock {
private:
    std::atomic_flag flag_ = ATOMIC_FLAG_INIT;

public:
    void lock() {
        // test_and_set: if flag is false, sets it to true and returns false (lock acquired)
        // If flag is already true, returns true (lock is held), so keep spinning
        while (flag_.test_and_set(std::memory_order_acquire)) {
            // Busy-Waiting
            // On x86, _mm_pause() can be used to reduce power consumption
            // yield can also be used to give other threads a chance
            std::this_thread::yield();
        }
    }

    void unlock() {
        flag_.clear(std::memory_order_release);
    }
};

// Example using the spin lock
SpinLock spin_lock;
int spin_counter = 0;

void spin_worker() {
    for (int i = 0; i < 1000; ++i) {
        spin_lock.lock();
        ++spin_counter;
        spin_lock.unlock();
    }
}

void test_spinlock() {
    std::thread t1(spin_worker);
    std::thread t2(spin_worker);
    std::thread t3(spin_worker);
    t1.join(); t2.join(); t3.join();
    std::cout << "SpinLock result: " << spin_counter
              << " (expected 3000)" << std::endl;
}


// ============================================================
// Example 12-18: The six memory orders (memory_order)
// ============================================================
#include <iostream>
#include <thread>
#include <atomic>

// There are six memory_order values, from weakest to strongest constraint:

// 1. memory_order_relaxed: guarantees atomicity only, no ordering
//    Use case: counters and other scenarios that don't depend on ordering
std::atomic<int> relaxed_count{0};

void relaxed_increment() {
    for (int i = 0; i < 1000; ++i) {
        relaxed_count.fetch_add(1, std::memory_order_relaxed);
    }
}

// 2. memory_order_acquire: acquire semantics
//    No memory operation after this load is reordered before it
//    Used to read shared flags

// 3. memory_order_release: release semantics
//    No memory operation before this store is reordered after it
//    Used to set shared flags

// 4. memory_order_acq_rel: both acquire and release semantics
//    Used for read-modify-write operations (such as fetch_add)

// 5. memory_order_seq_cst: sequential consistency (the default)
//    All threads see all atomic operations in the same order
//    Most expensive, but easiest to reason about

// Classic acquire/release pairing (guarding shared data)
std::atomic<bool> data_ready{false};
std::string shared_string;  // Non-atomic data

void writer_thread() {
    shared_string = "Important data from writer";  // Non-atomic write
    // release: guarantees the write to shared_string is visible to all threads before data_ready
    data_ready.store(true, std::memory_order_release);
}

void reader_thread() {
    // acquire: guarantees operations after reading data_ready are not executed before the read
    while (!data_ready.load(std::memory_order_acquire)) {
        std::this_thread::yield();
    }
    // At this point we are guaranteed to see the shared_string written by the writer
    std::cout << "Reader sees: " << shared_string << std::endl;
}

// 6. memory_order_consume: data-dependency ordering (rarely used; almost all compilers implement it as acquire)
//    C++17 discourages its use

void memory_order_demo() {
    std::thread w(writer_thread);
    std::thread r(reader_thread);
    w.join(); r.join();

    std::cout << "memory_order demo complete." << std::endl;
}


int main() {
    atomic_basics();
    test_spinlock();
    memory_order_demo();
    return 0;
}

// ============================================================
// Example 12-19: Comparing async, future, promise, and packaged_task
// ============================================================
#include <iostream>
#include <thread>
#include <future>
#include <chrono>
#include <vector>
#include <numeric>

// ---- Way 1: std::async, automatic asynchronous execution ----
int compute_sum(int from, int to) {
    int sum = 0;
    for (int i = from; i <= to; ++i) {
        sum += i;
        std::this_thread::sleep_for(std::chrono::milliseconds(1));
    }
    return sum;
}

void async_example() {
    // std::launch::async: force execution on a new thread
    std::future<int> result1 = std::async(std::launch::async,
                                           compute_sum, 1, 100);

    // std::launch::deferred: deferred execution; runs on the current thread at get/wait
    std::future<int> result2 = std::async(std::launch::deferred,
                                           compute_sum, 1, 50);

    // Default (async | deferred): up to the implementation
    std::future<int> result3 = std::async(compute_sum, 1, 200);

    std::cout << "Waiting for async results..." << std::endl;

    // get(): blocks waiting for the result; can be called only once, after which the future is invalid
    std::cout << "Result1 (async): " << result1.get() << std::endl;

    // wait(): blocks until the task completes, without retrieving the result
    result2.wait();
    std::cout << "Result2 (deferred): " << result2.get() << std::endl;

    // wait_for: wait with a timeout
    auto status = result3.wait_for(std::chrono::milliseconds(10));
    if (status == std::future_status::ready) {
        std::cout << "Result3 is ready: " << result3.get() << std::endl;
    } else if (status == std::future_status::timeout) {
        std::cout << "Result3 is still running..." << std::endl;
        // Can keep waiting
        std::cout << "Result3 finally: " << result3.get() << std::endl;
    } else if (status == std::future_status::deferred) {
        std::cout << "Result3 is deferred." << std::endl;
    }
}


// ---- Way 2: std::promise, setting the result manually ----

        // Set the result: promise.set_value(...)
        prom.set_value(42);
        // promise.set_value(43);  // Error! Can only be set once

        // Or set an exception
        // prom.set_exception(
        //     std::make_exception_ptr(std::runtime_error("Error!"))
        // );
    });

    std::cout << "Waiting for promise..." << std::endl;
    int result = fut.get();  // Blocks until the promise sets a result
    std::cout << "Promise result: " << result << std::endl;

}


// ---- Way 3: std::packaged_task wraps a callable ----
void packaged_task_example() {
    // packaged_task wraps a function and provides a future
    std::packaged_task<int(int, int)> task([](int a, int b) {
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        return a * b;
    });

    std::future<int> fut = task.get_future();

    // Run the task on a new thread
    std::thread worker(std::move(task), 6, 7);  // task is moved and unusable afterwards
    // It can also run directly on the current thread: task(6, 7);

// ---- Way 4: std::shared_future, retrieving the result multiple times ----
void shared_future_example() {
    std::promise<std::string> prom;
    // shared_future can be copied and read multiple times
    std::shared_future<std::string> shared_fut = prom.get_future().share();

    std::thread setter([&prom]() {
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        prom.set_value("Shared result");
    });

    auto reader = [shared_fut](int id) {
        // shared_future's get() returns a const reference and can be called multiple times
        std::cout << "Reader " << id << " got: "
                  << shared_fut.get() << std::endl;
    };

    std::thread r1(reader, 1);
    std::thread r2(reader, 2);
    std::thread r3(reader, 3);

    setter.join(); r1.join(); r2.join(); r3.join();
}


// ---- std::future's valid() and share() ----
void future_management() {
    std::future<int> fut = std::async([]() { return 100; });

    // valid(): checks whether the future is associated with a shared state (i.e. can be get)
    std::cout << "future is valid: " << fut.valid() << std::endl;  // true

    int result = fut.get();
    std::cout << "future is valid after get: "
              << fut.valid() << std::endl;  // false

    // share(): converts the future into a shared_future
    std::future<int> fut2 = std::async([]() { return 200; });
    std::shared_future<int> shared = fut2.share();
    // fut2.valid() is now false

    std::cout << "Shared future result: " << shared.get() << std::endl;
}

// ============================================================

#include <iostream>

#include <mutex>
#include <vector>

// thread_local variables: each thread has its own independent copy

// Global thread_local variable
thread_local int tls_counter = 0;

void worker_with_tls(int id) {

[Example 1] Concurrent Counter - Website Visit Statistics
================================================================================

Scenario: an e-commerce website needs to count total visits. Multiple front-end server
threads increment the visit counter at the same time. Without synchronization, the
count would be inaccurate.

// ============================================================
// Case 12-1: Concurrent visit counter (complete code)
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>
#include <vector>
#include <chrono>
#include <atomic>
#include <iomanip>

namespace visit_counter {

// Isolated with a namespace (Chapter 1: namespaces)
class VisitCounter {
private:
    mutable std::mutex mtx_;
    long long total_visits_ = 0;
    std::atomic<long long> atomic_visits_{0};

public:
    // Locked increment (thread-safe)
    void increment_safe() {
        std::lock_guard<std::mutex> lock(mtx_);
        ++total_visits_;
    }

    // Unlocked increment (not thread-safe)
    void increment_unsafe() {
        ++total_visits_;
    }

    // Atomic increment (no mutex needed, better performance)
    void increment_atomic() {
        atomic_visits_.fetch_add(1, std::memory_order_relaxed);
    }

    long long get_safe() const {
        std::lock_guard<std::mutex> lock(mtx_);
        return total_visits_;
    }

    long long get_unsafe() const { return total_visits_; }
    long long get_atomic() const { return atomic_visits_.load(); }

    void reset() {
        std::lock_guard<std::mutex> lock(mtx_);
        total_visits_ = 0;
        atomic_visits_.store(0);
    }
};

void simulate_visits(VisitCounter& counter,
                     int thread_id, int num_visits) {
    for (int i = 0; i < num_visits; ++i) {
        counter.increment_unsafe();
        counter.increment_safe();
        counter.increment_atomic();
        // Simulate the interval between visits
        std::this_thread::sleep_for(std::chrono::microseconds(10));
    }
}

void run() {
    const int NUM_THREADS = 8;
    const int VISITS_PER_THREAD = 10000;

    VisitCounter counter;

    std::cout << "\n========== Website Visit Counter ==========\n";
    std::cout << "Simulating " << NUM_THREADS << " front-end server threads\n";
    std::cout << "Each thread records " << VISITS_PER_THREAD << " visits\n\n";

    std::vector<std::thread> threads;
    for (int i = 0; i < NUM_THREADS; ++i) {
        threads.emplace_back(simulate_visits,
                             std::ref(counter),
                             i, VISITS_PER_THREAD);
    }

    for (auto& t : threads) t.join();

    long long expected = static_cast<long long>(NUM_THREADS)
                         * VISITS_PER_THREAD;
    std::cout << "Expected total visits: " << expected << "\n";
    std::cout << "Unlocked result:      " << counter.get_unsafe()
              << " (wrong! updates lost to data races)\n";
    std::cout << "Mutex result:         " << counter.get_safe()
              << " (correct!)\n";
    std::cout << "Atomic result:        " << counter.get_atomic()
              << " (correct! and faster)\n";
    std::cout << "========================================\n";
}

} // namespace visit_counter


[Example 2] Producer-Consumer Model - A Restaurant Kitchen
================================================================================

Scenario: in a restaurant kitchen, chefs (producers) cook dishes and place them in the
serving window (the buffer), and waiters (consumers) take dishes from the window to the
tables. Implemented with a ring buffer.

// ============================================================
// Case 12-2: Restaurant kitchen producer-consumer model
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <vector>
#include <queue>
#include <chrono>
#include <string>
#include <random>
#include <sstream>

namespace restaurant {

// Ring buffer class template (Chapter 8: templates)
template <typename T, size_t Capacity>
class RingBuffer {
private:
    T buffer_[Capacity];
    size_t head_ = 0;  // Consumer position
    size_t tail_ = 0;  // Producer position
    size_t count_ = 0;

    mutable std::mutex mtx_;
    std::condition_variable cv_not_full_;
    std::condition_variable cv_not_empty_;

public:
    // Producer puts data in
    void put(const T& item) {
        std::unique_lock<std::mutex> lock(mtx_);
        // If the buffer is full, wait for a consumer to take something
        cv_not_full_.wait(lock, [this]() {
            return count_ < Capacity;
        });

        buffer_[tail_] = item;
        tail_ = (tail_ + 1) % Capacity;
        ++count_;

        // Notify a consumer that new data is available
        cv_not_empty_.notify_one();
    }

    // Consumer takes data out
    T take() {
        std::unique_lock<std::mutex> lock(mtx_);
        // If the buffer is empty, wait for a producer to put something
        cv_not_empty_.wait(lock, [this]() {
            return count_ > 0;
        });

        T item = buffer_[head_];
        head_ = (head_ + 1) % Capacity;
        --count_;

        // Notify a producer that a slot is free
        cv_not_full_.notify_one();
        return item;
    }

    size_t size() const {
        std::lock_guard<std::mutex> lock(mtx_);
        return count_;
    }
};

// Dish class (Chapter 4: classes and objects)
class Dish {
private:
    std::string name_;
    int prep_time_;     // Preparation time (ms)
    int dish_id_;
    static std::atomic<int> next_id_;

public:
    Dish(const std::string& name, int prep_time)
        : name_(name), prep_time_(prep_time), dish_id_(next_id_++) {}

    std::string get_name() const { return name_; }
    int get_prep_time() const { return prep_time_; }
    int get_id() const { return dish_id_; }

    std::string to_string() const {
        std::ostringstream oss;
        oss << "[" << dish_id_ << "]" << name_
            << "(" << prep_time_ << "ms)";
        return oss.str();
    }
};

std::atomic<int> Dish::next_id_{1};

// Chef class -- the producer (Chapter 4: classes and objects; Chapter 6: polymorphism -- extensible Chef subclasses)
class Chef {
private:
    int id_;
    RingBuffer<Dish, 10>& buffer_;
    std::atomic<bool>& running_;
    std::vector<std::string> menu_;
    mutable std::mt19937 rng_;

public:
    Chef(int id, RingBuffer<Dish, 10>& buffer,
         std::atomic<bool>& running)
        : id_(id), buffer_(buffer), running_(running),
          menu_({"Kung Pao Chicken", "Yu-Shiang Pork", "Braised Pork Belly", "Steamed Fish",
                 "Mapo Tofu", "Sweet and Sour Ribs"}),
          rng_(std::random_device{}()) {}

    void cook() {
        while (running_.load(std::memory_order_relaxed)) {
            // Pick a random dish
            std::uniform_int_distribution<size_t> dist(0, menu_.size() - 1);
            std::uniform_int_distribution<int> time_dist(50, 300);
            std::string dish_name = menu_[dist(rng_)];
            int prep_time = time_dist(rng_);

            Dish dish(dish_name, prep_time);
            std::cout << "[Chef " << id_ << "] Cooking: "
                      << dish.to_string() << std::endl;

            // Simulate cooking time
            std::this_thread::sleep_for(
                std::chrono::milliseconds(prep_time));

            std::cout << "[Chef " << id_ << "] Finished: "
                      << dish.to_string() << std::endl;
            buffer_.put(dish);
        }
        std::cout << "[Chef " << id_ << "] Off duty." << std::endl;
    }
};

// Waiter class -- the consumer
class Waiter {
private:
    int id_;
    RingBuffer<Dish, 10>& buffer_;
    std::atomic<bool>& running_;
    std::atomic<int>& dishes_served_;

public:
    Waiter(int id, RingBuffer<Dish, 10>& buffer,
           std::atomic<bool>& running,
           std::atomic<int>& dishes_served)
        : id_(id), buffer_(buffer), running_(running),
          dishes_served_(dishes_served) {}

    void serve() {
        while (running_.load(std::memory_order_relaxed)) {
            Dish dish = buffer_.take();
            dishes_served_.fetch_add(1, std::memory_order_relaxed);
            std::cout << "  [Waiter " << id_ << "] Serving: "
                      << dish.to_string()
                      << " (" << dishes_served_.load() << " served so far)"
                      << std::endl;
            // Simulate serving time
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
        }
        std::cout << "  [Waiter " << id_ << "] Off duty." << std::endl;
    }
};

void run() {
    std::cout << "\n========== Restaurant Kitchen Simulation ==========\n\n";

    RingBuffer<Dish, 10> serving_window;  // Serving window
    std::atomic<bool> running{true};
    std::atomic<int> dishes_served{0};

    // 2 chefs (producers)
    std::vector<std::thread> chefs;
    Chef chef1(1, serving_window, running);
    Chef chef2(2, serving_window, running);
    chefs.emplace_back(&Chef::cook, &chef1);
    chefs.emplace_back(&Chef::cook, &chef2);

    // 3 waiters (consumers)
    std::vector<std::thread> waiters;
    Waiter waiter1(1, serving_window, running, dishes_served);
    Waiter waiter2(2, serving_window, running, dishes_served);
    Waiter waiter3(3, serving_window, running, dishes_served);
    waiters.emplace_back(&Waiter::serve, &waiter1);
    waiters.emplace_back(&Waiter::serve, &waiter2);
    waiters.emplace_back(&Waiter::serve, &waiter3);

    // Stop after running for 5 seconds
    std::this_thread::sleep_for(std::chrono::seconds(5));
    std::cout << "\n[System] Closing time, we're closed!\n";
    running.store(false);

    // Finish all threads
    for (auto& t : chefs) t.join();
    for (auto& t : waiters) t.join();

    std::cout << "\n[System] " << dishes_served.load()
              << " dishes served today. Great work, everyone!\n";
    std::cout << "=========================================\n";
}

// File search engine
class FileSearcher {
private:
    SearchResultCollector results_;
    std::string keyword_;         // Search keyword
    std::mutex dir_queue_mtx_;
    std::queue<std::string> dir_queue_;
    std::atomic<size_t> files_scanned_{0};
    std::atomic<size_t> dirs_processed_{0};
    std::atomic<bool> stop_requested_{false};

public:
    FileSearcher(const std::string& keyword) : keyword_(keyword) {}

    // Add a directory to search
    void add_directory(const std::string& path) {
        std::lock_guard<std::mutex> lock(dir_queue_mtx_);
        dir_queue_.push(path);
    }

    // Search a single file (check whether it matches the keyword)
    bool match_file(const std::string& filepath) {
        // Simplified: check whether the file name contains the keyword
        size_t pos = filepath.rfind('/');
        std::string filename = (pos != std::string::npos)
                               ? filepath.substr(pos + 1)
                               : filepath;
        // Case-insensitive match
        std::string lower_name = filename;
        std::string lower_key = keyword_;
        std::transform(lower_name.begin(), lower_name.end(),
                       lower_name.begin(), ::tolower);
        std::transform(lower_key.begin(), lower_key.end(),
                       lower_key.begin(), ::tolower);
        return lower_name.find(lower_key) != std::string::npos;
    }

    // Search the contents of a single file (Chapter 10: file I/O)
    bool match_file_content(const std::string& filepath) {
        // Read the file contents and search for the keyword (simplified demo)
        // std::ifstream file(filepath);
        // std::string content((std::istreambuf_iterator<char>(file)),
        //                      std::istreambuf_iterator<char>());
        // return content.find(keyword_) != std::string::npos;
        return false;  // The demo does not actually read files
    }

    // Worker thread function (searches the directory tree)
    void search_worker(int thread_id) {
        while (!stop_requested_.load(std::memory_order_relaxed)) {
            std::string current_dir;

            {
                std::lock_guard<std::mutex> lock(dir_queue_mtx_);
                if (dir_queue_.empty()) break;  // No more directories
                current_dir = dir_queue_.front();
                dir_queue_.pop();
            }

            std::cout << "[Thread " << thread_id
                      << "] Searching directory: " << current_dir << std::endl;

            dirs_processed_.fetch_add(1, std::memory_order_relaxed);

            // Get the files in this directory
            auto files = DirectoryLister::files(current_dir);

            // Get the subdirectories and add them to the queue
            auto subdirs = DirectoryLister::subdirectories(current_dir);
            {
                std::lock_guard<std::mutex> lock(dir_queue_mtx_);
                for (const auto& sub : subdirs) {
                    dir_queue_.push(current_dir + "/" + sub);
                }
            }

            // Check whether each file matches
            for (const auto& file : files) {
                std::string fullpath = current_dir + "/" + file;
                if (match_file(fullpath)) {
                    SearchResult sr;
                    sr.filepath = fullpath;
                    sr.filesize = 0;  // A real application would get the file size
                    sr.directory = current_dir;
                    results_.add(sr);
                }
                files_scanned_.fetch_add(1, std::memory_order_relaxed);
            }
        }
    }

    // Parallel search (using async)
    std::future<void> search_parallel_async() {
        return std::async(std::launch::async,
                          &FileSearcher::search_with_threads, this, 4);
    }

    // Launch multiple threads to search
    void search_with_threads(int num_threads) {
        std::vector<std::thread> threads;
        for (int i = 0; i < num_threads; ++i) {
            threads.emplace_back(&FileSearcher::search_worker, this, i);

        for (auto& t : threads) t.join();

        std::cout << "Matching files:\n";

            std::cout << "  " << r.to_string() << "\n";
        }
        std::cout << "==========================\n";

    void save_report(const std::string& report_path) const {

        }

    }

        stop_requested_.store(true, std::memory_order_relaxed);

};

[Example 4] Thread Pool Implementation - Web Server Request Handling
================================================================================

Scenario: a web server needs to handle a large number of concurrent HTTP requests.
Creating a new thread for every request is too expensive, so a thread pool creates a
fixed number of worker threads up front and reuses them.

// ============================================================
// Case 12-4: A general-purpose thread pool implementation
// ============================================================
#include <iostream>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <vector>
#include <queue>
#include <functional>
#include <future>
#include <memory>
#include <stdexcept>
#include <type_traits>
#include <atomic>
#include <chrono>

namespace thread_pool_demo {

// Abstract task interface (Chapter 6: polymorphism)
class ITask {
public:
    virtual ~ITask() = default;
    virtual void execute() = 0;
};

// Thread pool class
class ThreadPool {
private:
    std::vector<std::thread> workers_;
    std::queue<std::function<void()>> tasks_;  // Task queue
    std::mutex queue_mtx_;
    std::condition_variable cv_;
    std::atomic<bool> stop_{false};
    std::atomic<size_t> active_tasks_{0};

    void worker_thread() {
        while (true) {
            std::function<void()> task;

            {
                std::unique_lock<std::mutex> lock(queue_mtx_);
                // Wait until a task is available or the pool is stopped
                cv_.wait(lock, [this]() {
                    return stop_.load() || !tasks_.empty();
                });

                if (stop_.load() && tasks_.empty()) {
                    return;  // Thread exits
                }

                task = std::move(tasks_.front());
                tasks_.pop();
                active_tasks_.fetch_add(1);
            }

            try {
                task();
            } catch (const std::exception& e) {
                std::cerr << "[ThreadPool] Task exception: "
                          << e.what() << std::endl;
            } catch (...) {
                std::cerr << "[ThreadPool] Unknown exception\n";
            }

            active_tasks_.fetch_sub(1);
        }
    }

public:
    explicit ThreadPool(size_t num_threads)
        : stop_(false) {
        if (num_threads == 0) {
            throw std::invalid_argument(
                "Number of threads must be greater than 0");
        }
        workers_.reserve(num_threads);
        for (size_t i = 0; i < num_threads; ++i) {
            workers_.emplace_back(
                &ThreadPool::worker_thread, this);
        }
    }

    ~ThreadPool() {
        shutdown();
    }

    // Non-copyable and non-movable
    ThreadPool(const ThreadPool&) = delete;
    ThreadPool& operator=(const ThreadPool&) = delete;

    // Submit a task (supports any callable and arguments; Chapter 8: variadic templates)
    template <typename F, typename... Args>
    auto submit(F&& f, Args&&... args)
        -> std::future<typename std::invoke_result<F, Args...>::type> {
        using return_type = typename std::invoke_result<F, Args...>::type;

        // Create a packaged_task wrapped in a shared_ptr
        // because packaged_task is not copyable, while the lambda must go into a function<void()> queue
        auto task = std::make_shared<std::packaged_task<return_type()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...)
        );

        std::future<return_type> result_future = task->get_future();

        {
            std::lock_guard<std::mutex> lock(queue_mtx_);
            if (stop_.load()) {
                throw std::runtime_error(
                    "Cannot submit a task to a stopped thread pool");
            }
            tasks_.emplace([task]() { (*task)(); });
        }

        cv_.notify_one();
        return result_future;
    }

    // Wait for all tasks to complete
    void wait_all() {
        while (true) {
            std::lock_guard<std::mutex> lock(queue_mtx_);
            if (tasks_.empty() && active_tasks_.load() == 0) {
                break;
            }
            std::this_thread::yield();
        }
    }

    // Graceful shutdown
    void shutdown() {
        stop_.store(true);
        cv_.notify_all();
        for (std::thread& worker : workers_) {
            if (worker.joinable()) {
                worker.join();
            }
        }
        workers_.clear();
    }

    size_t queue_size() const {
        std::lock_guard<std::mutex> lock(queue_mtx_);
        return tasks_.size();
    }

    size_t active_count() const {
        return active_tasks_.load();
    }
};

void handle_request(int request_id, int processing_time_ms) {
    std::cout << "  [Request #" << request_id
              << "] Processing started (thread "
              << std::this_thread::get_id() << ")\n";
    std::this_thread::sleep_for(
        std::chrono::milliseconds(processing_time_ms));
    std::cout << "  [Request #" << request_id << "] Processing complete\n";
}

int compute_heavy_task(int input, int iterations) {
    long long result = 0;  // Use long long to prevent overflow
    for (int i = 0; i < iterations; ++i) {
        result += (input * i) % 10007;
    }
    return static_cast<int>(result % 10000);
}

// Forward declaration
class Account;

// Transaction type enum
enum class TransactionType {
    DEPOSIT,
    WITHDRAW,
    TRANSFER
};

// Transaction log struct
struct TransactionLog {
    std::chrono::system_clock::time_point timestamp;
    TransactionType type;
    int from_account;
    int to_account;
    double amount;
    double from_balance_after;
    double to_balance_after;
};

// Account class (Chapter 4: classes and objects)
class Account {
private:
    int account_id_;
    std::string owner_name_;
    double balance_;
    mutable std::mutex mtx_;

public:
    Account(int id, const std::string& name, double initial_balance)
        : account_id_(id), owner_name_(name), balance_(initial_balance) {}

    Account(const Account&) = delete;
    Account& operator=(const Account&) = delete;
    Account(Account&&) = default;
    Account& operator=(Account&&) = default;

    int get_id() const { return account_id_; }
    std::string get_owner() const { return owner_name_; }

    // Get the balance (thread-safe)
    double get_balance() const {
        std::lock_guard<std::mutex> lock(mtx_);
        return balance_;
    }

    // Deposit (thread-safe)
    void deposit(double amount) {
        if (amount <= 0) {
            throw std::invalid_argument("Deposit amount must be positive");
        }
        std::lock_guard<std::mutex> lock(mtx_);
        balance_ += amount;
    }

    // Withdraw (thread-safe)
    void withdraw(double amount) {
        if (amount <= 0) {
            throw std::invalid_argument("Withdrawal amount must be positive");
        }
        std::lock_guard<std::mutex> lock(mtx_);
        if (balance_ < amount) {
            throw std::runtime_error("Insufficient balance");
        }
        balance_ -= amount;
    }

    // Get the mutex for locking (used for external deadlock avoidance)
    std::mutex& get_mutex() const { return mtx_; }

    // Adjust the balance internally (the caller holds the lock)
    void adjust_balance(double delta) {
        balance_ += delta;
    }
};

// Banking transaction system (Chapters 4/5/6: OOP design)
class BankingSystem {
private:
    std::vector<std::unique_ptr<Account>> accounts_;
    mutable std::mutex log_mtx_;
    std::vector<TransactionLog> transaction_logs_;
    std::atomic<unsigned long long> total_transactions_{0};
    std::atomic<double> total_volume_{0.0};
    std::atomic<bool> running_{true};
    std::string log_file_path_;

    // Write a transaction log to file (Chapter 10: file I/O)
    void write_log_to_file(const TransactionLog& log) {
        // A real production environment may need asynchronous writes or durable flushing
        std::lock_guard<std::mutex> lock(log_mtx_);
        transaction_logs_.push_back(log);
    }

    static std::string timestamp_to_string(
        const std::chrono::system_clock::time_point& tp) {
        auto time_t_val = std::chrono::system_clock::to_time_t(tp);
        std::ostringstream oss;
        oss << std::put_time(std::localtime(&time_t_val), "%Y-%m-%d %H:%M:%S");
        return oss.str();
    }

public:
    BankingSystem(const std::string& log_file = "transactions.log")
        : log_file_path_(log_file) {}

    // Create an account
    Account& create_account(int id, const std::string& name,
                            double initial_balance) {
        auto acc = std::make_unique<Account>(id, name, initial_balance);
        Account& ref = *acc;
        accounts_.push_back(std::move(acc));
        return ref;
    }

    // Transfer (core method; uses std::lock to avoid deadlock)
    bool transfer(int from_id, int to_id, double amount) {
        if (from_id == to_id) {
            throw std::invalid_argument("Cannot transfer to yourself");
        }

        // Look up the accounts
        Account* from = nullptr;
        Account* to = nullptr;
        for (auto& acc : accounts_) {
            if (acc->get_id() == from_id) from = acc.get();
            if (acc->get_id() == to_id) to = acc.get();
        }

        if (!from || !to) {
            throw std::invalid_argument("Account does not exist");
        }

        try {
            // Key: use std::lock to lock both accounts' mutexes at once, avoiding deadlock
            // std::lock uses a deadlock-avoidance algorithm (try_lock + back-off)
            std::lock(from->get_mutex(), to->get_mutex());

            // adopt_lock: tells lock_guard that the mutex is already locked
            std::lock_guard<std::mutex> lock_from(from->get_mutex(),
                                                    std::adopt_lock);
            std::lock_guard<std::mutex> lock_to(to->get_mutex(),
                                                  std::adopt_lock);

            // Or in C++17 just use scoped_lock:
            // std::scoped_lock lock(from->get_mutex(), to->get_mutex());

            // Check the balance
            if (from->get_balance() < amount) {
                return false;  // Insufficient balance
            }

            // Perform the transfer (both locks are now held, so the operation is atomic)
            from->adjust_balance(-amount);
            to->adjust_balance(amount);

            // Record the transaction
            TransactionLog log;
            log.timestamp = std::chrono::system_clock::now();
            log.type = TransactionType::TRANSFER;
            log.from_account = from_id;
            log.to_account = to_id;
            log.amount = amount;
            log.from_balance_after = from->get_balance();
            log.to_balance_after = to->get_balance();

            write_log_to_file(log);
            total_transactions_.fetch_add(1, std::memory_order_relaxed);
            // Compare-exchange on an atomic double (note: floating-point atomic arithmetic needs a CAS loop)
            double expected = total_volume_.load(std::memory_order_relaxed);
            while (!total_volume_.compare_exchange_weak(
                       expected, expected + amount,
                       std::memory_order_relaxed)) {
                // Retry in a loop until it succeeds
            }

            return true;
        } catch (...) {
            throw;
        }
    }

    // Print the status of all accounts
    void print_accounts() const {
        std::cout << "\n====== Account Overview ======\n";
        for (const auto& acc : accounts_) {
            std::cout << "Account " << acc->get_id()
                      << " [" << acc->get_owner() << "]: "
                      << std::fixed << std::setprecision(2)
                      << acc->get_balance() << " yuan\n";
        }
        std::cout << "========================\n";
    }

    // Print statistics
    void print_statistics() const {
        std::cout << "\n====== Transaction Statistics ======\n";
        std::cout << "Total transactions: " << total_transactions_.load() << "\n";
        std::cout << "Total volume: " << std::fixed << std::setprecision(2)
                  << total_volume_.load() << " yuan\n";
        std::cout << "Log entries: " << transaction_logs_.size() << "\n";
        std::cout << "=======================\n";
    }

    // Save all transaction logs to a file (Chapter 10)
    void save_logs_to_file() {
        std::ofstream file(log_file_path_);
        if (!file.is_open()) {
            throw std::runtime_error("Cannot open log file: " + log_file_path_);
        }

        std::lock_guard<std::mutex> lock(log_mtx_);
        file << "Time,Type,From,To,Amount,FromBalance,ToBalance\n";
        for (const auto& log : transaction_logs_) {
            std::string type_str;
            switch (log.type) {
                case TransactionType::TRANSFER: type_str = "Transfer"; break;
                case TransactionType::DEPOSIT: type_str = "Deposit"; break;
                case TransactionType::WITHDRAW: type_str = "Withdraw"; break;
            }
            file << timestamp_to_string(log.timestamp) << ","
                 << type_str << ","
                 << log.from_account << ","
                 << log.to_account << ","
                 << log.amount << ","
                 << log.from_balance_after << ","
                 << log.to_balance_after << "\n";
        }
        file.close();
        std::cout << "Transaction log saved to: " << log_file_path_ << "\n";
    }

    void request_stop() {
        running_.store(false, std::memory_order_relaxed);
    }

    bool is_running() const {
        return running_.load(std::memory_order_relaxed);
    }
};

// Transfer thread function
void transfer_worker(BankingSystem& bank,
                     int thread_id, int num_transfers,
                     std::atomic<int>& success_count,
                     std::atomic<int>& failure_count) {
    std::mt19937 rng(std::random_device{}() + thread_id);
    std::uniform_int_distribution<int> from_dist(1001, 1010);
    std::uniform_int_distribution<int> to_dist(1001, 1010);
    std::uniform_real_distribution<double> amount_dist(10.0, 500.0);

    for (int i = 0; i < num_transfers; ++i) {
        if (!bank.is_running()) break;

        int from = from_dist(rng);
        int to = to_dist(rng);
        while (from == to) {
            to = to_dist(rng);
        }
        double amount = amount_dist(rng);

        try {
            if (bank.transfer(from, to, amount)) {
                success_count.fetch_add(1, std::memory_order_relaxed);
            } else {
                failure_count.fetch_add(1, std::memory_order_relaxed);
            }
        } catch (const std::exception& e) {
            std::cerr << "[Thread " << thread_id
                      << "] Transfer exception: " << e.what() << std::endl;
            failure_count.fetch_add(1, std::memory_order_relaxed);
        }
    }
}

namespace web_crawler {

// A crawl task
struct CrawlTask {
    std::string url;
    int depth;
};

// A crawl result
struct CrawlResult {
    std::string url;
    std::string content;
    std::vector<std::string> links;
    bool success;
    int http_status;
};

// Thread pool (simplified; see the implementation in Example 4)
class SimpleThreadPool {
private:
    std::vector<std::thread> workers_;
    std::queue<std::function<void()>> tasks_;
    std::mutex mtx_;
    std::condition_variable cv_;
    std::atomic<bool> stop_{false};

    void worker_loop() {
        while (true) {
            std::function<void()> task;
            {
                std::unique_lock<std::mutex> lock(mtx_);
                cv_.wait(lock, [this] {
                    return stop_.load() || !tasks_.empty();
                });
                if (stop_.load() && tasks_.empty()) return;
                task = std::move(tasks_.front());
                tasks_.pop();
            }
            task();
        }
    }

public:
    explicit SimpleThreadPool(size_t num_threads) {
        for (size_t i = 0; i < num_threads; ++i) {
            workers_.emplace_back(&SimpleThreadPool::worker_loop, this);
        }
    }

    ~SimpleThreadPool() { shutdown(); }

    void submit(std::function<void()> task) {
        {
            std::lock_guard<std::mutex> lock(mtx_);
            tasks_.emplace(std::move(task));
        }
        cv_.notify_one();
    }

    void shutdown() {
        stop_.store(true);
        cv_.notify_all();
        for (auto& w : workers_) {
            if (w.joinable()) w.join();
        }
    }
};

// Main WebCrawler class
class WebCrawler {
private:
    SimpleThreadPool pool_;
    std::queue<CrawlTask> url_queue_;
    std::unordered_set<std::string> visited_urls_;
    std::unordered_map<std::string, int> domain_last_access_;

    mutable std::mutex queue_mtx_;
    mutable std::mutex visited_mtx_;
    mutable std::mutex file_mtx_;
    std::condition_variable cv_;
    std::atomic<bool> running_{false};
    std::atomic<bool> stop_requested_{false};

    // Statistics
    std::atomic<int> pages_crawled_{0};
    std::atomic<int> pages_succeeded_{0};
    std::atomic<int> pages_failed_{0};
    std::atomic<int> links_found_{0};
    std::atomic<long long> bytes_downloaded_{0};

    int max_depth_;
    int politeness_delay_ms_;
    std::string output_dir_;
    std::ofstream index_file_;

    // Signal handling (Chapter 11)
    static void signal_handler(int sig) {
        std::cout << "\nReceived signal " << sig << ", stopping gracefully...\n";
        // Notify the crawler to stop (requires a static/global reference)
    }

    // Download a page (simulated)
    CrawlResult download(const std::string& url) {
        CrawlResult result;
        result.url = url;
        result.success = true;
        result.http_status = 200;

        // Simulate the download
        std::this_thread::sleep_for(
            std::chrono::milliseconds(100 + rand() % 200));

        result.content = "<html>Simulated page content " + url + "</html>";

        // Simulate extracting links
        result.links.push_back(url + "/page1");
        result.links.push_back(url + "/page2");

        bytes_downloaded_.fetch_add(result.content.size());
        return result;
    }

    // Parse HTML and extract links
    std::vector<std::string> extract_links(const std::string& html) {
        std::vector<std::string> links;
        // Use regex to extract <a href="..."> tags (Chapter 14: regex)
        std::regex link_regex("<a\\s+href=[\"']([^\"']+)[\"']",
                              std::regex::icase);
        std::smatch match;
        std::string::const_iterator search_start(html.cbegin());
        while (std::regex_search(search_start, html.cend(),
                                  match, link_regex)) {
            links.push_back(match[1].str());
            search_start = match.suffix().first;
        }
        links_found_.fetch_add(links.size());
        return links;
    }

    // Save a page to a file
    void save_page(const CrawlResult& result) {
        std::string filename = output_dir_ + "/" +
            std::to_string(std::hash<std::string>{}(result.url)) + ".html";
        std::lock_guard<std::mutex> lock(file_mtx_);
        std::ofstream file(filename);
        if (file.is_open()) {
            file << result.content;
            file.close();
            index_file_ << result.url << " -> " << filename << "\n";
        }
    }

    // Task to crawl a single page
    void crawl_task(const CrawlTask& task) {
        try {
            // Check the depth limit
            if (task.depth > max_depth_) return;

            // Simulated download
            CrawlResult result = download(task.url);

            if (result.success) {
                pages_crawled_.fetch_add(1);
                pages_succeeded_.fetch_add(1);

                // Save the page
                save_page(result);

                // Extract links and add them to the queue
                auto links = extract_links(result.content);
                for (const auto& link : links) {
                    add_url(link, task.depth + 1);
                }
            } else {
                pages_failed_.fetch_add(1);
                std::cerr << "Download failed " << task.url
                          << " (HTTP " << result.http_status << ")\n";
            }
        } catch (const std::exception& e) {
            pages_failed_.fetch_add(1);
            std::cerr << "Crawl exception " << task.url
                      << ": " << e.what() << "\n";
        }
    }

    // Add a URL to the queue (thread-safe)
    void add_url(const std::string& url, int depth) {
        {
            std::lock_guard<std::mutex> lock(visited_mtx_);
            if (visited_urls_.count(url) > 0) return;
            visited_urls_.insert(url);
        }
        {
            std::lock_guard<std::mutex> lock(queue_mtx_);
            url_queue_.push({url, depth});
        }
        cv_.notify_one();
    }

    // Get the next URL (thread-safe, blocking)
    bool get_next_url(CrawlTask& task) {
        std::unique_lock<std::mutex> lock(queue_mtx_);
        cv_.wait(lock, [this] {
            return stop_requested_.load() || !url_queue_.empty();
        });
        if (stop_requested_.load() && url_queue_.empty()) {
            return false;
        }
        task = url_queue_.front();
        url_queue_.pop();
        return true;
    }

    // Worker thread
    void worker_loop() {
        while (running_.load()) {
            CrawlTask task;
            if (!get_next_url(task)) break;
            crawl_task(task);
        }
    }

public:
    WebCrawler(size_t max_threads, int max_depth,
               int politeness_delay, const std::string& output_dir)
        : pool_(max_threads),
          max_depth_(max_depth),
          politeness_delay_ms_(politeness_delay),
          output_dir_(output_dir) {
        // Open the index file
        index_file_.open(output_dir + "/index.txt",
                          std::ios::out | std::ios::app);
    }

    ~WebCrawler() {
        stop();
        if (index_file_.is_open()) {
            index_file_.close();
        }
    }

    // Add a seed URL
    void add_seed_url(const std::string& url) {
        add_url(url, 0);
    }

    // Start the crawler
    void start() {
        running_.store(true);
        int num_threads = std::thread::hardware_concurrency();
        for (int i = 0; i < num_threads; ++i) {
            pool_.submit([this]() { worker_loop(); });
        }
        std::cout << "Crawler started, using " << num_threads
                  << " worker threads\n";
    }

    // Stop the crawler
    void stop() {
        stop_requested_.store(true);
        cv_.notify_all();
        running_.store(false);
        std::cout << "Crawler stopped\n";
    }

    // Wait for all tasks to complete
    void wait() {
        while (!url_queue_.empty()) {
            std::this_thread::sleep_for(std::chrono::milliseconds(100));
        }
    }

    // Print statistics
    void print_stats() const {
        std::cout << "\n========== Crawler Statistics ==========\n";
        std::cout << "Pages crawled:      " << pages_crawled_.load() << "\n";
        std::cout << "Succeeded:          " << pages_succeeded_.load() << "\n";
        std::cout << "Failed:             " << pages_failed_.load() << "\n";
        std::cout << "Links found:        " << links_found_.load() << "\n";
        std::cout << "Bytes downloaded:   " << bytes_downloaded_.load() << "\n";
        std::cout << "===============================\n";
    }
};

// Usage example
void run_demo() {
    try {
        WebCrawler crawler(4, 3, 500, "./crawled_pages");

        // Set up signal handling
        std::signal(SIGINT, WebCrawler::signal_handler);
        std::signal(SIGTERM, WebCrawler::signal_handler);

        // Add seed URLs
        crawler.add_seed_url("https://example.com");
        crawler.add_seed_url("https://example.org");

        auto start_time = std::chrono::steady_clock::now();

        crawler.start();
        crawler.wait();
        crawler.stop();

        auto end_time = std::chrono::steady_clock::now();
        auto elapsed = std::chrono::duration_cast<
            std::chrono::seconds>(end_time - start_time);

        crawler.print_stats();
        std::cout << "Total elapsed: " << elapsed.count() << " seconds\n";

    } catch (const std::exception& e) {
        std::cerr << "Crawler runtime exception: " << e.what() << std::endl;
    }
}

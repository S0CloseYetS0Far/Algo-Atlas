[13.1.1 Basic Example: TCP Client]

Network programming in C++ requires the operating system's Socket API. On Linux you use POSIX sockets,
and on Windows you use Winsock2. Below is a basic TCP client example:

// ============================================================
// Example 13-1: The simplest TCP echo client (Linux)
// ============================================================
#include <iostream>
#include <string>
#include <cstring>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <netdb.h>

#define PORT 8080
#define SERVER_IP "127.0.0.1"

int tcp_client_basic() {
    // 1. Create the socket
    // AF_INET: IPv4 protocol family
    // SOCK_STREAM: connection-oriented TCP socket
    // 0: choose the protocol automatically (TCP)
    int sock = socket(AF_INET, SOCK_STREAM, 0);
    if (sock < 0) {
        std::perror("socket() failed");
        return -1;
    }

    // 2. Set up the server address
    struct sockaddr_in server_addr;
    std::memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;           // IPv4
    server_addr.sin_port = htons(PORT);          // Port number (network byte order)

    // Convert the string IP to binary form
    if (inet_pton(AF_INET, SERVER_IP,
                  &server_addr.sin_addr) <= 0) {
        std::perror("inet_pton() failed");
        close(sock);
        return -1;
    }

    // 3. Connect to the server
    if (connect(sock, (struct sockaddr*)&server_addr,
                sizeof(server_addr)) < 0) {
        std::perror("connect() failed");
        close(sock);
        return -1;
    }

    std::cout << "Connected to server " << SERVER_IP << ":"
              << PORT << std::endl;

    // 4. Send data
    std::string message = "Hello, Server!";
    ssize_t bytes_sent = send(sock, message.c_str(),
                               message.size(), 0);
    std::cout << "Sent " << bytes_sent << " bytes: "
              << message << std::endl;

    // 5. Receive the response
    char buffer[1024] = {0};
    ssize_t bytes_received = recv(sock, buffer,
                                   sizeof(buffer) - 1, 0);
    if (bytes_received > 0) {
        buffer[bytes_received] = '\0';
        std::cout << "Received response (" << bytes_received << " bytes): "
                  << buffer << std::endl;
    }

    // 6. Close the connection
    close(sock);

    return 0;
}

// Compile: g++ -std=c++17 13-1.cpp -o 13-1


[13.1.2 Basic Example: TCP Server]

// ============================================================
// Example 13-2: TCP Echo Server
// ============================================================
#include <iostream>
#include <cstring>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

#define PORT 8080
#define BACKLOG 10  // Maximum length of the pending-connection queue

void handle_client(int client_sock, struct sockaddr_in client_addr) {
    char client_ip[INET_ADDRSTRLEN];
    inet_ntop(AF_INET, &client_addr.sin_addr,
              client_ip, sizeof(client_ip));

    std::cout << "Client connected: " << client_ip << ":"
              << ntohs(client_addr.sin_port) << std::endl;

    char buffer[1024];
    while (true) {
        std::memset(buffer, 0, sizeof(buffer));
        ssize_t bytes = recv(client_sock, buffer,
                              sizeof(buffer) - 1, 0);

        if (bytes <= 0) {
            if (bytes == 0) {
                std::cout << "Client disconnected\n";
            } else {
                std::perror("recv() failed");
            }
            break;
        }

        std::cout << "Received " << bytes << " bytes: "
                  << buffer << std::endl;

        // Echo (send the received data back unchanged)
        send(client_sock, buffer, bytes, 0);
    }

    close(client_sock);
}

int tcp_server_basic() {
    // 1. Create the socket
    int server_sock = socket(AF_INET, SOCK_STREAM, 0);
    if (server_sock < 0) {
        std::perror("socket() failed");
        return -1;
    }

    // Set SO_REUSEADDR to allow port reuse (avoids the TIME_WAIT problem)
    int opt = 1;
    if (setsockopt(server_sock, SOL_SOCKET, SO_REUSEADDR,
                   &opt, sizeof(opt)) < 0) {
        std::perror("setsockopt() failed");
    }

    // 2. Bind the address and port
    struct sockaddr_in server_addr;
    std::memset(&server_addr, 0, sizeof(server_addr));
    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = INADDR_ANY;  // Listen on all network interfaces
    server_addr.sin_port = htons(PORT);

    if (bind(server_sock, (struct sockaddr*)&server_addr,
             sizeof(server_addr)) < 0) {
        std::perror("bind() failed");
        close(server_sock);
        return -1;
    }

    // 3. Start listening
    if (listen(server_sock, BACKLOG) < 0) {
        std::perror("listen() failed");
        close(server_sock);
        return -1;
    }

    std::cout << "Echo server listening on 0.0.0.0:" << PORT
              << std::endl;

    // 4. Accept-connection loop
    while (true) {
        struct sockaddr_in client_addr;
        socklen_t addr_len = sizeof(client_addr);

        int client_sock = accept(server_sock,
                                  (struct sockaddr*)&client_addr,
                                  &addr_len);

        if (client_sock < 0) {
            std::perror("accept() failed");
            continue;
        }

        handle_client(client_sock, client_addr);
        // Note: clients are handled serially here! A real server needs multithreading (Chapter 12)
    }

    close(server_sock);
    return 0;
}

// ============================================================
// Example 13-3: Cross-platform socket client wrapper
// ============================================================
#ifdef _WIN32
    #include <winsock2.h>
    #include <ws2tcpip.h>
    #pragma comment(lib, "ws2_32.lib")
    using socklen_t = int;
#else
    #include <unistd.h>
    #include <sys/socket.h>
    #include <netinet/in.h>
    #include <arpa/inet.h>
    #include <netdb.h>
    #define SOCKET_ERROR (-1)
    using SOCKET = int;
#endif

#include <iostream>
#include <string>
#include <cstring>
#include <stdexcept>

class SocketInitializer {
public:
    SocketInitializer() {
#ifdef _WIN32
        WSADATA wsaData;
        if (WSAStartup(MAKEWORD(2, 2), &wsaData) != 0) {
            throw std::runtime_error("WSAStartup failed");
        }
#endif
    }
    ~SocketInitializer() {
#ifdef _WIN32
        WSACleanup();
#endif
    }
};

// Cross-platform socket close
void close_socket(SOCKET sock) {
#ifdef _WIN32
    closesocket(sock);
#else
    close(sock);
#endif
}

// Cross-platform TCP client
class TcpClient {
private:
    SOCKET sock_;
    bool connected_;

public:
    TcpClient() : sock_(SOCKET_ERROR), connected_(false) {}

    ~TcpClient() {
        disconnect();
    }

    bool connect_to(const std::string& host, int port) {
        sock_ = socket(AF_INET, SOCK_STREAM, 0);
        if (sock_ == SOCKET_ERROR) {
            return false;
        }

        struct sockaddr_in addr;
        std::memset(&addr, 0, sizeof(addr));
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port);

        if (inet_pton(AF_INET, host.c_str(),
                      &addr.sin_addr) <= 0) {
            close_socket(sock_);
            return false;
        }

        if (connect(sock_, (struct sockaddr*)&addr,
                    sizeof(addr)) == SOCKET_ERROR) {
            close_socket(sock_);
            return false;
        }

        connected_ = true;
        return true;
    }

    int send_data(const std::string& data) {
        if (!connected_) return -1;
        int sent = send(sock_, data.c_str(), data.size(), 0);
        return sent;
    }

    std::string receive_data(size_t max_size = 4096) {
        if (!connected_) return "";
        std::vector<char> buffer(max_size);
        int received = recv(sock_, buffer.data(),
                             buffer.size() - 1, 0);
        if (received > 0) {
            buffer[received] = '\0';
            return std::string(buffer.data());
        }
        return "";
    }

    void disconnect() {
        if (connected_) {
            close_socket(sock_);
            connected_ = false;
        }
    }
};

int cross_platform_example() {
    try {
        SocketInitializer init;
        TcpClient client;

        if (client.connect_to("127.0.0.1", 80)) {
            client.send_data("GET / HTTP/1.0\r\n\r\n");
            std::string response = client.receive_data();
            std::cout << "Response: " << response << std::endl;
        }
    } catch (const std::exception& e) {
        std::cerr << "Error: " << e.what() << std::endl;
    }

    return 0;
}

[13.1.4 Under the Hood: The Layers of the TCP/IP Protocol Stack]
================================================================================

The TCP/IP protocol stack is the foundation of Internet communication. Understanding the layered model is essential for network programming.

I. The TCP/IP Four-Layer Model (top to bottom)

┌───────────────────────────────────────────────────────────┐
│  Application layer                                        │
│  HTTP, HTTPS, FTP, SMTP, DNS, WebSocket                   │
│  User data is produced and consumed here                  │
├───────────────────────────────────────────────────────────┤
│  Transport layer                                          │
│  TCP (reliable), UDP (unreliable)                         │
│  Port numbers identify apps; provides end-to-end delivery │
├───────────────────────────────────────────────────────────┤
│  Network / Internet layer                                 │
│  IP (IPv4, IPv6), ICMP, ARP                               │
│  IP addresses identify hosts; routes/forwards packets     │
├───────────────────────────────────────────────────────────┤
│  Data link layer (Link / Network Access)                  │
│  Ethernet, Wi-Fi, PPP                                     │
│  MAC addresses identify devices on a LAN; moves frames    │
├───────────────────────────────────────────────────────────┤
│  Physical layer                                           │
│  Twisted pair, optical fiber, radio waves                 │
│  The transmission medium for the bit stream               │
└───────────────────────────────────────────────────────────┘

II. Data Encapsulation (Encapsulation / Decapsulation)

When sending data (top-down):
1. The application layer creates the data (e.g. an HTTP request)
2. Transport layer: adds a TCP header (source port, destination port, sequence number, etc.) → TCP Segment
3. Network layer: adds an IP header (source IP, destination IP, TTL, etc.) → IP Packet
4. Data link layer: adds a MAC frame header (source MAC, destination MAC, CRC checksum) → Ethernet Frame
5. Physical layer: converts the frame into electrical/optical/radio signals

When receiving data (the bottom-up reverse process), each layer strips its corresponding header, and the data is finally delivered to the application layer.

III. Why Layering?
   1. Separation of concerns: each layer only needs to care about its own responsibilities
   2. Standardization: implementations from different vendors can interoperate
   3. Flexibility: one layer's implementation can be replaced without affecting the others
      (e.g. switching from Ethernet to Wi-Fi requires no change to TCP/IP above it)
   4. Easier debugging: network problems can be located layer by layer

[13.1.5 Under the Hood: TCP Three-Way Handshake and Four-Way Teardown]
================================================================================

I. TCP Three-Way Handshake (establishing a connection)

   Client                                    Server
     |                                         |
     |─────── SYN (seq=x) ───────────────────→| State: SYN_SENT → SYN_RCVD
     |                                         |
     |←────── SYN+ACK (seq=y, ack=x+1) ───────| State: SYN_RCVD → received
     |                                         |
     |─────── ACK (ack=y+1) ──────────────────→| State: ESTABLISHED
     |                                         |
     |         ★ Connection established ★       |

   The purpose of each step of the handshake:
   - 1st: Client → Server: "I want to connect to you" (the Client proves it can send)
   - 2nd: Server → Client: "Got it, I'm willing to connect" (the Server proves it can both receive and send)
   - 3rd: Client → Server: "I received your acknowledgment" (the Client proves it can receive)

   Why three steps instead of two?
   Suppose there were only a two-way handshake (Client sends SYN, Server replies SYN+ACK and the connection is established).
   Imagine the Client's first SYN gets retransmitted due to network delay, and this old SYN only
   reaches the Server after the connection has been closed. The Server replies with SYN+ACK and allocates resources
   for a connection, but the Client no longer needs it. With a three-way handshake the Client simply
   never sends the final ACK, and the Server knows the connection request was invalid.

II. TCP Four-Way Teardown (releasing a connection)

   Client                                    Server
     |                                         |
     |─────── FIN (seq=m) ────────────────────→| State: FIN_WAIT_1
     |                                         |
     |←────── ACK (ack=m+1) ──────────────────| State: FIN_WAIT_2 / CLOSE_WAIT
     |                                         |
     |←────── FIN (seq=n) ────────────────────| State: LAST_ACK
     |                                         |
     |─────── ACK (ack=n+1) ──────────────────→| State: TIME_WAIT / CLOSED
     |                                         |
     |  (TIME_WAIT lasts 2MSL, ~1-4 minutes)   |

   Why four steps instead of three?
   Because TCP is full-duplex: both sides can send data independently. When one side sends FIN, it means
   "I have no more data to send", but the other side may still have data to send. Therefore you need:
   - The first FIN+ACK: active closer → passive side, "I'm done sending"
   - The second FIN+ACK: passive closer → active side, "I'm done sending" (possibly after some time)

   The meaning of the TIME_WAIT state (the active closer waits 2MSL):
   1. Ensure the final ACK reaches the peer (if the ACK is lost, the peer will resend FIN)
   2. Let late segments of the old connection "die out" in the network so they aren't confused with a new connection

   This is also why SO_REUSEADDR is needed: it allows a restarting server to bind the port immediately
   without waiting for TIME_WAIT to end.

[13.1.6 Under the Hood: What a Socket Really Is]
================================================================================

I. What is a Socket?

A socket is the network programming interface the operating system provides to applications. On Linux,
a socket is essentially a special file descriptor (FD). When you call socket(), the kernel
creates a data structure (struct socket) associated with the following kernel resources:

- Send Buffer: application data is staged here before being sent
- Receive Buffer: data received from the network is staged here, waiting for the application to read it
- TCB (TCP Control Block): maintains the TCP connection state
  (state machine, sequence numbers, window size, RTT estimation, etc.)
- Wait queues: a connection wait queue (for accept) and a data wait queue (for read/recv)
- Links to the underlying network protocol stack (e.g. congestion-control state in the TCP layer)

The kernel maintains two buffers for each socket:
                        ┌─────────────────┐
  user space send() ──→ │ Send buffer     │ ──→ protocol stack → NIC
                        │ (SO_SNDBUF)     │
                        └─────────────────┘

                        ┌─────────────────┐
  user space recv() ←── │ Receive buffer  │ ←── NIC → protocol stack
                        │ (SO_RCVBUF)     │
                        └─────────────────┘

Default buffer sizes (Linux):
- SO_RCVBUF: usually about 212992 bytes (see /proc/sys/net/core/rmem_default)
- SO_SNDBUF: usually about 212992 bytes

II. How Sockets Appear in the File System

Like regular files, sockets are file descriptors. This means:
- You can close a socket with close() (releasing the FD and related kernel resources)
- You can set non-blocking mode with fcntl()
- You can multiplex with select/poll/epoll
- You can view a process's socket file descriptors under /proc/<pid>/fd/

In /proc/<pid>/fd/, a socket is shown as socket:[inode_number]

III. Blocking I/O vs. Non-Blocking I/O

Blocking mode (default):
- When recv() has no data, the process goes to sleep (TASK_INTERRUPTIBLE)
- When the send() buffer is full, the process also sleeps
- When accept() has no new connection, the process sleeps

Non-blocking mode (set O_NONBLOCK via fcntl):
- When recv() has no data, it returns -1 immediately with errno=EAGAIN/EWOULDBLOCK
- When the send() buffer is full, it returns -1 immediately
- The application must poll, or use epoll/select

Setting non-blocking mode:
int flags = fcntl(sock, F_GETFL, 0);
fcntl(sock, F_SETFL, flags | O_NONBLOCK);

[13.1.7 Under the Hood: The epoll Multiplexing Mechanism]
================================================================================

When a server must handle tens of thousands of connections at once, creating one thread per connection
is impractical (thread stack memory overhead + context-switch overhead).

Linux's epoll (Event Poll) solves this problem. It lets a single thread efficiently
monitor events on a large number of file descriptors.

I. The Three Core epoll System Calls

1. epoll_create1(int flags)
   - Creates an epoll instance and returns a file descriptor
   - The kernel allocates an eventpoll struct, containing:
     * A red-black tree (rbtree): stores the monitored file descriptors, supporting O(log n) insert/delete/lookup
     * A ready list (rdllist): stores the file descriptors that have pending events

2. epoll_ctl(int epfd, int op, int fd, struct epoll_event* event)
   - Adds/modifies/removes file descriptors to be monitored by the epoll instance
   - Registered event types: EPOLLIN (readable), EPOLLOUT (writable), EPOLLERR, etc.
   - When an event occurs on an fd, the kernel adds that fd to the ready list

3. epoll_wait(int epfd, struct epoll_event* events, int maxevents, int timeout)
   - Waits for events and returns the list of ready file descriptors
   - Unlike select/poll, epoll only copies the ready fds (rather than all monitored fds)
   - This is the key to epoll's O(1) performance

II. The Two epoll Trigger Modes

1. Level Triggered (LT) — the default mode
   - As long as the file descriptor is readable/writable, epoll_wait keeps returning that fd
   - Behaves like select/poll; simple to use
   - If you don't read all the data this time, the next epoll_wait will notify you again

2. Edge Triggered (ET)
   - Notifies only once, when the state changes (from nothing to something)
   - Must be used together with non-blocking I/O
   - You must read all the data in one go (loop on recv until EAGAIN)
   - Higher performance (fewer system calls), but harder to program

III. epoll vs. select/poll

| Feature        | select          | poll            | epoll              |
|----------------|-----------------|-----------------|--------------------|
| Data structure | fd_set bitmap   | pollfd array    | rb-tree + list     |
| Max fds        | 1024(FD_SETSIZE)| unlimited       | unlimited          |
| Complexity     | O(n)            | O(n)            | O(1) active conns  |
| Memory copying | all fds per call| all fds per call| only active fds    |
| Kernel impl.   | scans all fds   | scans all fds   | events + callbacks |

n is the total number of monitored file descriptors. epoll's O(1) means the time to obtain ready fds is independent of the total number of fds.

IV. epoll Example Code

#include <sys/epoll.h>

// Create an epoll instance
int epfd = epoll_create1(0);

// Register the fd to monitor
struct epoll_event ev;
ev.events = EPOLLIN | EPOLLET;  // Read event + edge triggered
ev.data.fd = listen_fd;
epoll_ctl(epfd, EPOLL_CTL_ADD, listen_fd, &ev);

// Event loop
struct epoll_event events[MAX_EVENTS];
while (true) {
    int nfds = epoll_wait(epfd, events, MAX_EVENTS, -1);
    for (int i = 0; i < nfds; ++i) {
        if (events[i].data.fd == listen_fd) {
            // New connection
            int client = accept(listen_fd, ...);
            // Register the client with epoll
            ev.events = EPOLLIN | EPOLLET;
            ev.data.fd = client;
            epoll_ctl(epfd, EPOLL_CTL_ADD, client, &ev);
        } else {
            // Client data
            int client_fd = events[i].data.fd;
            // Edge triggered: keep reading until EAGAIN
            while (true) {
                ssize_t n = recv(client_fd, buf, sizeof(buf), 0);
                if (n <= 0) { /* handle disconnect or error */ break; }
                // Process data...
            }
        }
    }
}

[13.1.8 Under the Hood: HTTP Message Format]
================================================================================

I. HTTP Request Format

A request consists of: request line + request headers + blank line + request body (optional)

  Request line:    METHOD /path HTTP/1.1\r\n
  Request headers: Host: www.example.com\r\n
          User-Agent: Mozilla/5.0\r\n
          Content-Type: application/json\r\n
          Content-Length: 27\r\n
  \r\n (blank line)
  Request body: {"key": "value"}\n

Request methods:
- GET: retrieve a resource (no request body)
- POST: submit data (has a request body)
- PUT: update a resource
- DELETE: delete a resource
- HEAD: like GET but returns only the headers
- OPTIONS: query the supported methods

II. HTTP Response Format

A response consists of: status line + response headers + blank line + response body (optional)

  Status line:      HTTP/1.1 200 OK\r\n
  Response headers: Content-Type: text/html\r\n
          Content-Length: 123\r\n
          Connection: keep-alive\r\n
  \r\n (blank line)
  Response body: <html><body>Hello World</body></html>

Status code classes:
- 1xx (Informational): request received, continuing to process
- 2xx (Success): 200 OK, 201 Created, 204 No Content
- 3xx (Redirection): 301 Moved Permanently, 302 Found (moved temporarily), 304 Not Modified
- 4xx (Client error): 400 Bad Request, 401 Unauthorized, 403 Forbidden,
                    404 Not Found, 405 Method Not Allowed
- 5xx (Server error): 500 Internal Server Error, 502 Bad Gateway,
                    503 Service Unavailable

III. The Full Process from Typing a URL to Displaying the Page

After the user types https://www.example.com/path into the browser:

1. DNS resolution: the browser queries a DNS server to turn the domain name into an IP address
   - Browser cache first → OS cache → router cache → ISP DNS → root DNS

2. TCP three-way handshake: the browser establishes a TCP connection to port 443 on the server IP

3. TLS/SSL handshake (HTTPS):
   - Client Hello → Server Hello → certificate verification → key exchange → encrypted channel established

4. Sending the HTTP request: the browser builds an HTTP GET request and sends it

5. Server processing: the web server receives the request, runs backend logic, and returns an HTTP response

6. Browser receives the response: parses the HTTP response and obtains the HTML content

7. Browser rendering: parse HTML → build the DOM tree → parse CSS → compute layout → paint

8. TCP four-way teardown: depending on the Connection header, the connection is kept alive or closed

II. bind(): Binding an Address

// ============================================================
// Example 13-5: Address structures for bind()
// ============================================================
#include <sys/socket.h>
#include <netinet/in.h>

void bind_examples() {
    int sock = socket(AF_INET, SOCK_STREAM, 0);

    // 1. Bind to a given port on all interfaces
    struct sockaddr_in addr1;
    addr1.sin_family = AF_INET;
    addr1.sin_addr.s_addr = INADDR_ANY;  // 0.0.0.0
    addr1.sin_port = htons(8080);
    bind(sock, (struct sockaddr*)&addr1, sizeof(addr1));

    // 2. Bind to a specific IP address (select a specific NIC on multi-NIC hosts)
    struct sockaddr_in addr2;
    addr2.sin_family = AF_INET;
    inet_pton(AF_INET, "192.168.1.100", &addr2.sin_addr);
    addr2.sin_port = htons(8080);
    // bind(sock, (struct sockaddr*)&addr2, sizeof(addr2));

    // 3. Let the kernel assign a port automatically (don't call bind, or bind port 0)
    struct sockaddr_in addr3;
    addr3.sin_family = AF_INET;
    addr3.sin_addr.s_addr = INADDR_ANY;
    addr3.sin_port = htons(0);  // The kernel picks a free port automatically
    // bind(sock, (struct sockaddr*)&addr3, sizeof(addr3));

    // Retrieve the automatically assigned port
    // struct sockaddr_in bound_addr;

// ============================================================
// Example 13-6: Using listen/accept
// ============================================================
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>

void listen_accept_example() {
    int server = socket(AF_INET, SOCK_STREAM, 0);

    struct sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = INADDR_ANY;
    addr.sin_port = htons(9090);
    bind(server, (struct sockaddr*)&addr, sizeof(addr));

    // backlog: maximum length of the completed-connection queue
    // Linux 5.4+: the default is in /proc/sys/net/core/somaxconn (usually 128)
    // If the queue is full, new SYNs are ignored (no RST is sent, so the client retries)
    listen(server, 10);

    // accept: take one connection from the completed-connection queue
    struct sockaddr_in client_addr;
    socklen_t addr_len = sizeof(client_addr);

    // accept4 (Linux-specific): can set SOCK_NONBLOCK | SOCK_CLOEXEC directly
    // int client = accept4(server,
    //     (struct sockaddr*)&client_addr, &addr_len,
    //     SOCK_NONBLOCK | SOCK_CLOEXEC);

    int client = accept(server,
                        (struct sockaddr*)&client_addr, &addr_len);

    if (client >= 0) {
        char ip[INET_ADDRSTRLEN];
        inet_ntop(AF_INET, &client_addr.sin_addr, ip, sizeof(ip));
        std::cout << "Client: " << ip << ":"
                  << ntohs(client_addr.sin_port) << std::endl;
        close(client);
    }

    close(server);
}

IV. connect(): Client Connection

// ============================================================
// Example 13-7: Handling connect timeouts
// ============================================================
#include <sys/socket.h>
#include <netinet/in.h>
#include <unistd.h>
#include <fcntl.h>
#include <errno.h>
#include <sys/select.h>

bool connect_with_timeout(int sock, struct sockaddr* addr,
                           socklen_t addr_len, int timeout_sec) {
    // Switch to non-blocking mode
    int flags = fcntl(sock, F_GETFL, 0);
    fcntl(sock, F_SETFL, flags | O_NONBLOCK);

    // Non-blocking connect
    int result = connect(sock, addr, addr_len);

    if (result == 0) {
        // Connected immediately (common for local connections)
        fcntl(sock, F_SETFL, flags);  // Restore blocking mode
        return true;
    }

    if (errno != EINPROGRESS) {
        // A real error
        fcntl(sock, F_SETFL, flags);
        return false;
    }

    // Connection in progress; wait with select
    fd_set write_fds;
    FD_ZERO(&write_fds);
    FD_SET(sock, &write_fds);

    struct timeval tv;
    tv.tv_sec = timeout_sec;
    tv.tv_usec = 0;

    result = select(sock + 1, nullptr, &write_fds, nullptr, &tv);

    if (result <= 0) {
        // Timeout or error
        fcntl(sock, F_SETFL, flags);
        return false;
    }

    // Check whether the connection actually succeeded
    int error = 0;
    socklen_t len = sizeof(error);
    getsockopt(sock, SOL_SOCKET, SO_ERROR, &error, &len);

    // Restore blocking mode
    fcntl(sock, F_SETFL, flags);

    return (error == 0);
}

void connect_example() {
    int sock = socket(AF_INET, SOCK_STREAM, 0);

    struct sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_port = htons(80);
    inet_pton(AF_INET, "93.184.216.34", &addr.sin_addr);

    if (connect_with_timeout(sock, (struct sockaddr*)&addr,
                              sizeof(addr), 5)) {
        std::cout << "Connected successfully!" << std::endl;
    } else {
        std::cout << "Connection failed or timed out." << std::endl;
    }

    close(sock);
}

    // write(sock, data, data_len);

    // sendto(sock, data, data_len, 0, (struct sockaddr*)&addr, sizeof(addr));

    char buffer[4096];

    // MSG_DONTWAIT: non-blocking receive
    // MSG_PEEK: look at the data without removing it (data stays in the buffer)
    // MSG_WAITALL: wait until the requested number of bytes has fully arrived
    // MSG_TRUNC: return the real data length (even if larger than the buffer)

    } else {

    }
}

    std::cout << "Host byte order: "
              << (is_little_endian ? "little-endian" : "big-endian") << std::endl;

    // htonl: Host TO Network Long (32-bit)
    // ntohs: Network TO Host Short
    // ntohl: Network TO Host Long

    uint16_t host_port = 8080;
    uint16_t net_port = htons(host_port);
    std::cout << "Host port: " << host_port
              << " → Network: " << net_port << std::endl;

              << " → Host: " << ntohs(net_port) << std::endl;

    uint32_t host_ip_int = 0xC0A80101;  // 192.168.1.1
    uint32_t net_ip_int = htonl(host_ip_int);
    std::cout << "Host IP hex: 0x" << std::hex << host_ip_int
              << " → Network: 0x" << net_ip_int << std::dec << std::endl;

    // Modern alternatives: htobe16/htole16 etc. (endian.h)
    // htobe64 / htole64 / be64toh / le64toh

    #include <endian.h>

    std::cout << "htobe64: " << std::hex << big
              << " → be64toh: " << back << std::dec << std::endl;

}

// ============================================================
// Example 13-11: Protocol-independent address resolution with getaddrinfo
// ============================================================
#include <sys/types.h>
#include <sys/socket.h>
#include <netdb.h>
#include <iostream>
#include <cstring>
#include <unistd.h>
#include <arpa/inet.h>

void getaddrinfo_example() {
    struct addrinfo hints;
    struct addrinfo* result = nullptr;

    std::memset(&hints, 0, sizeof(hints));
    hints.ai_family = AF_UNSPEC;     // Support both IPv4 and IPv6
    hints.ai_socktype = SOCK_STREAM; // TCP stream
    hints.ai_flags = AI_PASSIVE;     // For bind; fill in the local address automatically

    // Get the address information
    int status = getaddrinfo("example.com", "80", &hints, &result);
    if (status != 0) {
        std::cerr << "getaddrinfo: " << gai_strerror(status)
                  << std::endl;
        return;
    }

    std::cout << "Results of resolving example.com:80:"
              << std::endl;

    // Iterate over all results
    for (struct addrinfo* rp = result; rp != nullptr;
         rp = rp->ai_next) {
        char addr_str[INET6_ADDRSTRLEN];
        void* addr_ptr = nullptr;
        int port = 0;

        if (rp->ai_family == AF_INET) {
            struct sockaddr_in* ipv4 =
                (struct sockaddr_in*)rp->ai_addr;
            addr_ptr = &ipv4->sin_addr;
            port = ntohs(ipv4->sin_port);
        } else if (rp->ai_family == AF_INET6) {
            struct sockaddr_in6* ipv6 =
                (struct sockaddr_in6*)rp->ai_addr;
            addr_ptr = &ipv6->sin6_addr;
            port = ntohs(ipv6->sin6_port);
        }

        inet_ntop(rp->ai_family, addr_ptr,
                  addr_str, sizeof(addr_str));
        std::cout << "  " << (rp->ai_family == AF_INET ? "IPv4" : "IPv6")
                  << ": " << addr_str << ":" << port
                  << " (socktype=" << rp->ai_socktype
                  << ", protocol=" << rp->ai_protocol << ")"
                  << std::endl;
    }

    // Try to connect to the first address that works
    int sock = -1;
    for (struct addrinfo* rp = result; rp != nullptr;
         rp = rp->ai_next) {
        sock = socket(rp->ai_family, rp->ai_socktype,
                      rp->ai_protocol);
        if (sock == -1) continue;

        if (connect(sock, rp->ai_addr, rp->ai_addrlen) != -1) {
            std::cout << "Successfully connected to: ";
            // Print the connected address...
            break;
        }

        close(sock);
        sock = -1;
    }

    if (sock != -1) close(sock);

    // Free the address information
    freeaddrinfo(result);
}

// Simplified example of creating a server with getaddrinfo
int create_server_socket(const char* service) {
    struct addrinfo hints, *result, *rp;

    std::memset(&hints, 0, sizeof(hints));
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_STREAM;
    hints.ai_flags = AI_PASSIVE;

    int s = getaddrinfo(nullptr, service, &hints, &result);
    if (s != 0) return -1;

    int sock = -1;
    for (rp = result; rp != nullptr; rp = rp->ai_next) {
        sock = socket(rp->ai_family, rp->ai_socktype,
                      rp->ai_protocol);
        if (sock == -1) continue;

        // Enable address reuse
        int yes = 1;
        setsockopt(sock, SOL_SOCKET, SO_REUSEADDR,
                   &yes, sizeof(yes));

        if (bind(sock, rp->ai_addr, rp->ai_addrlen) == 0) {
            break;  // Bind succeeded
        }

        close(sock);
        sock = -1;
    }

    freeaddrinfo(result);
    return sock;
}

// ============================================================
// Example 13-12: Common setsockopt options

#include <sys/socket.h>
#include <netinet/in.h>
#include <netinet/tcp.h>
#include <iostream>
#include <unistd.h>

void setsockopt_examples(int sock) {
    // 1. SO_REUSEADDR: allow reuse of a local address (can bind during TIME_WAIT)
    int reuse = 1;
    setsockopt(sock, SOL_SOCKET, SO_REUSEADDR,
               &reuse, sizeof(reuse));

    // 2. SO_REUSEPORT (Linux 3.9+): multiple sockets may bind the same port
    // Used for simple multi-process/multi-thread load balancing (kernel-level distribution)
    // int reuse_port = 1;
    // setsockopt(sock, SOL_SOCKET, SO_REUSEPORT,
    //            &reuse_port, sizeof(reuse_port));

    // 3. SO_KEEPALIVE: enable TCP keepalive (detects dead connections)
    int keepalive = 1;
    setsockopt(sock, SOL_SOCKET, SO_KEEPALIVE,
               &keepalive, sizeof(keepalive));

    // TCP keepalive parameters (Linux-specific)
    // int keepidle = 7200;  // Start probing after 7200 seconds without data
    // int keepintvl = 75;   // Probe interval of 75 seconds
    // int keepcnt = 9;      // Disconnect after 9 failed probes
    // setsockopt(sock, IPPROTO_TCP, TCP_KEEPIDLE, &keepidle, sizeof(keepidle));
    // setsockopt(sock, IPPROTO_TCP, TCP_KEEPINTVL, &keepintvl, sizeof(keepintvl));
    // setsockopt(sock, IPPROTO_TCP, TCP_KEEPCNT, &keepcnt, sizeof(keepcnt));

    // 4. SO_LINGER: control the behavior of close
    struct linger ling;
    ling.l_onoff = 1;   // Enable linger
    ling.l_linger = 5;  // Wait 5 seconds
    setsockopt(sock, SOL_SOCKET, SO_LINGER,
               &ling, sizeof(ling));
    // l_onoff=0: default; close returns immediately and the kernel tries to send remaining data
    // l_onoff=1, l_linger=0: close returns immediately, discards buffered data, sends RST
    // l_onoff=1, l_linger>0: close blocks until the data is sent or the timeout expires

    // 5. TCP_NODELAY: disable the Nagle algorithm (reduces latency for small packets)
    // Suitable for latency-sensitive applications (e.g. SSH, games, real-time communication)
    int nodelay = 1;
    setsockopt(sock, IPPROTO_TCP, TCP_NODELAY,
               &nodelay, sizeof(nodelay));

    // Nagle algorithm: coalesce small packets before sending (wait for an ACK or a full buffer)
    // Pro: fewer small packets on the network
    // Con: increased latency
    // Typical effect: with a write-write-read pattern, the second write's data may
    // wait for the first write's ACK before being sent, causing unnecessary delay

    // 6. SO_RCVBUF / SO_SNDBUF: adjust the buffer sizes
    int rcvbuf = 256 * 1024;  // 256KB
    setsockopt(sock, SOL_SOCKET, SO_RCVBUF,
               &rcvbuf, sizeof(rcvbuf));

    int sndbuf = 256 * 1024;
    setsockopt(sock, SOL_SOCKET, SO_SNDBUF,
               &sndbuf, sizeof(sndbuf));

    // 7. SO_RCVTIMEO / SO_SNDTIMEO: set receive/send timeouts
    struct timeval tv;
    tv.tv_sec = 5;
    tv.tv_usec = 0;
    setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO,
               &tv, sizeof(tv));
    setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO,
               &tv, sizeof(tv));
}

// ============================================================
// Example 13-13: A complete RAII socket wrapper class
// ============================================================
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <unistd.h>
#include <fcntl.h>
#include <cstring>
#include <string>
#include <stdexcept>
#include <iostream>
#include <memory>

namespace net {

// Socket exception class (Chapter 9: custom exceptions)
class SocketException : public std::runtime_error {
private:
    int err_code_;
public:
    SocketException(const std::string& msg, int err = errno)
        : std::runtime_error(msg + ": " + std::strerror(err)),
          err_code_(err) {}
    int error_code() const { return err_code_; }
};

// RAII socket wrapper class (Chapter 4: classes and objects, Chapter 3: resource management)
class Socket {
private:
    int fd_;
    bool owned_;

    Socket(int fd, bool owned) : fd_(fd), owned_(owned) {}

public:
    Socket() : fd_(-1), owned_(false) {}

    explicit Socket(int domain, int type, int protocol = 0)
        : fd_(socket(domain, type, protocol)), owned_(true) {
        if (fd_ < 0) {
            throw SocketException("socket() failed");
        }
    }

    ~Socket() noexcept {
        if (owned_ && fd_ >= 0) {
            ::close(fd_);
        }
    }

    // Move semantics
    Socket(Socket&& other) noexcept
        : fd_(other.fd_), owned_(other.owned_) {
        other.fd_ = -1;
        other.owned_ = false;
    }

    Socket& operator=(Socket&& other) noexcept {
        if (this != &other) {
            if (owned_ && fd_ >= 0) ::close(fd_);
            fd_ = other.fd_;
            owned_ = other.owned_;
            other.fd_ = -1;
            other.owned_ = false;
        }
        return *this;
    }

    // Copying is disabled
    Socket(const Socket&) = delete;
    Socket& operator=(const Socket&) = delete;

    int fd() const { return fd_; }

    void set_reuse_addr(bool enable = true) {
        int opt = enable ? 1 : 0;
        if (setsockopt(fd_, SOL_SOCKET, SO_REUSEADDR,
                       &opt, sizeof(opt)) < 0) {
            throw SocketException("setsockopt SO_REUSEADDR failed");
        }
    }

    void set_nonblocking(bool enable = true) {
        int flags = fcntl(fd_, F_GETFL, 0);
        if (flags < 0) throw SocketException("fcntl F_GETFL failed");
        if (enable)
            flags |= O_NONBLOCK;
        else
            flags &= ~O_NONBLOCK;
        if (fcntl(fd_, F_SETFL, flags) < 0) {
            throw SocketException("fcntl F_SETFL failed");
        }
    }

    void bind(const std::string& ip, int port) {
        struct sockaddr_in addr;
        std::memset(&addr, 0, sizeof(addr));
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port);

        if (ip.empty() || ip == "0.0.0.0") {
            addr.sin_addr.s_addr = INADDR_ANY;
        } else {
            if (inet_pton(AF_INET, ip.c_str(),
                          &addr.sin_addr) <= 0) {
                throw SocketException("inet_pton() failed");
            }
        }

        if (::bind(fd_, (struct sockaddr*)&addr,
                   sizeof(addr)) < 0) {
            throw SocketException("bind() failed");
        }
    }

    void listen(int backlog = SOMAXCONN) {
        if (::listen(fd_, backlog) < 0) {
            throw SocketException("listen() failed");
        }
    }

    Socket accept() {
        struct sockaddr_in client_addr;
        socklen_t addr_len = sizeof(client_addr);
        int client_fd = ::accept(fd_,
                                  (struct sockaddr*)&client_addr,
                                  &addr_len);
        if (client_fd < 0) {
            throw SocketException("accept() failed");
        }
        return Socket(client_fd, true);
    }

    void connect(const std::string& ip, int port) {
        struct sockaddr_in addr;
        std::memset(&addr, 0, sizeof(addr));
        addr.sin_family = AF_INET;
        addr.sin_port = htons(port);

        if (inet_pton(AF_INET, ip.c_str(),
                      &addr.sin_addr) <= 0) {
            throw SocketException("inet_pton() failed");
        }

        if (::connect(fd_, (struct sockaddr*)&addr,
                      sizeof(addr)) < 0) {
            throw SocketException("connect() failed");
        }
    }

    ssize_t send(const void* data, size_t len,
                 int flags = 0) const {
        ssize_t result = ::send(fd_, data, len, MSG_NOSIGNAL | flags);
        if (result < 0) {
            throw SocketException("send() failed");
        }
        return result;
    }

    ssize_t send(const std::string& data) const {
        return this->send(data.c_str(), data.size());
    }

    ssize_t recv(void* buffer, size_t len,
                 int flags = 0) const {
        ssize_t result = ::recv(fd_, buffer, len, flags);
        if (result < 0) {
            throw SocketException("recv() failed");
        }
        return result;
    }

    std::string recv_str(size_t max_len = 4096) const {
        std::vector<char> buffer(max_len);
        ssize_t received = recv(buffer.data(), buffer.size() - 1, 0);
        if (received >= 0) {
            buffer[received] = '\0';
            return std::string(buffer.data());
        }
        return "";
    }

    void close() {
        if (owned_ && fd_ >= 0) {
            ::close(fd_);
            fd_ = -1;
        }
    }

    // Get the remote address
    std::string peer_address() const {
        struct sockaddr_in addr;
        socklen_t len = sizeof(addr);
        if (getpeername(fd_, (struct sockaddr*)&addr, &len) == 0) {
            char ip[INET_ADDRSTRLEN];
            inet_ntop(AF_INET, &addr.sin_addr, ip, sizeof(ip));
            return std::string(ip) + ":" +
                   std::to_string(ntohs(addr.sin_port));
        }
        return "unknown";
    }

    // Get the local address
    std::string local_address() const {
        struct sockaddr_in addr;
        socklen_t len = sizeof(addr);
        if (getsockname(fd_, (struct sockaddr*)&addr, &len) == 0) {
            char ip[INET_ADDRSTRLEN];
            inet_ntop(AF_INET, &addr.sin_addr, ip, sizeof(ip));
            return std::string(ip) + ":" +
                   std::to_string(ntohs(addr.sin_port));
        }
        return "unknown";
    }
};

} // namespace net

        case Method::POST:    return "POST";
        case Method::PUT:     return "PUT";
        case Method::DELETE:  return "DELETE";
        case Method::HEAD:    return "HEAD";
        case Method::OPTIONS: return "OPTIONS";
        case Method::PATCH:   return "PATCH";
        default:              return "UNKNOWN";
    }
}

// HTTP request
struct HttpRequest {
    Method method = Method::UNKNOWN;
    std::string path = "/";
    std::string version = "HTTP/1.1";
    std::map<std::string, std::string> headers;
    std::string body;

    std::string get_header(const std::string& key,
                           const std::string& default_val = "") const {
        auto it = headers.find(key);
        return (it != headers.end()) ? it->second : default_val;
    }
};

// HTTP response
struct HttpResponse {
    int status_code = 200;
    std::string status_message = "OK";
    std::string version = "HTTP/1.1";
    std::map<std::string, std::string> headers;
    std::string body;

    void set_header(const std::string& key,
                    const std::string& value) {
        headers[key] = value;
    }

    void set_content_type(const std::string& mime) {
        headers["Content-Type"] = mime;
    }

    void set_body(const std::string& content,
                  const std::string& content_type = "text/plain") {
        body = content;
        headers["Content-Type"] = content_type;
        headers["Content-Length"] = std::to_string(content.size());
    }
};

// HTTP request parser
class HttpParser {
public:
    // Parse the raw bytes of an HTTP request
    static HttpRequest parse_request(const std::string& raw) {
        HttpRequest req;
        std::istringstream stream(raw);
        std::string line;

        // 1. Parse the request line: METHOD PATH VERSION
        if (!std::getline(stream, line) || line.empty()) {
            throw std::runtime_error("Empty request");
        }

        // Strip \r
        if (!line.empty() && line.back() == '\r') {
            line.pop_back();
        }

        std::istringstream request_line(line);
        std::string method_str, path, version;
        request_line >> method_str >> path >> version;

        req.method = method_from_string(method_str);
        req.path = path;
        req.version = version;

        // 2. Parse the headers
        while (std::getline(stream, line)) {
            if (!line.empty() && line.back() == '\r') {
                line.pop_back();
            }
            if (line.empty()) break;  // A blank line marks the end of the headers

            size_t colon_pos = line.find(':');
            if (colon_pos != std::string::npos) {
                std::string key = line.substr(0, colon_pos);
                std::string value = line.substr(colon_pos + 1);
                // Trim leading whitespace
                size_t start = value.find_first_not_of(" \t");
                if (start != std::string::npos)
                    value = value.substr(start);
                req.headers[key] = value;
            }
        }

        // 3. Parse the request body (if there is a Content-Length)
        auto it = req.headers.find("Content-Length");
        if (it != req.headers.end()) {
            size_t len = std::stoul(it->second);
            req.body = raw.substr(raw.find("\r\n\r\n") + 4, len);
        }

        return req;
    }

    // Serialize an HttpRequest into a byte string
    static std::string serialize_request(const HttpRequest& req) {
        std::ostringstream oss;

        // Request line
        oss << method_to_string(req.method) << " "
            << req.path << " " << req.version << "\r\n";

        // Headers
        for (const auto& [key, value] : req.headers) {
            oss << key << ": " << value << "\r\n";
        }

        // Blank line + request body
        oss << "\r\n" << req.body;

        return oss.str();
    }

    // Serialize an HttpResponse into a byte string
    static std::string serialize_response(const HttpResponse& res) {
        std::ostringstream oss;

        // Status line
        oss << res.version << " " << res.status_code
            << " " << res.status_message << "\r\n";

        // Headers
        for (const auto& [key, value] : res.headers) {
            oss << key << ": " << value << "\r\n";
        }

        // Blank line + response body
        oss << "\r\n" << res.body;

        return oss.str();
    }
};

// Mapping of HTTP status codes to messages
struct HttpStatus {
    static std::string message(int code) {
        switch (code) {
            case 200: return "OK";
            case 201: return "Created";
            case 204: return "No Content";
            case 301: return "Moved Permanently";
            case 302: return "Found";
            case 304: return "Not Modified";
            case 400: return "Bad Request";
            case 401: return "Unauthorized";
            case 403: return "Forbidden";
            case 404: return "Not Found";
            case 405: return "Method Not Allowed";
            case 500: return "Internal Server Error";
            case 502: return "Bad Gateway";
            case 503: return "Service Unavailable";
            default:  return "Unknown";
        }
    }
};

The C++ ecosystem has some excellent networking libraries that can greatly simplify web programming:

   - Supports both HTTP clients and servers

   - Supports HTTPS (requires OpenSSL)
   - Good for quick prototypes and small projects

4. POCO C++ Libraries — an enterprise-grade C++ networking framework
   - Comprehensive features (HTTP/HTTPS/SMTP/FTP, etc.)
   - Mature ecosystem
   - Suited to large enterprise applications

5. Drogon — a high-performance C++ web framework
   - Supports HTTP/HTTPS/WebSocket
   - Asynchronous and non-blocking, based on epoll
   - Built-in ORM and template engine
   - Performance close to native C++

// URL parser
struct Url {
    std::string protocol;
    std::string host;
    int port;
    std::string path;

    static Url parse(const std::string& url) {
        Url result;
        result.protocol = "http";
        result.port = 80;

        size_t pos = 0;

        // Parse the protocol
        size_t proto_end = url.find("://");
        if (proto_end != std::string::npos) {
            result.protocol = url.substr(0, proto_end);
            pos = proto_end + 3;
        }

        if (result.protocol == "https") {
            result.port = 443;
            throw std::runtime_error("HTTPS not supported in this demo");
        }

        // Parse the host and path
        size_t path_start = url.find('/', pos);
        std::string host_port;

        if (path_start != std::string::npos) {
            host_port = url.substr(pos, path_start - pos);
            result.path = url.substr(path_start);
        } else {
            host_port = url.substr(pos);
            result.path = "/";
        }

        // Parse the port
        size_t port_pos = host_port.find(':');
        if (port_pos != std::string::npos) {
            result.host = host_port.substr(0, port_pos);
            result.port = std::stoi(host_port.substr(port_pos + 1));
        } else {
            result.host = host_port;
        }

        return result;
    }
};

// HTTP client (socket managed with RAII, Chapters 3/4)
class HttpClient {
private:
    int sock_;

    std::string resolve_host(const std::string& host) {
        struct addrinfo hints, *result;
        std::memset(&hints, 0, sizeof(hints));
        hints.ai_family = AF_INET;
        hints.ai_socktype = SOCK_STREAM;

        if (getaddrinfo(host.c_str(), nullptr, &hints, &result) != 0) {
            throw std::runtime_error("DNS resolution failed: " + host);
        }

        char ip[INET_ADDRSTRLEN];
        struct sockaddr_in* addr = (struct sockaddr_in*)result->ai_addr;
        inet_ntop(AF_INET, &addr->sin_addr, ip, sizeof(ip));

        std::string ip_str(ip);
        freeaddrinfo(result);
        return ip_str;
    }

public:
    HttpClient() : sock_(-1) {}

    ~HttpClient() {
        if (sock_ >= 0) close(sock_);
    }

    // Copying is disabled
    HttpClient(const HttpClient&) = delete;
    HttpClient& operator=(const HttpClient&) = delete;

    // GET request
    std::string get(const std::string& url) {
        Url parsed = Url::parse(url);

        std::cout << "Parsed URL:\n"
                  << "  Protocol: " << parsed.protocol << "\n"
                  << "  Host: " << parsed.host << "\n"
                  << "  Port: " << parsed.port << "\n"
                  << "  Path: " << parsed.path << std::endl;

        // Resolve DNS
        std::string ip = resolve_host(parsed.host);
        std::cout << "DNS resolved: " << parsed.host << " → "
                  << ip << std::endl;

        // Create the socket and connect
        sock_ = socket(AF_INET, SOCK_STREAM, 0);
        if (sock_ < 0) {
            throw std::runtime_error("socket() failed");
        }

        struct sockaddr_in addr;
        std::memset(&addr, 0, sizeof(addr));
        addr.sin_family = AF_INET;
        addr.sin_port = htons(parsed.port);
        inet_pton(AF_INET, ip.c_str(), &addr.sin_addr);

        if (connect(sock_, (struct sockaddr*)&addr,
                    sizeof(addr)) < 0) {
            throw std::runtime_error("connect() failed");
        }

        std::cout << "Connected!" << std::endl;

        // Build the HTTP GET request
        std::ostringstream request;
        request << "GET " << parsed.path << " HTTP/1.1\r\n";
        request << "Host: " << parsed.host << "\r\n";
        request << "User-Agent: SimpleHttpClient/1.0\r\n";
        request << "Accept: */*\r\n";
        request << "Connection: close\r\n";
        request << "\r\n";

        std::string req_str = request.str();
        std::cout << "Sending request:\n" << req_str << std::endl;

        // Send the request
        send(sock_, req_str.c_str(), req_str.size(), 0);

        // Receive the response
        std::string response;
        char buffer[4096];
        int received;
        while ((received = recv(sock_, buffer,
                                 sizeof(buffer) - 1, 0)) > 0) {
            buffer[received] = '\0';
            response += buffer;
        }

        close(sock_);
        sock_ = -1;

        return response;
    }

    // Parse an HTTP response, separating headers and body
    struct ParsedResponse {
        int status_code;
        std::string status_message;
        std::map<std::string, std::string> headers;
        std::string body;
    };

    static ParsedResponse parse_response(
        const std::string& raw_response) {
        ParsedResponse result;
        std::istringstream stream(raw_response);
        std::string line;

        // Parse the status line
        if (std::getline(stream, line)) {
            if (!line.empty() && line.back() == '\r')
                line.pop_back();
            size_t pos1 = line.find(' ');
            size_t pos2 = line.find(' ', pos1 + 1);
            result.status_code = std::stoi(
                line.substr(pos1 + 1, pos2 - pos1 - 1));
            result.status_message = line.substr(pos2 + 1);
        }

        // Parse the headers
        while (std::getline(stream, line)) {
            if (!line.empty() && line.back() == '\r')
                line.pop_back();
            if (line.empty()) break;

            size_t colon = line.find(':');
            if (colon != std::string::npos) {
                std::string key = line.substr(0, colon);
                std::string value = line.substr(colon + 2);
                result.headers[key] = value;
            }
        }

        // Read the body
        std::string remaining;
        while (std::getline(stream, line)) {
            remaining += line + "\n";
        }
        result.body = remaining;

        return result;
    }
};

void run() {
    std::cout << "\n========== Simple HTTP Client ==========\n\n";

    try {
        HttpClient client;
        std::string response = client.get(
            "http://example.com/");

        auto parsed = HttpClient::parse_response(response);

        std::cout << "\n===== HTTP Response =====\n";
        std::cout << "Status: " << parsed.status_code
                  << " " << parsed.status_message << "\n\n";

        std::cout << "Response headers:\n";
        for (const auto& [key, value] : parsed.headers) {
            std::cout << "  " << key << ": " << value << "\n";
        }

        std::cout << "\nResponse body (first 500 characters):\n";
        std::cout << parsed.body.substr(0, 500) << "\n";

    } catch (const std::exception& e) {
        std::cerr << "Error: " << e.what() << std::endl;
    }

    std::cout << "=====================================\n";
}

// MIME type mapping
std::string get_mime_type(const std::string& path) {
    static const std::map<std::string, std::string> mime_types = {
        {".html", "text/html"},
        {".htm",  "text/html"},
        {".css",  "text/css"},
        {".js",   "application/javascript"},
        {".json", "application/json"},
        {".png",  "image/png"},
        {".jpg",  "image/jpeg"},
        {".jpeg", "image/jpeg"},
        {".gif",  "image/gif"},
        {".svg",  "image/svg+xml"},
        {".ico",  "image/x-icon"},
        {".txt",  "text/plain"},
        {".pdf",  "application/pdf"},
        {".zip",  "application/zip"},
        {".mp3",  "audio/mpeg"},
        {".mp4",  "video/mp4"},
    };

    size_t dot = path.rfind('.');
    if (dot != std::string::npos) {
        std::string ext = path.substr(dot);
        auto it = mime_types.find(ext);
        if (it != mime_types.end()) {
            return it->second;
        }
    }
    return "application/octet-stream";
}

// HTTP response builder
class ResponseBuilder {
public:
    static std::string ok(const std::string& content,
                          const std::string& content_type) {
        std::ostringstream resp;
        resp << "HTTP/1.1 200 OK\r\n";
        resp << "Content-Type: " << content_type << "\r\n";
        resp << "Content-Length: " << content.size() << "\r\n";
        resp << "Connection: close\r\n";
        resp << "Server: SimpleStaticServer/1.0\r\n";
        resp << "\r\n";
        resp << content;
        return resp.str();
    }

    static std::string file_response(const std::string& filepath,
                                       const std::string& doc_root) {
        // Security check: guard against directory traversal attacks
        std::string full_path = doc_root + filepath;

        // If a directory is requested, try to return index.html
        if (filepath == "/" || filepath.back() == '/') {
            full_path = doc_root + filepath + "index.html";
        }

        std::ifstream file(full_path, std::ios::binary);
        if (!file.is_open()) {
            return not_found();
        }

        std::string content((std::istreambuf_iterator<char>(file)),
                            std::istreambuf_iterator<char>());

        std::string mime = get_mime_type(full_path);
        return ok(content, mime);
    }

    static std::string not_found() {
        std::string body = "<html><body><h1>404 Not Found</h1>"
                           "</body></html>";
        std::ostringstream resp;
        resp << "HTTP/1.1 404 Not Found\r\n";
        resp << "Content-Type: text/html\r\n";
        resp << "Content-Length: " << body.size() << "\r\n";
        resp << "Connection: close\r\n";
        resp << "\r\n";
        resp << body;
        return resp.str();
    }

    static std::string bad_request() {
        std::string body = "<html><body><h1>400 Bad Request</h1>"
                           "</body></html>";
        std::ostringstream resp;
        resp << "HTTP/1.1 400 Bad Request\r\n";
        resp << "Content-Type: text/html\r\n";
        resp << "Content-Length: " << body.size() << "\r\n";
        resp << "Connection: close\r\n";
        resp << "\r\n";
        resp << body;
        return resp.str();
    }

    static std::string internal_error() {
        std::string body = "<html><body><h1>500 Internal Server "
                           "Error</h1></body></html>";
        std::ostringstream resp;
        resp << "HTTP/1.1 500 Internal Server Error\r\n";
        resp << "Content-Type: text/html\r\n";
        resp << "Content-Length: " << body.size() << "\r\n";
        resp << "Connection: close\r\n";
        resp << "\r\n";
        resp << body;
        return resp.str();
    }
};

// Static file server
class StaticFileServer {
private:
    int server_sock_;
    std::string doc_root_;
    int port_;
    std::atomic<bool> running_{true};
    std::atomic<int> requests_served_{0};

    // Parse the HTTP request to get the path
    std::string parse_request_path(const std::string& request) {
        // Get the first line
        size_t first_line_end = request.find("\r\n");
        if (first_line_end == std::string::npos) return "/";

        std::string first_line = request.substr(0, first_line_end);

        // GET /path HTTP/1.1
        size_t method_end = first_line.find(' ');
        if (method_end == std::string::npos) return "/";

        size_t path_end = first_line.find(' ', method_end + 1);
        if (path_end == std::string::npos) return "/";

        return first_line.substr(method_end + 1,
                                  path_end - method_end - 1);
    }

    // Handle a single client connection
    void handle_client(int client_sock,
                       const std::string& client_ip) {
        char buffer[8192];
        int bytes = recv(client_sock, buffer, sizeof(buffer) - 1, 0);

        if (bytes > 0) {
            buffer[bytes] = '\0';
            std::string request(buffer);

            std::string path = parse_request_path(request);

            std::cout << "[" << client_ip << "] "
                      << path << std::endl;

            std::string response;
            try {
                response = ResponseBuilder::file_response(path, doc_root_);
            } catch (const std::exception&) {
                response = ResponseBuilder::internal_error();
            }

            send(client_sock, response.c_str(), response.size(), 0);
            requests_served_.fetch_add(1);
        }

        close(client_sock);
    }

public:
    StaticFileServer(const std::string& doc_root, int port = 8080)
        : server_sock_(-1), doc_root_(doc_root), port_(port) {}

    ~StaticFileServer() {
        stop();
    }

    void start() {
        // Create the socket
        server_sock_ = socket(AF_INET, SOCK_STREAM, 0);
        if (server_sock_ < 0) {
            throw std::runtime_error("socket() failed");
        }

        // Enable address reuse
        int reuse = 1;
        setsockopt(server_sock_, SOL_SOCKET, SO_REUSEADDR,
                   &reuse, sizeof(reuse));

        // Bind
        struct sockaddr_in addr;
        std::memset(&addr, 0, sizeof(addr));
        addr.sin_family = AF_INET;
        addr.sin_addr.s_addr = INADDR_ANY;
        addr.sin_port = htons(port_);

        if (bind(server_sock_, (struct sockaddr*)&addr,
                 sizeof(addr)) < 0) {
            throw std::runtime_error("bind() failed");
        }

        // Listen
        if (listen(server_sock_, SOMAXCONN) < 0) {
            throw std::runtime_error("listen() failed");
        }

        std::cout << "HTTP static file server started\n";
        std::cout << "Document root: " << doc_root_ << "\n";
        std::cout << "Listening on port: " << port_ << "\n";
        std::cout << "Visit http://localhost:" << port_ << "/\n\n";

        // Accept-connection loop
        while (running_.load()) {
            struct sockaddr_in client_addr;
            socklen_t addr_len = sizeof(client_addr);

            int client = accept(server_sock_,
                                 (struct sockaddr*)&client_addr,
                                 &addr_len);

            if (client < 0) {
                if (running_.load()) {
                    std::perror("accept failed");
                }
                continue;
            }

            // Get the client IP
            char ip[INET_ADDRSTRLEN];
            inet_ntop(AF_INET, &client_addr.sin_addr,
                      ip, sizeof(ip));

            // Handle each connection in its own thread (Chapter 12)
            std::thread(&StaticFileServer::handle_client,
                        this, client, std::string(ip)).detach();
        }
    }

    void stop() {
        running_.store(false);
        if (server_sock_ >= 0) {
            close(server_sock_);
            server_sock_ = -1;
        }
    }

    int requests_served() const {
        return requests_served_.load();
    }
};

// Signal handling (Chapter 11)
namespace {
    StaticFileServer* g_server = nullptr;

    void sig_handler(int sig) {
        std::cout << "\nReceived signal " << sig << ", stopping server...\n";
        if (g_server) {
            g_server->stop();
        }
    }
}

void run() {

    try {
        StaticFileServer server("./public", 8080);
        g_server = &server;

        std::signal(SIGINT, sig_handler);
        std::signal(SIGTERM, sig_handler);

        server.start();

        std::cout << "\nServer stopped. Handled "
                  << server.requests_served() << " requests.\n";

    } catch (const std::exception& e) {
        std::cerr << "Server error: " << e.what() << std::endl;
    }

    g_server = nullptr;
    std::cout << "===========================================\n";
}

// Simple JSON utilities (not a full JSON library; for demonstration only)
namespace json {
    std::string escape(const std::string& s) {
        std::string result;
        for (char c : s) {
            if (c == '"') result += "\\\"";
            else if (c == '\\') result += "\\\\";
            else if (c == '\n') result += "\\n";
            else result += c;
        }
        return result;
    }

    std::string object(const std::map<std::string, std::string>& kv) {
        std::ostringstream oss;
        oss << "{";
        bool first = true;
        for (const auto& [k, v] : kv) {
            if (!first) oss << ", ";
            oss << "\"" << k << "\": \"" << escape(v) << "\"";
            first = false;
        }
        oss << "}";
        return oss.str();
    }

    std::string array(const std::vector<std::string>& items) {
        std::ostringstream oss;
        oss << "[";
        for (size_t i = 0; i < items.size(); ++i) {
            if (i > 0) oss << ", ";
            oss << items[i];
        }
        oss << "]";
        return oss.str();
    }
}

// HTTP request
struct Request {
    std::string method;
    std::string path;
    std::map<std::string, std::string> headers;
    std::string body;
};

// HTTP response
struct Response {
    int status_code = 200;
    std::string body;
    std::string content_type = "text/plain";
};

// Route handler interface (Chapter 6: polymorphism)
class IHandler {
public:
    virtual ~IHandler() = default;
    virtual Response handle(const Request& req) = 0;
};

// === TODO data model ===
struct TodoItem {
    int id;
    std::string title;
    bool completed;
    std::string created_at;

    std::string to_json() const {
        return json::object({
            {"id", std::to_string(id)},
            {"title", title},
            {"completed", completed ? "true" : "false"},
            {"created_at", created_at}
        });
    }
};

// TODO storage (thread-safe)
class TodoStore {
private:
    std::vector<TodoItem> items_;
    mutable std::mutex mtx_;
    int next_id_ = 1;

public:
    std::vector<TodoItem> get_all() const {
        std::lock_guard<std::mutex> lock(mtx_);
        return items_;
    }

    TodoItem get_by_id(int id) const {
        std::lock_guard<std::mutex> lock(mtx_);
        for (const auto& item : items_) {
            if (item.id == id) return item;
        }
        throw std::runtime_error("TODO not found");
    }

    TodoItem create(const std::string& title) {
        std::lock_guard<std::mutex> lock(mtx_);
        TodoItem item;
        item.id = next_id_++;
        item.title = title;
        item.completed = false;
        item.created_at = "2024-01-01T00:00:00Z";
        items_.push_back(item);
        return item;
    }

    TodoItem update(int id, const std::string& title,
                     bool completed) {
        std::lock_guard<std::mutex> lock(mtx_);
        for (auto& item : items_) {
            if (item.id == id) {
                item.title = title;
                item.completed = completed;
                return item;
            }
        }
        throw std::runtime_error("TODO not found");
    }

    bool remove(int id) {
        std::lock_guard<std::mutex> lock(mtx_);
        auto it = std::find_if(items_.begin(), items_.end(),
            [id](const TodoItem& item) { return item.id == id; });
        if (it != items_.end()) {
            items_.erase(it);
            return true;
        }
        return false;
    }
};

// === Router ===
class Router {
private:
    std::map<std::string,
             std::map<std::string,
                      std::unique_ptr<IHandler>>> routes_;

public:
    void add_route(const std::string& method,
                   const std::string& path_pattern,
                   std::unique_ptr<IHandler> handler) {
        routes_[method][path_pattern] = std::move(handler);
    }

    Response route(const Request& req) {
        Response resp;
        resp.status_code = 404;
        resp.body = "Not Found";

        auto method_it = routes_.find(req.method);
        if (method_it == routes_.end()) {
            resp.status_code = 405;
            resp.body = "Method Not Allowed";
            return resp;
        }

        // First check for an exact match
        auto& method_routes = method_it->second;
        auto exact = method_routes.find(req.path);
        if (exact != method_routes.end()) {
            return exact->second->handle(req);
        }

        // Then check parameterized routes (e.g. /api/todos/123)
        for (auto& [pattern, handler] : method_routes) {
            // Simple pattern: if pattern ends with /api/todos/ and req.path is longer
            if (pattern == "/api/todos/" ||
                pattern == "/api/todos") {
                // Special handling
            }
        }

        return resp;
    }
};

// Parse an HTTP request
Request parse_http_request(const std::string& raw) {
    Request req;
    std::istringstream stream(raw);
    std::string line;

    // Request line
    if (std::getline(stream, line)) {
        if (!line.empty() && line.back() == '\r')
            line.pop_back();
        std::istringstream req_line(line);
        req_line >> req.method >> req.path;
    }

    // Headers
    while (std::getline(stream, line)) {
        if (!line.empty() && line.back() == '\r')
            line.pop_back();
        if (line.empty()) break;
        size_t colon = line.find(':');
        if (colon != std::string::npos) {
            std::string key = line.substr(0, colon);
            std::string val = line.substr(colon + 2);
            req.headers[key] = val;
        }
    }

    // Request body
    auto it = req.headers.find("Content-Length");
    if (it != req.headers.end()) {
        int len = std::stoi(it->second);
        req.body.resize(len);
        stream.read(&req.body[0], len);
    }

    return req;
}

// Build an HTTP response
std::string build_http_response(const Response& resp) {
    std::ostringstream oss;
    oss << "HTTP/1.1 " << resp.status_code;
    switch (resp.status_code) {
        case 200: oss << " OK"; break;
        case 201: oss << " Created"; break;
        case 400: oss << " Bad Request"; break;
        case 404: oss << " Not Found"; break;
        case 405: oss << " Method Not Allowed"; break;
        case 500: oss << " Internal Server Error"; break;
    }
    oss << "\r\n";
    oss << "Content-Type: " << resp.content_type << "\r\n";
    oss << "Content-Length: " << resp.body.size() << "\r\n";
    oss << "Connection: close\r\n";
    oss << "Access-Control-Allow-Origin: *\r\n";
    oss << "\r\n";
    oss << resp.body;
    return oss.str();
}

// API server
class ApiServer {
private:
    Router router_;
    int port_;
    std::atomic<bool> running_{true};

public:
    explicit ApiServer(int port = 8080) : port_(port) {}

    Router& get_router() { return router_; }

    void start() {
        int server_sock = socket(AF_INET, SOCK_STREAM, 0);

        int reuse = 1;
        setsockopt(server_sock, SOL_SOCKET, SO_REUSEADDR,
                   &reuse, sizeof(reuse));

        struct sockaddr_in addr;
        std::memset(&addr, 0, sizeof(addr));
        addr.sin_family = AF_INET;
        addr.sin_addr.s_addr = INADDR_ANY;
        addr.sin_port = htons(port_);

        bind(server_sock, (struct sockaddr*)&addr, sizeof(addr));
        listen(server_sock, 10);

        std::cout << "REST API server started on port " << port_
                  << "\n";

        while (running_.load()) {
            struct sockaddr_in client_addr;
            socklen_t len = sizeof(client_addr);
            int client = accept(server_sock,
                                 (struct sockaddr*)&client_addr, &len);
            if (client < 0) continue;

            // Handle the request (in a detached thread)
            std::thread([this, client]() {
                char buffer[8192];
                int bytes = recv(client, buffer,
                                  sizeof(buffer) - 1, 0);
                if (bytes > 0) {
                    buffer[bytes] = '\0';
                    Request req = parse_http_request(std::string(buffer));

                    Response resp;
                    try {
                        resp = router_.route(req);
                    } catch (const std::exception& e) {
                        resp.status_code = 500;
                        resp.body = json::object(
                            {{"error", e.what()}});
                        resp.content_type = "application/json";
                    }

                    std::string http_resp = build_http_response(resp);
                    send(client, http_resp.c_str(),
                         http_resp.size(), 0);
                }
                close(client);
            }).detach();
        }

        close(server_sock);
    }

    void stop() { running_.store(false); }
};

void run() {
    std::cout << "\n========== RESTful API Server ==========\n\n";

    TodoStore store;
    ApiServer server(8080);

    // Register routes (Chapter 6: polymorphism)
    Router& router = server.get_router();
    router.add_route("GET", "/api/todos",
        std::make_unique<GetAllTodosHandler>(store));
    router.add_route("POST", "/api/todos",
        std::make_unique<CreateTodoHandler>(store));
    // Parameterized routes are simplified into a generic pattern
    router.add_route("GET", "/api/todos/",
        std::make_unique<GetTodoHandler>(store));
    router.add_route("DELETE", "/api/todos/",
        std::make_unique<DeleteTodoHandler>(store));

    // Add some sample data
    store.create("Learn C++ network programming");
    store.create("Finish the HTTP server project");
    store.create("Read the socket programming docs");

    std::cout << "Available endpoints:\n";
    std::cout << "  GET    /api/todos      - Get all TODOs\n";
    std::cout << "  POST   /api/todos      - Create a new TODO\n";
    std::cout << "  GET    /api/todos/1    - Get TODO details\n";
    std::cout << "  DELETE /api/todos/1    - Delete a TODO\n\n";

    server.start();

    std::cout << "==========================================\n";
}

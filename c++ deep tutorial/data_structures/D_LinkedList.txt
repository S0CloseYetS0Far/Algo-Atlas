## Chapter Overview

A linked list is a linear data structure whose elements are linked together by pointers, rather than
being stored contiguously in memory like an array. Each node of a linked list contains a data field and a
pointer field, and the pointer points to the next node.

Advantages of linked lists: insertion and removal are efficient (O(1)) and don't require moving other
elements; space is allocated dynamically, with no need to pre-allocate.
Disadvantages of linked lists: no random access (reaching an arbitrary element takes O(n)); each node needs
extra storage for pointers; they are cache-unfriendly (nodes are scattered throughout memory).

1.1 Basic Linked List Concepts
------------------------------

Common kinds of linked lists:

Singly Linked List: each node contains data + a pointer to the next node.
Doubly Linked List: each node contains data + pointers to the previous and next nodes.
Circular Linked List: the tail node points back to the head node, forming a ring.

```cpp
#include <iostream>

struct SinglyNode {
    int data;
    SinglyNode* next;

    SinglyNode(int val) : data(val), next(nullptr) {}
};

struct DoublyNode {
    int data;
    DoublyNode* prev;
    DoublyNode* next;

    DoublyNode(int val) : data(val), prev(nullptr), next(nullptr) {}
};

int main() {
    // Create a singly linked list: 10 -> 20 -> 30
    SinglyNode* head = new SinglyNode(10);
    head->next = new SinglyNode(20);
    head->next->next = new SinglyNode(30);

    // Traverse the singly linked list
    std::cout << "Singly linked list: ";
    SinglyNode* cur = head;
    while (cur) {
        std::cout << cur->data;
        if (cur->next) std::cout << " -> ";
        cur = cur->next;
    }
    std::cout << std::endl;

    // Create a doubly linked list: 10 <-> 20 <-> 30
    DoublyNode* dhead = new DoublyNode(10);
    DoublyNode* dsecond = new DoublyNode(20);
    DoublyNode* dthird = new DoublyNode(30);

    dhead->next = dsecond;
    dsecond->prev = dhead;
    dsecond->next = dthird;
    dthird->prev = dsecond;

    // Forward traversal
    std::cout << "Doubly linked list (forward): ";
    DoublyNode* dcur = dhead;
    while (dcur) {
        std::cout << dcur->data;
        if (dcur->next) std::cout << " <-> ";
        dcur = dcur->next;
    }
    std::cout << std::endl;

    // Backward traversal
    std::cout << "Doubly linked list (backward): ";
    dcur = dthird;
    while (dcur) {
        std::cout << dcur->data;
        if (dcur->prev) std::cout << " <-> ";
        dcur = dcur->prev;
    }
    std::cout << std::endl;

    // Memory cleanup (omitted)

    return 0;
}
```

1.2 Under the Hood: Memory Layout
---------------------------------

Linked list nodes are stored non-contiguously in memory. Each node is allocated independently on the heap via new.

```
Memory layout illustration (addresses are arbitrary):

0x1000: [10 | ptr(0x2000)]       <- head node
0x2000: [20 | ptr(0x3000)]       <- second node
0x3000: [30 | ptr(nullptr)]      <- tail node

Array (for comparison):
0x1000: [10]                      <- arr[0]
0x1004: [20]                      <- arr[1]
0x1008: [30]                      <- arr[2]
        contiguous addresses, right next to each other
```

This non-contiguous storage means:
1. Elements can't be accessed directly by index (you have to traverse from the head)
2. Low CPU cache hit rate (poor spatial locality)
3. Insertion/removal only requires updating pointers, not moving data

1.3 Implementing a Singly Linked List by Hand
---------------------------------------------

```cpp
#include <iostream>
#include <stdexcept>

template<typename T>
class LinkedList {
private:
    struct Node {
        T data;
        Node* next;
        Node(const T& val) : data(val), next(nullptr) {}
    };

    Node* head;
    size_t count;

public:
    LinkedList() : head(nullptr), count(0) {}

    ~LinkedList() {
        while (head) {
            Node* temp = head;
            head = head->next;
            delete temp;
        }
    }

    // Insert at the head
    void push_front(const T& value) {
        Node* new_node = new Node(value);
        new_node->next = head;
        head = new_node;
        ++count;
    }

    // Insert at the tail
    void push_back(const T& value) {
        Node* new_node = new Node(value);
        if (!head) {
            head = new_node;
        } else {
            Node* cur = head;
            while (cur->next) cur = cur->next;
            cur->next = new_node;
        }
        ++count;
    }

    // Insert in the middle (after the given position)
    void insert_after(size_t index, const T& value) {
        if (index >= count) {
            throw std::out_of_range("Index out of range");
        }
        Node* cur = head;
        for (size_t i = 0; i < index; ++i) cur = cur->next;

        Node* new_node = new Node(value);
        new_node->next = cur->next;
        cur->next = new_node;
        ++count;
    }

    // Remove from the head
    void pop_front() {
        if (!head) throw std::underflow_error("List is empty");
        Node* temp = head;
        head = head->next;
        delete temp;
        --count;
    }

    // Remove from the tail
    void pop_back() {
        if (!head) throw std::underflow_error("List is empty");
        if (!head->next) {
            delete head;
            head = nullptr;
        } else {
            Node* cur = head;
            while (cur->next->next) cur = cur->next;
            delete cur->next;
            cur->next = nullptr;
        }
        --count;
    }

    // Remove by value (removes the first match)
    void remove(const T& value) {
        if (!head) return;

        if (head->data == value) {
            Node* temp = head;
            head = head->next;
            delete temp;
            --count;
            return;
        }

        Node* cur = head;
        while (cur->next && cur->next->data != value) {
            cur = cur->next;
        }
        if (cur->next) {
            Node* temp = cur->next;
            cur->next = cur->next->next;
            delete temp;
            --count;
        }
    }

    // Search
    bool contains(const T& value) const {
        Node* cur = head;
        while (cur) {
            if (cur->data == value) return true;
            cur = cur->next;
        }
        return false;
    }

    // Reverse the list
    void reverse() {
        Node* prev = nullptr;
        Node* cur = head;
        Node* next = nullptr;

        while (cur) {
            next = cur->next;
            cur->next = prev;
            prev = cur;
            cur = next;
        }
        head = prev;
    }

    // Get the element at index (0-based)
    T& at(size_t index) {
        if (index >= count) throw std::out_of_range("Index out of range");
        Node* cur = head;
        for (size_t i = 0; i < index; ++i) cur = cur->next;
        return cur->data;
    }

    void print() const {
        Node* cur = head;
        while (cur) {
            std::cout << cur->data;
            if (cur->next) std::cout << " -> ";
            cur = cur->next;
        }
        std::cout << " (size=" << count << ")" << std::endl;
    }

    size_t size() const { return count; }
    bool empty() const { return count == 0; }
};

int main() {
    LinkedList<int> lst;

    lst.push_back(10);
    lst.push_back(20);
    lst.push_back(30);
    lst.push_front(5);
    lst.print();

    lst.insert_after(1, 15);
    lst.print();

    lst.pop_front();
    lst.print();

    lst.reverse();
    std::cout << "After reversing: ";
    lst.print();

    std::cout << "Contains 20? " << lst.contains(20) << std::endl;
    std::cout << "at(2) = " << lst.at(2) << std::endl;

    lst.remove(20);
    lst.print();

    return 0;
}
```

1.4 Implementing a Doubly Linked List by Hand
---------------------------------------------

```cpp
#include <iostream>
#include <stdexcept>

template<typename T>
class DoublyLinkedList {
private:
    struct Node {
        T data;
        Node* prev;
        Node* next;
        Node(const T& val) : data(val), prev(nullptr), next(nullptr) {}
    };

    Node* head;
    Node* tail;
    size_t count;

public:
    DoublyLinkedList() : head(nullptr), tail(nullptr), count(0) {}

    ~DoublyLinkedList() {
        while (head) {
            Node* temp = head;
            head = head->next;
            delete temp;
        }
    }

    void push_back(const T& value) {
        Node* new_node = new Node(value);
        if (!head) {
            head = tail = new_node;
        } else {
            new_node->prev = tail;
            tail->next = new_node;
            tail = new_node;
        }
        ++count;
    }

    void push_front(const T& value) {
        Node* new_node = new Node(value);
        if (!head) {
            head = tail = new_node;
        } else {
            new_node->next = head;
            head->prev = new_node;
            head = new_node;
        }
        ++count;
    }

    void pop_back() {
        if (!tail) throw std::underflow_error("List is empty");
        Node* temp = tail;
        tail = tail->prev;
        if (tail) tail->next = nullptr;
        else head = nullptr;
        delete temp;
        --count;
    }

    void print_forward() const {
        Node* cur = head;
        while (cur) {
            std::cout << cur->data;
            if (cur->next) std::cout << " <-> ";
            cur = cur->next;
        }
        std::cout << std::endl;
    }

    void print_backward() const {
        Node* cur = tail;
        while (cur) {
            std::cout << cur->data;
            if (cur->prev) std::cout << " <-> ";
            cur = cur->prev;
        }
        std::cout << std::endl;
    }

    size_t size() const { return count; }
};

int main() {
    DoublyLinkedList<int> dll;
    dll.push_back(10);
    dll.push_back(20);
    dll.push_back(30);
    dll.push_front(5);

    std::cout << "Forward: ";
    dll.print_forward();
    std::cout << "Backward: ";
    dll.print_backward();

    return 0;
}
```

2.1 std::forward_list -- Singly Linked List
-------------------------------------------

Introduced in C++11, it uses less memory than list (only one pointer per node).

```cpp
#include <iostream>
#include <forward_list>

int main() {
    std::forward_list<int> flst = {1, 2, 3, 4, 5};

    // Operations at the head
    flst.push_front(0);
    flst.pop_front();

    // Operations after a given position (require the predecessor iterator)
    auto it = flst.before_begin();  // head sentinel
    flst.insert_after(it, 99);     // insert after the head position

    // Find and insert
    auto prev = flst.before_begin();
    for (auto cur = flst.begin(); cur != flst.end(); ++cur, ++prev) {
        if (*cur == 3) {
            flst.insert_after(cur, 100);  // insert 100 after 3
            break;
        }
    }

    // Removal
    flst.erase_after(flst.before_begin());  // erase the first element

    // Specific operations
    flst.sort();
    flst.unique();        // remove consecutive duplicate elements
    flst.reverse();

    // Splicing
    std::forward_list<int> other = {200, 300};
    flst.splice_after(flst.before_begin(), other);  // splice other onto the front

    // Merge (both lists must already be sorted)
    std::forward_list<int> a = {1, 3, 5};
    std::forward_list<int> b = {2, 4, 6};
    a.merge(b);

    for (int x : a) std::cout << x << " ";
    std::cout << std::endl;

    return 0;
}
```

2.2 std::list -- Doubly Linked List
-----------------------------------

```cpp
#include <iostream>
#include <list>
#include <algorithm>

int main() {
    std::list<int> lst = {3, 1, 4, 1, 5, 9};

    // Operations at both ends
    lst.push_front(0);
    lst.push_back(10);
    lst.pop_front();
    lst.pop_back();

    // Insertion (before pos)
    auto it = std::find(lst.begin(), lst.end(), 4);
    if (it != lst.end()) {
        lst.insert(it, 100);      // insert before 4
        lst.emplace(it, 200);     // C++11 in-place construction
    }

    // Removal
    lst.remove(1);                // remove all elements equal to 1
    lst.remove_if([](int x) { return x > 5; });  // conditional removal

    // List-specific operations
    lst.sort();
    lst.unique();                 // remove duplicates
    lst.reverse();

    // Splicing (moves other's elements into lst)
    std::list<int> other = {100, 200};
    lst.splice(lst.end(), other);  // other becomes empty

    // Merge (both lists must already be sorted)
    std::list<int> a = {1, 3, 5};
    std::list<int> b = {2, 4, 6};
    a.merge(b);  // a becomes {1,2,3,4,5,6}, b is empty

    // resize
    lst.resize(10);    // extend or truncate
    lst.resize(20, -1); // fill new elements with -1

    // Assignment
    lst.assign(5, 100);  // replace everything with five 100s

    // Traversal
    for (int x : lst) std::cout << x << " ";
    std::cout << std::endl;

    return 0;
}
```

2.3 Linked List Algorithms (Common Operations)
----------------------------------------------

```cpp
#include <iostream>
#include <forward_list>
#include <unordered_set>

// 1. Detect whether a linked list has a cycle (fast and slow pointers)
struct ListNode {
    int val;
    ListNode* next;
    ListNode(int v) : val(v), next(nullptr) {}
};

bool hasCycle(ListNode* head) {
    ListNode* slow = head;
    ListNode* fast = head;

    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return true;
    }
    return false;
}

// 2. Find the middle node of a linked list
ListNode* findMiddle(ListNode* head) {
    ListNode* slow = head;
    ListNode* fast = head;

    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
    }
    return slow;
}

// 3. Merge two sorted linked lists
ListNode* mergeTwoLists(ListNode* l1, ListNode* l2) {
    ListNode dummy(0);
    ListNode* tail = &dummy;

    while (l1 && l2) {
        if (l1->val <= l2->val) {
            tail->next = l1;
            l1 = l1->next;
        } else {
            tail->next = l2;
            l2 = l2->next;
        }
        tail = tail->next;
    }
    tail->next = l1 ? l1 : l2;

    return dummy.next;
}

// 4. Remove the Nth node from the end
ListNode* removeNthFromEnd(ListNode* head, int n) {
    ListNode dummy(0);
    dummy.next = head;
    ListNode* fast = &dummy;
    ListNode* slow = &dummy;

    // fast moves n+1 steps ahead first
    for (int i = 0; i <= n; ++i) fast = fast->next;

    while (fast) {
        fast = fast->next;
        slow = slow->next;
    }

    ListNode* to_delete = slow->next;
    slow->next = slow->next->next;
    delete to_delete;

    return dummy.next;
}

// Helper function
void printList(ListNode* head) {
    while (head) {
        std::cout << head->val;
        if (head->next) std::cout << " -> ";
        head = head->next;
    }
    std::cout << std::endl;
}

int main() {
    // Merge sorted linked lists
    ListNode* l1 = new ListNode(1);
    l1->next = new ListNode(3);
    l1->next->next = new ListNode(5);

    ListNode* l2 = new ListNode(2);
    l2->next = new ListNode(4);
    l2->next->next = new ListNode(6);

    ListNode* merged = mergeTwoLists(l1, l2);
    std::cout << "After merging: ";
    printList(merged);

    // Middle node
    ListNode* mid = findMiddle(merged);
    std::cout << "Middle node: " << mid->val << std::endl;

    // Remove the 3rd node from the end
    ListNode* after_remove = removeNthFromEnd(merged, 3);
    std::cout << "After removing the 3rd from the end: ";
    printList(after_remove);

    return 0;
}
```

Example 1: Browser Back and Forward (Doubly Linked List Version)
----------------------------------------------------------------

Unlike the stack implementation, here a doubly linked list is used to manage page history more intuitively:

```cpp
#include <iostream>
#include <string>

class BrowserHistory {
private:
    struct Page {
        std::string url;
        Page* prev;
        Page* next;
        Page(const std::string& u) : url(u), prev(nullptr), next(nullptr) {}
    };

    Page* current;

public:
    BrowserHistory(const std::string& homepage) {
        current = new Page(homepage);
    }

    void visit(const std::string& url) {
        // Clear the forward history
        Page* temp = current->next;
        while (temp) {
            Page* to_del = temp;
            temp = temp->next;
            delete to_del;
        }

        current->next = new Page(url);
        current->next->prev = current;
        current = current->next;
        std::cout << "Visit: " << url << std::endl;
    }

    std::string back(int steps) {
        while (steps-- > 0 && current->prev) {
            current = current->prev;
        }
        std::cout << "Back to: " << current->url << std::endl;
        return current->url;
    }

    std::string forward(int steps) {
        while (steps-- > 0 && current->next) {
            current = current->next;
        }
        std::cout << "Forward to: " << current->url << std::endl;
        return current->url;
    }

    ~BrowserHistory() {
        while (current->prev) current = current->prev;
        while (current) {
            Page* temp = current;
            current = current->next;
            delete temp;
        }
    }
};

int main() {
    BrowserHistory bh("google.com");

    bh.visit("github.com");
    bh.visit("stackoverflow.com");
    bh.visit("cppreference.com");

    bh.back(1);   // stackoverflow.com
    bh.back(1);   // github.com
    bh.forward(1); // stackoverflow.com
    bh.visit("reddit.com");  // forward history is cleared

    bh.back(2);   // github.com -> google.com

    return 0;
}
```


Example 2: The Josephus Problem (Circular Linked List)
------------------------------------------------------

The classic Josephus problem: n people stand in a circle and count off starting from the first person;
whoever counts m is eliminated. Find the last person remaining.

```cpp
#include <iostream>

struct Person {
    int id;
    Person* next;
    Person(int i) : id(i), next(nullptr) {}
};

int josephus(int n, int m) {
    // Build the circular linked list
    Person* head = new Person(1);
    Person* prev = head;

    for (int i = 2; i <= n; ++i) {
        Person* p = new Person(i);
        prev->next = p;
        prev = p;
    }
    prev->next = head;  // close the ring

    // Start the game
    Person* cur = head;
    Person* last = prev;

    while (cur->next != cur) {  // stop when only one person is left
        // Count up to m-1
        for (int i = 1; i < m; ++i) {
            last = cur;
            cur = cur->next;
        }

        // cur is eliminated
        std::cout << cur->id << " eliminated" << std::endl;
        last->next = cur->next;
        delete cur;
        cur = last->next;
    }

    int survivor = cur->id;
    delete cur;
    return survivor;
}

int main() {
    int n = 7, m = 3;
    int survivor = josephus(n, m);
    std::cout << n << " people, counting to " << m << ", survivor: " << survivor << std::endl;

    return 0;
}
```


Example 3: Representing and Operating on Polynomials with Linked Lists
----------------------------------------------------------------------

Using a linked list to represent polynomials and implementing addition:

```cpp
#include <iostream>

struct Term {
    int coefficient;  // coefficient
    int exponent;     // exponent
    Term* next;
    Term(int c, int e) : coefficient(c), exponent(e), next(nullptr) {}
};

class Polynomial {
private:
    Term* head;

    // Helper: append a term at the tail
    void append(int coeff, int exp) {
        if (coeff == 0) return;  // terms with a coefficient of 0 are not added
        if (!head) {
            head = new Term(coeff, exp);
            return;
        }
        Term* cur = head;
        while (cur->next) cur = cur->next;
        cur->next = new Term(coeff, exp);
    }

public:
    Polynomial() : head(nullptr) {}

    Polynomial(const std::initializer_list<std::pair<int, int>>& terms) : head(nullptr) {
        for (auto& [c, e] : terms) {
            append(c, e);
        }
    }

    ~Polynomial() {
        while (head) {
            Term* temp = head;
            head = head->next;
            delete temp;
        }
    }

    // Polynomial addition
    Polynomial add(const Polynomial& other) const {
        Polynomial result;
        Term* p = head;
        Term* q = other.head;

        while (p && q) {
            if (p->exponent > q->exponent) {
                result.append(p->coefficient, p->exponent);
                p = p->next;
            } else if (p->exponent < q->exponent) {
                result.append(q->coefficient, q->exponent);
                q = q->next;
            } else {
                result.append(p->coefficient + q->coefficient, p->exponent);
                p = p->next;
                q = q->next;
            }
        }

        while (p) { result.append(p->coefficient, p->exponent); p = p->next; }
        while (q) { result.append(q->coefficient, q->exponent); q = q->next; }

        return result;
    }

    void print() const {
        if (!head) { std::cout << "0"; return; }
        Term* cur = head;
        while (cur) {
            if (cur != head && cur->coefficient > 0) std::cout << " + ";
            if (cur->coefficient < 0) std::cout << " - ";
            int abs_c = cur->coefficient > 0 ? cur->coefficient : -cur->coefficient;

            if (cur->exponent == 0) {
                std::cout << abs_c;
            } else if (cur->exponent == 1) {
                std::cout << abs_c << "x";
            } else {
                std::cout << abs_c << "x^" << cur->exponent;
            }
            cur = cur->next;
        }
    }
};

int main() {
    Polynomial p1({{3, 2}, {2, 1}, {1, 0}});     // 3x^2 + 2x + 1
    Polynomial p2({{5, 3}, {-1, 2}, {4, 0}});    // 5x^3 - x^2 + 4

    Polynomial sum = p1.add(p2);

    std::cout << "P1 = "; p1.print(); std::cout << std::endl;
    std::cout << "P2 = "; p2.print(); std::cout << std::endl;
    std::cout << "Sum = "; sum.print(); std::cout << std::endl;

    return 0;
}
```

## Chapter Overview

A Graph is a non-linear data structure made up of vertices and the edges that connect
them. Graphs are among the most flexible data structures and can represent all kinds of
complex relationships: social networks, map navigation, web page links, dependencies,
circuit design, and more.

Graph theory is an important branch of computer science. This chapter starts with the
basic concepts of graphs, digs into how graphs are stored and traversed, covers the
common graph algorithms in depth, and finally reinforces what you've learned with examples and exercises.

1.1 Basic Concepts of Graphs
--------------------

Components of a graph:
- Vertex: a node in the graph
- Edge: a line connecting two vertices

Kinds of graphs:
- Directed vs undirected
- Weighted vs unweighted
- Connected vs disconnected
- Dense vs sparse

Related terminology:
- Degree: the number of edges incident to a vertex
- Out-degree: in a directed graph, the number of edges leaving the vertex
- In-degree: in a directed graph, the number of edges pointing into the vertex
- Path: a sequence of vertices where each pair of adjacent vertices is connected by an edge
- Cycle: a path whose start vertex equals its end vertex

1.2 Ways to Store a Graph
--------------------

(1) Adjacency Matrix
The graph is represented by a 2D array: arr[i][j] = 1 (or a weight) means there is an edge from vertex i to j.

Pros: checking whether two vertices are connected is O(1)
Cons: O(V^2) space, which wastes memory for sparse graphs

```cpp
#include <iostream>
#include <vector>

class AdjacencyMatrixGraph {
private:
    int vertices;
    std::vector<std::vector<int>> matrix;  // 0 = no edge, 1 = edge

public:
    AdjacencyMatrixGraph(int v) : vertices(v), matrix(v, std::vector<int>(v, 0)) {}

    void addEdge(int u, int v, bool directed = false) {
        matrix[u][v] = 1;
        if (!directed) matrix[v][u] = 1;
    }

    void removeEdge(int u, int v) {
        matrix[u][v] = 0;
        matrix[v][u] = 0;
    }

    bool hasEdge(int u, int v) const {
        return matrix[u][v] != 0;
    }

    void print() const {
        std::cout << "   ";
        for (int i = 0; i < vertices; ++i) std::cout << i << " ";
        std::cout << std::endl;

        for (int i = 0; i < vertices; ++i) {
            std::cout << i << ": ";
            for (int j = 0; j < vertices; ++j) {
                std::cout << matrix[i][j] << " ";
            }
            std::cout << std::endl;
        }
    }
};

int main() {
    AdjacencyMatrixGraph g(4);
    g.addEdge(0, 1);
    g.addEdge(0, 2);
    g.addEdge(1, 2);
    g.addEdge(2, 3);

    g.print();

    std::cout << "Edge 0-2? " << g.hasEdge(0, 2) << std::endl;
    std::cout << "Edge 0-3? " << g.hasEdge(0, 3) << std::endl;

    return 0;
}
```

(2) Adjacency List
Each vertex maintains a list/vector of the vertices adjacent to it.

Pros: O(V+E) space, well suited to sparse graphs
Cons: checking whether two vertices are connected takes O(degree)

```cpp
#include <iostream>
#include <vector>
#include <list>

class AdjacencyListGraph {
private:
    int vertices;
    std::vector<std::list<int>> adj_list;

public:
    AdjacencyListGraph(int v) : vertices(v), adj_list(v) {}

    void addEdge(int u, int v, bool directed = false) {
        adj_list[u].push_back(v);
        if (!directed) adj_list[v].push_back(u);
    }

    void removeEdge(int u, int v) {
        adj_list[u].remove(v);
        adj_list[v].remove(u);
    }

    bool hasEdge(int u, int v) const {
        for (int w : adj_list[u]) {
            if (w == v) return true;
        }
        return false;
    }

    std::list<int> getNeighbors(int v) const {
        return adj_list[v];
    }

    void print() const {
        for (int i = 0; i < vertices; ++i) {
            std::cout << i << ": ";
            for (int neighbor : adj_list[i]) {
                std::cout << neighbor << " ";
            }
            std::cout << std::endl;
        }
    }
};

int main() {
    AdjacencyListGraph g(5);
    g.addEdge(0, 1);
    g.addEdge(0, 4);
    g.addEdge(1, 2);
    g.addEdge(1, 3);
    g.addEdge(1, 4);
    g.addEdge(2, 3);
    g.addEdge(3, 4);

    std::cout << "Adjacency list:" << std::endl;
    g.print();

    std::cout << "\nNeighbors of vertex 1: ";
    for (int v : g.getNeighbors(1)) std::cout << v << " ";
    std::cout << std::endl;

    return 0;
}
```

1.3 Graph Traversal: Breadth-First Search (BFS)
---------------------------------------

BFS uses a queue and traverses "level by level"; it can be used to find shortest paths (in unweighted graphs).

```cpp
#include <iostream>
#include <vector>
#include <list>
#include <queue>

class Graph {
private:
    int vertices;
    std::vector<std::list<int>> adj_list;

public:
    Graph(int v) : vertices(v), adj_list(v) {}

    void addEdge(int u, int v, bool directed = false) {
        adj_list[u].push_back(v);
        if (!directed) adj_list[v].push_back(u);
    }

    // BFS traversal
    void bfs(int start) const {
        std::vector<bool> visited(vertices, false);
        std::queue<int> q;

        visited[start] = true;
        q.push(start);

        std::cout << "BFS starting from " << start << ": ";
        while (!q.empty()) {
            int current = q.front();
            q.pop();
            std::cout << current << " ";

            for (int neighbor : adj_list[current]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    q.push(neighbor);
                }
            }
        }
        std::cout << std::endl;
    }

    // Shortest path with BFS (unweighted graph)
    std::vector<int> shortestPath(int start, int end) const {
        std::vector<bool> visited(vertices, false);
        std::vector<int> parent(vertices, -1);
        std::queue<int> q;

        visited[start] = true;
        q.push(start);

        while (!q.empty()) {
            int current = q.front();
            q.pop();

            if (current == end) break;

            for (int neighbor : adj_list[current]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    parent[neighbor] = current;
                    q.push(neighbor);
                }
            }
        }

        // Reconstruct the path
        std::vector<int> path;
        if (!visited[end]) return path;  // unreachable

        for (int v = end; v != -1; v = parent[v]) {
            path.push_back(v);
        }
        std::reverse(path.begin(), path.end());
        return path;
    }
};

int main() {
    Graph g(6);
    g.addEdge(0, 1);
    g.addEdge(0, 2);
    g.addEdge(1, 3);
    g.addEdge(2, 3);
    g.addEdge(2, 4);
    g.addEdge(3, 4);
    g.addEdge(3, 5);
    g.addEdge(4, 5);

    g.bfs(0);

    auto path = g.shortestPath(0, 5);
    std::cout << "Shortest path 0->5: ";
    for (int v : path) std::cout << v << " ";
    std::cout << std::endl;

    return 0;
}
```

1.4 Graph Traversal: Depth-First Search (DFS)
---------------------------------------

DFS uses a stack (recursion or an explicit stack): it follows one path as far as it goes and then backtracks.

```cpp
#include <iostream>
#include <vector>
#include <list>
#include <stack>

class GraphDFS {
private:
    int vertices;
    std::vector<std::list<int>> adj_list;

    void dfsRecursive(int v, std::vector<bool>& visited) const {
        visited[v] = true;
        std::cout << v << " ";

        for (int neighbor : adj_list[v]) {
            if (!visited[neighbor]) {
                dfsRecursive(neighbor, visited);
            }
        }
    }

public:
    GraphDFS(int v) : vertices(v), adj_list(v) {}

    void addEdge(int u, int v, bool directed = false) {
        adj_list[u].push_back(v);
        if (!directed) adj_list[v].push_back(u);
    }

    // Recursive DFS
    void dfsRecursive(int start) const {
        std::vector<bool> visited(vertices, false);
        std::cout << "DFS (recursive) starting from " << start << ": ";
        dfsRecursive(start, visited);
        std::cout << std::endl;
    }

    // Iterative DFS (using a stack)
    void dfsIterative(int start) const {
        std::vector<bool> visited(vertices, false);
        std::stack<int> stk;

        stk.push(start);

        std::cout << "DFS (iterative) starting from " << start << ": ";
        while (!stk.empty()) {
            int v = stk.top();
            stk.pop();

            if (visited[v]) continue;
            visited[v] = true;
            std::cout << v << " ";

            // Push in reverse order to keep the same traversal order as the recursive version
            for (auto it = adj_list[v].rbegin(); it != adj_list[v].rend(); ++it) {
                if (!visited[*it]) {
                    stk.push(*it);
                }
            }
        }
        std::cout << std::endl;
    }

    // Detect whether there is a cycle (directed graph)
    bool hasCycle() const {
        std::vector<bool> visited(vertices, false);
        std::vector<bool> on_path(vertices, false);

        for (int i = 0; i < vertices; ++i) {
            if (!visited[i]) {
                if (dfsCycleDetect(i, visited, on_path)) return true;
            }
        }
        return false;
    }

private:
    bool dfsCycleDetect(int v, std::vector<bool>& visited,
                        std::vector<bool>& on_path) const {
        visited[v] = true;
        on_path[v] = true;

        for (int neighbor : adj_list[v]) {
            if (!visited[neighbor]) {
                if (dfsCycleDetect(neighbor, visited, on_path)) return true;
            } else if (on_path[neighbor]) {
                return true;  // cycle found
            }
        }

        on_path[v] = false;
        return false;
    }
};

int main() {
    GraphDFS g(5);
    g.addEdge(0, 1, true);
    g.addEdge(1, 2, true);
    g.addEdge(2, 0, true);   // forms the cycle 0->1->2->0
    g.addEdge(1, 3, true);
    g.addEdge(3, 4, true);

    g.dfsRecursive(0);
    g.dfsIterative(0);

    std::cout << "Has a cycle? " << g.hasCycle() << std::endl;

    return 0;
}
```

2.1 Weighted Graphs and Shortest Paths (Dijkstra's Algorithm)
-----------------------------------------

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <limits>
#include <list>

class WeightedGraph {
private:
    int vertices;
    std::vector<std::list<std::pair<int, int>>> adj_list;  // {neighbor, weight}

public:
    WeightedGraph(int v) : vertices(v), adj_list(v) {}

    void addEdge(int u, int v, int weight, bool directed = false) {
        adj_list[u].push_back({v, weight});
        if (!directed) adj_list[v].push_back({u, weight});
    }

    // Dijkstra's shortest path algorithm
    std::vector<int> dijkstra(int start) const {
        std::vector<int> dist(vertices, std::numeric_limits<int>::max());
        std::vector<bool> visited(vertices, false);

        // Min-heap: {distance, vertex}
        std::priority_queue<std::pair<int, int>,
                            std::vector<std::pair<int, int>>,
                            std::greater<std::pair<int, int>>> pq;

        dist[start] = 0;
        pq.push({0, start});

        while (!pq.empty()) {
            int u = pq.top().second;
            pq.pop();

            if (visited[u]) continue;
            visited[u] = true;

            for (const auto& [v, w] : adj_list[u]) {
                if (!visited[v] && dist[u] + w < dist[v]) {
                    dist[v] = dist[u] + w;
                    pq.push({dist[v], v});
                }
            }
        }

        return dist;
    }

    // Bellman-Ford algorithm (handles negative edge weights)
    std::vector<int> bellmanFord(int start) const {
        std::vector<int> dist(vertices, std::numeric_limits<int>::max());
        dist[start] = 0;

        // Relax V-1 times
        for (int i = 0; i < vertices - 1; ++i) {
            for (int u = 0; u < vertices; ++u) {
                for (const auto& [v, w] : adj_list[u]) {
                    if (dist[u] != std::numeric_limits<int>::max() &&
                        dist[u] + w < dist[v]) {
                        dist[v] = dist[u] + w;
                    }
                }
            }
        }

        // Check for negative-weight cycles
        for (int u = 0; u < vertices; ++u) {
            for (const auto& [v, w] : adj_list[u]) {
                if (dist[u] != std::numeric_limits<int>::max() &&
                    dist[u] + w < dist[v]) {
                    std::cout << "The graph contains a negative-weight cycle!" << std::endl;
                    return {};
                }
            }
        }

        return dist;
    }
};

int main() {
    WeightedGraph g(6);
    g.addEdge(0, 1, 4);
    g.addEdge(0, 2, 2);
    g.addEdge(1, 2, 1);
    g.addEdge(1, 3, 5);
    g.addEdge(2, 3, 8);
    g.addEdge(2, 4, 10);
    g.addEdge(3, 4, 2);
    g.addEdge(3, 5, 6);
    g.addEdge(4, 5, 3);

    auto dist = g.dijkstra(0);

    std::cout << "Shortest distances from 0 to every vertex:" << std::endl;
    for (int i = 0; i < dist.size(); ++i) {
        std::cout << "  0 -> " << i << ": "
                  << (dist[i] == std::numeric_limits<int>::max() ? "INF" :
                      std::to_string(dist[i])) << std::endl;
    }

    return 0;
}
```

2.2 Minimum Spanning Trees (Prim's Algorithm & Kruskal's Algorithm)
---------------------------------------------

```cpp
#include <iostream>
#include <vector>
#include <list>
#include <queue>
#include <algorithm>

class MST {
private:
    // Union-Find (disjoint set)
    struct UnionFind {
        std::vector<int> parent, rank;
        UnionFind(int n) : parent(n), rank(n, 0) {
            for (int i = 0; i < n; ++i) parent[i] = i;
        }

        int find(int x) {
            if (parent[x] != x)
                parent[x] = find(parent[x]);
            return parent[x];
        }

        bool unite(int x, int y) {
            int px = find(x), py = find(y);
            if (px == py) return false;
            if (rank[px] < rank[py]) std::swap(px, py);
            parent[py] = px;
            if (rank[px] == rank[py]) ++rank[px];
            return true;
        }
    };

public:
    // Kruskal's algorithm
    static std::vector<std::tuple<int, int, int>>
    kruskalMST(int vertices, std::vector<std::tuple<int, int, int>>& edges) {
        // Sort by weight
        std::sort(edges.begin(), edges.end(),
                  [](const auto& a, const auto& b) {
                      return std::get<2>(a) < std::get<2>(b);
                  });

        UnionFind uf(vertices);
        std::vector<std::tuple<int, int, int>> mst;

        for (const auto& [u, v, w] : edges) {
            if (uf.unite(u, v)) {
                mst.push_back({u, v, w});
                if (mst.size() == (size_t)vertices - 1) break;
            }
        }

        return mst;
    }

    // Prim's algorithm
    static std::vector<std::pair<int, int>>
    primMST(const std::vector<std::list<std::pair<int, int>>>& adj_list) {
        int vertices = adj_list.size();
        std::vector<bool> in_mst(vertices, false);
        std::vector<int> key(vertices, std::numeric_limits<int>::max());
        std::vector<int> parent(vertices, -1);

        // Min-heap: {weight, vertex}
        std::priority_queue<std::pair<int, int>,
                            std::vector<std::pair<int, int>>,
                            std::greater<std::pair<int, int>>> pq;

        key[0] = 0;
        pq.push({0, 0});

        while (!pq.empty()) {
            int u = pq.top().second;
            pq.pop();

            if (in_mst[u]) continue;
            in_mst[u] = true;

            for (const auto& [v, w] : adj_list[u]) {
                if (!in_mst[v] && w < key[v]) {
                    key[v] = w;
                    parent[v] = u;
                    pq.push({key[v], v});
                }
            }
        }

        // Build the list of MST edges
        std::vector<std::pair<int, int>> mst;
        for (int i = 1; i < vertices; ++i) {
            if (parent[i] != -1) {
                mst.push_back({parent[i], i});
            }
        }
        return mst;
    }
};

int main() {
    int V = 5;

    // Edges: (u, v, w)
    std::vector<std::tuple<int, int, int>> edges = {
        {0, 1, 2}, {0, 3, 6}, {1, 2, 3},
        {1, 3, 8}, {1, 4, 5}, {2, 4, 7}, {3, 4, 9}
    };

    auto mst = MST::kruskalMST(V, edges);
    int total_weight = 0;

    std::cout << "Kruskal minimum spanning tree:" << std::endl;
    for (const auto& [u, v, w] : mst) {
        std::cout << "  " << u << " - " << v << "  weight: " << w << std::endl;
        total_weight += w;
    }
    std::cout << "Total weight: " << total_weight << std::endl;

    return 0;
}
```

2.3 Topological Sort
-------------------------------------

```cpp
#include <iostream>
#include <vector>
#include <list>
#include <queue>

class TopologicalSort {
private:
    int vertices;
    std::vector<std::list<int>> adj_list;

public:
    TopologicalSort(int v) : vertices(v), adj_list(v) {}

    void addEdge(int u, int v) {
        adj_list[u].push_back(v);
    }

    // Kahn's algorithm (based on in-degrees)
    std::vector<int> topologicalSort() {
        std::vector<int> in_degree(vertices, 0);

        for (int u = 0; u < vertices; ++u) {
            for (int v : adj_list[u]) {
                ++in_degree[v];
            }
        }

        std::queue<int> q;
        for (int i = 0; i < vertices; ++i) {
            if (in_degree[i] == 0) q.push(i);
        }

        std::vector<int> result;
        while (!q.empty()) {
            int u = q.front();
            q.pop();
            result.push_back(u);

            for (int v : adj_list[u]) {
                if (--in_degree[v] == 0) {
                    q.push(v);
                }
            }
        }

        if (result.size() != (size_t)vertices) {
            std::cout << "The graph contains a cycle; topological sort is impossible!" << std::endl;
            return {};
        }

        return result;
    }
};

int main() {
    // Course dependency graph
    TopologicalSort ts(6);
    ts.addEdge(0, 2);  // C++ Basics -> Data Structures
    ts.addEdge(0, 3);  // C++ Basics -> Algorithms
    ts.addEdge(1, 2);  // Mathematics -> Data Structures
    ts.addEdge(2, 3);  // Data Structures -> Algorithms
    ts.addEdge(2, 4);  // Data Structures -> Databases
    ts.addEdge(3, 5);  // Algorithms -> Computer Graphics
    ts.addEdge(4, 5);  // Databases -> Computer Graphics

    auto order = ts.topologicalSort();

    std::cout << "Study order: ";
    for (int course : order) {
        std::cout << "C" << course << " ";
    }
    std::cout << std::endl;

    return 0;
}
```

Example 1: Influence Analysis in a Social Network
------------------------------------

```cpp
#include <iostream>
#include <vector>
#include <list>
#include <queue>
#include <unordered_map>
#include <string>
#include <algorithm>

class SocialNetwork {
private:
    std::unordered_map<std::string, int> name_to_id;
    std::vector<std::string> id_to_name;
    std::vector<std::list<int>> friends;

    int getOrCreateId(const std::string& name) {
        auto it = name_to_id.find(name);
        if (it != name_to_id.end()) return it->second;
        int id = id_to_name.size();
        name_to_id[name] = id;
        id_to_name.push_back(name);
        friends.emplace_back();
        return id;
    }

public:
    void addFriendship(const std::string& a, const std::string& b) {
        int id_a = getOrCreateId(a);
        int id_b = getOrCreateId(b);
        friends[id_a].push_back(id_b);
        friends[id_b].push_back(id_a);
    }

    // Compute influence (influence = followers + followers of followers * 0.5)
    std::vector<std::pair<std::string, double>> getInfluenceRank() {
        int n = id_to_name.size();
        std::vector<double> influence(n, 0);

        for (int i = 0; i < n; ++i) {
            influence[i] = friends[i].size();  // direct followers

            // Indirect followers (second-degree connections)
            for (int f : friends[i]) {
                influence[i] += friends[f].size() * 0.5;
            }
        }

        std::vector<std::pair<std::string, double>> result;
        for (int i = 0; i < n; ++i) {
            result.push_back({id_to_name[i], influence[i]});
        }

        std::sort(result.begin(), result.end(),
                  [](const auto& a, const auto& b) {
                      return a.second > b.second;
                  });

        return result;
    }

    // Find the shortest connection path (six degrees of separation)
    std::vector<std::string> findConnectionPath(const std::string& from,
                                                 const std::string& to) {
        if (name_to_id.find(from) == name_to_id.end() ||
            name_to_id.find(to) == name_to_id.end()) {
            return {};
        }

        int start = name_to_id[from];
        int end = name_to_id[to];
        int n = id_to_name.size();

        std::vector<bool> visited(n, false);
        std::vector<int> parent(n, -1);
        std::queue<int> q;

        visited[start] = true;
        q.push(start);

        while (!q.empty()) {
            int cur = q.front();
            q.pop();

            if (cur == end) break;

            for (int neighbor : friends[cur]) {
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    parent[neighbor] = cur;
                    q.push(neighbor);
                }
            }
        }

        if (!visited[end]) return {};

        std::vector<std::string> path;
        for (int v = end; v != -1; v = parent[v]) {
            path.push_back(id_to_name[v]);
        }
        std::reverse(path.begin(), path.end());
        return path;
    }
};

int main() {
    SocialNetwork sn;

    sn.addFriendship("Alice", "Bob");
    sn.addFriendship("Alice", "Charlie");
    sn.addFriendship("Bob", "David");
    sn.addFriendship("Charlie", "David");
    sn.addFriendship("David", "Eve");
    sn.addFriendship("Eve", "Frank");
    sn.addFriendship("Bob", "Grace");

    std::cout << "Influence ranking:" << std::endl;
    for (const auto& [name, score] : sn.getInfluenceRank()) {
        std::cout << "  " << name << ": " << score << std::endl;
    }

    std::cout << "\nConnection path from Alice to Frank: ";
    for (const auto& name : sn.findConnectionPath("Alice", "Frank")) {
        std::cout << name << " -> ";
    }
    std::cout << "End" << std::endl;

    return 0;
}
```

```cpp
#include <iostream>

#include <queue>

class MapNavigator {
private:
    struct Location {
        std::string name;
        double lat, lon;
    };

    std::vector<Location> locations;
    std::vector<std::list<std::pair<int, double>>> roads;  // {neighbor, distance(km)}
    std::unordered_map<std::string, int> name_to_id;

public:
    int addLocation(const std::string& name, double lat, double lon) {
        int id = locations.size();
        name_to_id[name] = id;
        locations.push_back({name, lat, lon});
        roads.emplace_back();
        return id;
    }

    void addRoad(const std::string& a, const std::string& b, double distance) {
        int id_a = name_to_id[a];
        int id_b = name_to_id[b];
        roads[id_a].push_back({id_b, distance});
        roads[id_b].push_back({id_a, distance});
    }

    std::pair<std::vector<std::string>, double>
    findShortestPath(const std::string& from, const std::string& to) {
        int start = name_to_id[from];
        int end = name_to_id[to];
        int n = locations.size();

        std::vector<double> dist(n, std::numeric_limits<double>::max());
        std::vector<int> parent(n, -1);
        std::vector<bool> visited(n, false);

        std::priority_queue<std::pair<double, int>,
                            std::vector<std::pair<double, int>>,
                            std::greater<std::pair<double, int>>> pq;

        dist[start] = 0;
        pq.push({0, start});

        while (!pq.empty()) {
            int u = pq.top().second;
            pq.pop();

            if (visited[u]) continue;
            visited[u] = true;

            if (u == end) break;

            for (const auto& [v, w] : roads[u]) {
                if (!visited[v] && dist[u] + w < dist[v]) {
                    dist[v] = dist[u] + w;
                    parent[v] = u;
                    pq.push({dist[v], v});
                }
            }
        }

        std::vector<std::string> path;
        if (dist[end] == std::numeric_limits<double>::max()) {
            return {path, -1};
        }

        for (int v = end; v != -1; v = parent[v]) {
            path.push_back(locations[v].name);
        }
        std::reverse(path.begin(), path.end());

        return {path, dist[end]};
    }
};

int main() {
    MapNavigator nav;

    nav.addLocation("Beijing", 39.9, 116.4);
    nav.addLocation("Tianjin", 39.1, 117.2);
    nav.addLocation("Shanghai", 31.2, 121.5);
    nav.addLocation("Nanjing", 32.1, 118.8);
    nav.addLocation("Jinan", 36.7, 117.0);

    nav.addRoad("Beijing", "Tianjin", 120);
    nav.addRoad("Beijing", "Jinan", 400);
    nav.addRoad("Tianjin", "Jinan", 320);
    nav.addRoad("Jinan", "Nanjing", 600);
    nav.addRoad("Nanjing", "Shanghai", 300);
    nav.addRoad("Tianjin", "Shanghai", 1000);

    auto [path, distance] = nav.findShortestPath("Beijing", "Shanghai");

    std::cout << "Shortest route from Beijing to Shanghai: ";
    for (size_t i = 0; i < path.size(); ++i) {
        std::cout << path[i];
        if (i < path.size() - 1) std::cout << " -> ";
    }
    std::cout << std::endl;
    std::cout << "Total distance: " << distance << " km" << std::endl;

    return 0;
}
```
